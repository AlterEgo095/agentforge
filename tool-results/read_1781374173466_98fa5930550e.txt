   169→  // ─── Core Pipeline ──────────────────────────────────────
   170→
   171→  protected async generate(
   172→    prompt: string,
   173→    context: Record<string, unknown>,
   174→    sessionId: string,
   175→  ): Promise<MarketingOutput> {
   176→    // Phase 1: Parse Campaign Brief
   177→    this.emitEvent('marketing_analysis', sessionId, { status: 'started' });
   178→    const brief = this.parseCampaignBrief(prompt, context);
   179→    this.emitEvent('marketing_analysis', sessionId, {
   180→      status: 'completed',
   181→      industry: brief.industry,
   182→      product: brief.product,
   183→      goals: brief.goals,
   184→    });
   185→
   186→    // Phase 2: Build Audience Profile
   187→    const audience = this.buildAudienceProfile(brief);
   188→    this.emitEvent('marketing_analysis', sessionId, {
   189→      status: 'audience_profiled',
   190→      segmentSize: audience.segmentSize,
   191→      preferredChannels: audience.preferredChannels,
   192→    });
   193→
   194→    // Phase 3: Develop Marketing Strategy
   195→    this.emitEvent('marketing_strategy', sessionId, { status: 'started' });
   196→    const strategy = this.developStrategy(brief, audience);
   197→    this.emitEvent('marketing_strategy', sessionId, {
   198→      status: 'completed',
   199→      positioning: strategy.positioning,
   200→      channels: strategy.channels.map(c => c.channel),
   201→      differentiators: strategy.differentiators,
   202→    });
   203→
   204→    // Phase 4: Generate Content Pieces
   205→    this.emitEvent('marketing_content', sessionId, { status: 'started' });
   206→    const channelNames = strategy.channels.map(c => c.channel);
   207→    const content = this.generateContent(strategy, channelNames);
   208→    this.emitEvent('marketing_content', sessionId, {
   209→      status: 'completed',
   210→      contentCount: content.length,
   211→      contentTypes: content.map(c => c.type),
   212→    });
   213→
   214→    // Phase 5: Create Campaign Calendar
   215→    this.emitEvent('marketing_calendar', sessionId, { status: 'started' });
   216→    const durationWeeks = brief.durationWeeks ?? 8;
   217→    const calendar = this.createCalendar(content, durationWeeks);
   218→    this.emitEvent('marketing_calendar', sessionId, {
   219→      status: 'completed',
   220→      weeks: calendar.length,
   221→      totalItems: calendar.reduce((sum, w) => sum + w.items.length, 0),
   222→    });
   223→
   224→    // Phase 6: Define KPIs
   225→    const kpis = this.defineKPIs(brief.goals.join(', '), channelNames);
   226→
   227→    // Phase 7: Competitive Analysis
   228→    const competitiveAnalysis = this.analyzeCompetition(brief.industry, brief.product);
   229→
   230→    // Phase 8: Budget Allocation
   231→    const totalBudget = brief.budget ?? 10000;
   232→    const budget = this.allocateBudget(strategy.channels, totalBudget);
   233→
   234→    // Phase 9: Calculate ROI Projection & Estimated Reach
   235→    const roiProjection = this.projectROI(strategy, budget);
   236→    const estimatedReach = this.estimateReach(audience, strategy, durationWeeks);
   237→
   238→    // Assemble output
   239→    const output: MarketingOutput = {
   240→      campaignName: this.generateCampaignName(brief),
   241→      objective: brief.goals[0] ?? 'Increase brand awareness and drive conversions',
   242→      targetAudience: audience,
   243→      strategy,
   244→      content,
   245→      calendar,
   246→      kpis,
   247→      competitiveAnalysis,
   248→      budget,
   249→      metadata: {
   250→        channels: channelNames,
   251→        duration: durationWeeks,
   252→        estimatedReach,
   253→        roiProjection,
   254→      },
   255→    };
   256→
   257→    this.emitEvent('marketing_package', sessionId, {
   258→      status: 'completed',
   259→      campaignName: output.campaignName,
   260→      contentPieces: content.length,
   261→      calendarWeeks: calendar.length,
   262→      kpis: kpis.length,
   263→    });
   264→
   265→    return output;
   266→  }

   268→  // ─── Validation ─────────────────────────────────────────
   269→
   270→  protected async validateOutput(output: MarketingOutput, prompt: string): Promise<QualityReport> {
   271→    const dimensions: Record<string, number> = {};
   272→    const suggestions: string[] = [];
   273→
   274→    // Strategy Quality (0-1): Validates strategy depth and coherence
   275→    dimensions['Strategy Quality'] = this.scoreStrategyQuality(output, suggestions);
   276→
   277→    // Content Quality (0-1): Validates content variety and CTAs
   278→    dimensions['Content Quality'] = this.scoreContentQuality(output, suggestions);
   279→
   280→    // Market Fit (0-1): Validates audience alignment
   281→    dimensions['Market Fit'] = this.scoreMarketFit(output, suggestions);
   282→
   283→    // Actionability (0-1): Validates calendar, KPIs, budget
   284→    dimensions['Actionability'] = this.scoreActionability(output, suggestions);
   285→
   286→    // ROI Potential (0-1): Validates budget allocation and projections
   287→    dimensions['ROI Potential'] = this.scoreROIPotential(output, suggestions);
   288→
   289→    const overall = this.computeWeightedScore(dimensions);
   290→    const passed = this.isQualityPassing(overall);
   291→
   292→    return { overall, dimensions, passed, suggestions };
   293→  }

   332→  private parseCampaignBrief(prompt: string, context?: Record<string, unknown>): CampaignBrief {
   333→    // Check if context provides a brief directly
   334→    if (context?.brief && typeof context.brief === 'object') {
   335→      const b = context.brief as CampaignBrief;
   336→      if (b.industry && b.product) {
   337→        return b;
   338→      }
   339→    }
   340→
   341→    // Check for structured JSON brief in the prompt
   342→    const jsonMatch = prompt.match(/```json\n([\s\S]*?)```/);
   343→    if (jsonMatch) {
   344→      try {
   345→        const parsed = JSON.parse(jsonMatch[1]);
   346→        if (parsed.industry && parsed.product) {
   347→          return {
   348→            industry: parsed.industry,
   349→            product: parsed.product,
   350→            goals: parsed.goals ?? ['Increase brand awareness'],
   351→            budget: parsed.budget,
   352→            durationWeeks: parsed.durationWeeks,
   353→            targetMarket: parsed.targetMarket,
   354→            competitors: parsed.competitors,
   355→          };
   356→        }
   357→      } catch {
   358→        // Fall through to NLP-style extraction
   359→      }
   360→    }
   361→
   362→    // Extract from natural language prompt
   363→    const lower = prompt.toLowerCase();
   364→
   365→    const industry = this.extractKeyword(
   366→      prompt,
   367→      [
   368→        'saas', 'e-commerce', 'fintech', 'healthcare', 'education', 'real estate',
   369→        'food & beverage', 'travel', 'fitness', 'fashion', 'technology', 'automotive',
   370→        'finance', 'entertainment', 'beauty', 'sports', 'non-profit', 'b2b', 'b2c',
   371→      ],
   372→      'technology',
   373→    );
   374→
   375→    const product = this.extractProductFromPrompt(prompt);
   376→
   377→    const goals: string[] = [];
   378→    if (/awareness|brand recognition|visibility/i.test(prompt)) {
   379→      goals.push('Increase brand awareness');
   380→    }
   381→    if (/lead|conversion|sign.?up|registr/i.test(prompt)) {
   382→      goals.push('Generate qualified leads');
   383→    }
   384→    if (/revenue|sales|purchase|buy/i.test(prompt)) {
   385→      goals.push('Drive revenue growth');
   386→    }
   387→    if (/engagement|interact|communit/i.test(prompt)) {
   388→      goals.push('Boost audience engagement');
   389→    }
   390→    if (/retention|loyalt|churn|repeat/i.test(prompt)) {
   391→      goals.push('Improve customer retention');
   392→    }
   393→    if (goals.length === 0) {
   394→      goals.push('Increase brand awareness', 'Generate qualified leads');
   395→    }
   396→
   397→    // Extract budget if mentioned
   398→    const budgetMatch = prompt.match(/\$?([\d,]+)\s*(?:dollar|budget|spend|invest)/i);
   399→    const budget = budgetMatch ? parseInt(budgetMatch[1].replace(/,/g, ''), 10) : undefined;
   400→
   401→    // Extract duration if mentioned
   402→    const durationMatch = prompt.match(/(\d+)\s*week/i);
   403→    const durationWeeks = durationMatch ? parseInt(durationMatch[1], 10) : undefined;
   404→
   405→    // Extract competitors if mentioned
   406→    const competitors: string[] = [];
   407→    const competitorMatch = prompt.match(/(?:competitors?|vs\.?|against)\s*:?\s*(.+)/i);
   408→    if (competitorMatch) {
   409→      competitors.push(
   410→        ...competitorMatch[1].split(/[,;]/).map(c => c.trim()).filter(Boolean),
   411→      );
   412→    }
   413→
   414→    return {
   415→      industry,
   416→      product,
   417→      goals,
   418→      budget,
   419→      durationWeeks: durationWeeks ?? this.inferDuration(goals),
   420→      competitors,
   421→    };
   422→  }
   423→
   424→  private extractProductFromPrompt(prompt: string): string {
   425→    // Try patterns like "for [Product]", "promote [Product]", "campaign for [Product]"
   426→    const patterns = [
   427→      /(?:for|promote|market|launch|campaign\s+for)\s+["']?([A-Z][\w\s&-]{2,40})["']?/i,
   428→      /["']([^"']{3,50})["']/,
   429→    ];
   430→
   431→    for (const pattern of patterns) {
   432→      const match = prompt.match(pattern);
   433→      if (match) {
   434→        return match[1].trim();
   435→      }
   436→    }
   437→
   438→    // Fallback: use first capitalized phrase
   439→    const capMatch = prompt.match(/\b([A-Z][a-z]+(?:\s+[A-Z][a-z]+)*)\b/);
   440→    return capMatch ? capMatch[1] : 'Product';
   441→  }
   442→
   443→  private extractKeyword(prompt: string, keywords: string[], fallback: string): string {
   444→    const lower = prompt.toLowerCase();
   445→    for (const keyword of keywords) {
   446→      if (lower.includes(keyword.toLowerCase())) {
   447→        return keyword;
   448→      }
   449→    }
   450→    return fallback;
   451→  }
   452→
   453→  private inferDuration(goals: string[]): number {
   454→    if (goals.some(g => /awareness|brand/i.test(g))) return 12;
   455→    if (goals.some(g => /lead|conversion/i.test(g))) return 8;
   456→    if (goals.some(g => /retention|loyalt/i.test(g))) return 6;
   457→    return 8;
   458→  }

   597→        budgetPct: 0.08,
   598→        expectedROI: 3.5,
   599→      },
   600→    };
   601→
   602→    const strategies: ChannelStrategy[] = preferredChannels
   603→      .filter(ch => channelTactics[ch])
   604→      .map(ch => ({
   605→        channel: ch as ChannelType,
   606→        tactics: channelTactics[ch]!.tactics,
   607→        budget: Math.round(channelTactics[ch]!.budgetPct * 100),
   608→        expectedROI: channelTactics[ch]!.expectedROI,

   611→    // Ensure at least 3 channels
   612→    if (strategies.length < 3) {
   613→      const defaultChannels: ChannelType[] = ['social-media', 'content', 'email'];
   614→      for (const ch of defaultChannels) {
   615→        if (!strategies.some(s => s.channel === ch) && channelTactics[ch]) {
   616→          strategies.push({
   617→            channel: ch,
   618→            tactics: channelTactics[ch]!.tactics,
   619→            budget: Math.round(channelTactics[ch]!.budgetPct * 100),
   620→            expectedROI: channelTactics[ch]!.expectedROI,
   621→          });
   622→        }
   623→      }
   624→    }
   625→
   626→    // Normalize budget percentages to sum to 100
   627→    const total = strategies.reduce((sum, s) => sum + s.budget, 0);
   628→    if (total > 0) {
   629→      const scale = 100 / total;
   630→      strategies.forEach(s => {
   631→        s.budget = Math.round(s.budget * scale);
   632→      });
   633→      // Fix rounding to exactly 100
   634→      const newTotal = strategies.reduce((sum, s) => sum + s.budget, 0);
   635→      if (newTotal !== 100 && strategies.length > 0) {
   636→        strategies[0].budget += 100 - newTotal;
   637→      }
   638→    }
   639→
   640→    return strategies;
   641→  }

   834→
   835→  private createCalendar(content: MarketingContent[], durationWeeks: number): CalendarEntry[] {
   836→    const calendar: CalendarEntry[] = [];
   837→    const startDate = new Date();
   838→    startDate.setDate(startDate.getDate() + 7); // Start next week
   839→
   840→    // Distribute content across weeks
   841→    const contentPerWeek = Math.ceil(content.length / durationWeeks);
   842→    let contentIndex = 0;
   843→
   844→    for (let week = 1; week <= durationWeeks; week++) {
   845→      const items: CalendarItem[] = [];
   846→      const weekStart = new Date(startDate);
   847→      weekStart.setDate(weekStart.getDate() + (week - 1) * 7);
   848→
   849→      const weekContent = content.slice(contentIndex, contentIndex + contentPerWeek);
   850→      contentIndex += contentPerWeek;
   851→
   852→      for (let i = 0; i < weekContent.length; i++) {
   853→        const item = weekContent[i]!;
   854→        const dayOffset = Math.min(i, 4); // Spread across weekdays (0=Mon, 4=Fri)
   855→        const scheduledDate = new Date(weekStart);
   856→        scheduledDate.setDate(scheduledDate.getDate() + dayOffset);
   857→
   858→        items.push({
   859→          contentId: item.id,
   860→          date: scheduledDate.toISOString().split('T')[0]!,
   861→          channel: item.channel,
   862→          action: this.inferAction(item.type),
   863→        });
   864→      }
   865→
   866→      calendar.push({ week, items });
   867→    }
   868→
   869→    return calendar;
   870→  }

  1004→  // ─── Helper: ROI Projection ─────────────────────────────
  1005→
  1006→  private projectROI(strategy: MarketingStrategy, budget: BudgetAllocation): number {
  1007→    if (budget.total === 0) return 0;
  1008→
  1009→    const weightedROI = strategy.channels.reduce((sum, ch) => {
  1010→      const allocation = budget.breakdown.find(b => b.channel === ch.channel);
  1011→      const share = allocation ? allocation.amount / budget.total : 0;
  1012→      return sum + ch.expectedROI * share;
  1013→    }, 0);
  1014→
  1015→    return Math.round(weightedROI * 100) / 100;
  1016→  }

  1018→  private estimateReach(audience: AudienceProfile, strategy: MarketingStrategy, durationWeeks: number): number {
  1019→    // Base reach from segment size, modulated by channel mix and duration
  1020→    const channelMultipliers: Record<string, number> = {
  1021→      'social-media': 0.15,
  1022→      email: 0.08,
  1023→      content: 0.10,
  1024→      ppc: 0.12,
  1025→      seo: 0.06,
  1026→      influencer: 0.09,
  1027→      events: 0.03,
  1028→      pr: 0.05,
  1029→    };
  1030→
  1031→    const totalMultiplier = strategy.channels.reduce(
  1032→      (sum, ch) => sum + (channelMultipliers[ch.channel] ?? 0.05),
  1033→      0,
  1034→    );
  1035→
  1036→    const durationFactor = Math.min(durationWeeks / 8, 1.5);
  1037→    return Math.round(audience.segmentSize * totalMultiplier * durationFactor);
  1038→  }

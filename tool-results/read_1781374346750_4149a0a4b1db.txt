    13→// ─── Output & Supporting Types ─────────────────────────────
    14→
    15→export interface RechercheSection {
    16→  id: string;
    17→  heading: string;
    18→  content: string; // 150+ words per section
    19→  sourceIds: string[];
    20→  confidence: number; // 0-1
    21→  type: 'overview' | 'methodology' | 'findings' | 'analysis' | 'conclusion' | 'gaps' | 'recommendations';
    22→}

    24→export type RechercheSourceType = 'academic' | 'news' | 'blog' | 'documentation' | 'social' | 'government' | 'other';
    25→export type RechercheSourceCredibility = 'high' | 'medium' | 'low';

    27→export interface RechercheSource {
    28→  id: string;
    29→  title: string;

    35→  publishedDate?: string;
    36→  keyClaims: string[];
    37→}
    38→
    39→export type RechercheFindingConsensus = 'strong' | 'moderate' | 'weak' | 'disputed';
    40→export type RechercheFindingType = 'fact' | 'opinion' | 'trend' | 'statistic' | 'claim';

    42→export interface RechercheFinding {
    43→  id: string;
    44→  statement: string;
    45→  evidence: string[];
    46→  confidence: number; // 0-1
    47→  sourceIds: string[];

   120→  // ─── Core Pipeline: generate() ─────────────────────────
   121→
   122→  protected async generate(
   123→    prompt: string,
   124→    context: Record<string, unknown>,
   125→    sessionId: string,
   126→  ): Promise<RechercheOutput> {
   127→    // ── Step 1: Parse & Classify the Query ──
   128→    this.emitEvent('research_query', sessionId, { query: prompt });
   129→    const classification = this.classifyQuery(prompt);
   130→
   131→    // ── Step 2: Plan Search Strategy ──
   132→    const depth = typeof context.depth === 'number' ? context.depth : 3;
   133→    const strategy = this.planSearchStrategy(classification, depth);
   134→
   135→    // ── Step 3: Simulate Web Search (generate structured search results) ──
   136→    this.emitEvent('research_search', sessionId, {
   137→      engines: strategy.engines,
   138→      keywords: strategy.keywords,
   139→      depth: strategy.depth,
   140→    });
   141→    const sources = this.generateSources(prompt, strategy);
   142→
   143→    // ── Step 4: Extract Findings from Sources ──
   144→    this.emitEvent('research_extract', sessionId, {
   145→      sourceCount: sources.length,
   146→    });
   147→    const findings = this.extractFindings(sources);
   148→
   149→    // ── Step 5: Cross-Reference & Compare Sources ──
   150→    const comparisons = this.compareSources(sources);
   151→
   152→    // ── Step 6: Synthesize — Executive Summary & Sections ──
   153→    this.emitEvent('research_synthesize', sessionId, {
   154→      findingCount: findings.length,
   155→      comparisonCount: comparisons.length,
   156→    });
   157→    const executiveSummary = this.synthesizeSummary(prompt, classification, findings, comparisons);
   158→    const sections = this.synthesizeSections(prompt, classification, sources, findings, comparisons);
   159→
   160→    // ── Step 7: Identify Knowledge Gaps ──
   161→    const gaps = this.identifyGaps(findings, sources);
   162→
   163→    // ── Step 8: Format References in Multiple Citation Styles ──
   164→    const references = this.generateCitations(sources, 'all');
   165→
   166→    // ── Step 9: Assemble Output & Emit Report ──
   167→    const confidenceLevel = this.computeConfidenceLevel(findings, sources);
   168→
   169→    const output: RechercheOutput = {
   170→      title: this.generateTitle(prompt, classification),
   171→      query: prompt,
   172→      executiveSummary,
   173→      sections,
   174→      sources,
   175→      findings,
   176→      comparisons,
   177→      references,
   178→      classification,
   179→      gaps,
   180→      metadata: {
   181→        sourceCount: sources.length,
   182→        findingCount: findings.length,
   183→        confidenceLevel,
   184→        depthLevel: strategy.depth,
   185→        searchEnginesUsed: strategy.engines,
   186→      },
   187→    };
   188→
   189→    this.emitEvent('research_report', sessionId, {
   190→      title: output.title,
   191→      sectionCount: sections.length,
   192→      sourceCount: sources.length,
   193→      findingCount: findings.length,
   194→      confidenceLevel,
   195→    });
   196→
   197→    return output;
   198→  }

   200→  // ─── Quality Validation: validateOutput() ──────────────
   201→
   202→  protected async validateOutput(output: RechercheOutput, prompt: string): Promise<QualityReport> {
   203→    const suggestions: string[] = [];
   204→
   205→    // 1. Source Quality (0-1): Checks credibility distribution
   206→    const sourceQuality = this.scoreSourceQuality(output, suggestions);
   207→
   208→    // 2. Comprehensiveness (0-1): Checks section coverage
   209→    const comprehensiveness = this.scoreComprehensiveness(output, suggestions);
   210→
   211→    // 3. Accuracy (0-1): Checks confidence levels and evidence
   212→    const accuracy = this.scoreAccuracy(output, suggestions);
   213→
   214→    // 4. Synthesis (0-1): Validates cross-references between sources
   215→    const synthesis = this.scoreSynthesis(output, suggestions);
   216→
   217→    // 5. Presentation (0-1): Validates structure and formatting
   218→    const presentation = this.scorePresentation(output, prompt, suggestions);
   219→
   220→    const dimensions: Record<string, number> = {
   221→      'Source Quality': sourceQuality,
   222→      'Comprehensiveness': comprehensiveness,
   223→      'Accuracy': accuracy,
   224→      'Synthesis': synthesis,
   225→      'Presentation': presentation,
   226→    };
   227→
   228→    const overall = this.computeWeightedScore(dimensions);
   229→    const passed = this.isQualityPassing(overall);
   230→
   231→    return { overall, dimensions, passed, suggestions };
   232→  }

   234→  // ─── Render: renderOutput() ────────────────────────────
   235→
   236→  protected async renderOutput(output: RechercheOutput, format?: string): Promise<unknown> {
   237→    switch (format) {
   238→      case 'summary':
   239→        return {
   240→          title: output.title,
   241→          query: output.query,
   242→          executiveSummary: output.executiveSummary,
   243→          confidenceLevel: output.metadata.confidenceLevel,
   244→          sourceCount: output.metadata.sourceCount,
   245→          findingCount: output.metadata.findingCount,
   246→          gaps: output.gaps,
   247→        };
   248→
   249→      case 'markdown':
   250→        return this.renderMarkdown(output);
   251→
   252→      case 'academic':
   253→        return this.renderAcademic(output);
   254→
   255→      default:
   256→        // JSON with full research report
   257→        return output;
   258→    }
   259→  }

   261→  // ─── Helper: classifyQuery() ───────────────────────────
   262→
   263→  private classifyQuery(query: string): RechercheClassification {
   264→    const queryLower = query.toLowerCase();
   265→
   266→    // Domain detection
   267→    const domainMap: Record<string, string> = {
   268→      'ai|machine learning|neural|model|gpt|llm': 'artificial-intelligence',
   269→      'climate|carbon|emission|environment|sustainability': 'environment',
   270→      'health|medical|disease|treatment|clinical': 'healthcare',
   271→      'finance|stock|market|investment|crypto|bitcoin': 'finance',
   272→      'education|learning|school|university|curriculum': 'education',
   273→      'law|legal|regulation|compliance|policy': 'legal',
   274→      'tech|software|hardware|startup|silicon': 'technology',
   275→      'science|research|study|experiment|peer-review': 'science',
   276→      'politics|government|election|democracy|policy': 'politics',
   277→      'business|company|corporate|strategy|management': 'business',
   278→    };
   279→
   280→    let domain = 'general';
   281→    for (const [pattern, d] of Object.entries(domainMap)) {
   282→      if (new RegExp(pattern, 'i').test(queryLower)) {
   283→        domain = d;
   284→        break;
   285→      }
   286→    }
   287→
   288→    // Primary topic: use the first meaningful noun phrase from the query
   289→    const primaryTopic = query.replace(/[?!.]/g, '').trim();
   290→
   291→    // Sub-topics: extract potential sub-areas from the query
   292→    const subTopics = this.extractSubTopics(query);
   293→
   294→    // Entity extraction (basic pattern matching)
   295→    const entities = this.extractEntities(query);
   296→
   297→    // Temporal focus detection
   298→    let temporalFocus: RechercheTemporalFocus = 'current';
   299→    if (/\b(history|historical|origin|past|evolution of)\b/i.test(queryLower)) {
   300→      temporalFocus = 'historical';
   301→    } else if (/\b(recent|latest|new|2024|2025|this year)\b/i.test(queryLower)) {
   302→      temporalFocus = 'recent';
   303→    } else if (/\b(future|predict|forecast|outlook|trend|2030|2040)\b/i.test(queryLower)) {
   304→      temporalFocus = 'future';
   305→    }
   306→
   307→    // Sentiment detection
   308→    let sentiment: RechercheSentiment = 'neutral';
   309→    if (/\b(benefit|improve|advance|progress|opportunity|positive|growth)\b/i.test(queryLower)) {
   310→      sentiment = 'positive';
   311→    } else if (/\b(risk|danger|threat|crisis|negative|decline|problem)\b/i.test(queryLower)) {
   312→      sentiment = 'negative';
   313→    } else if (/\b(vs|versus|debate|controversy|compare|pros.*cons)\b/i.test(queryLower)) {
   314→      sentiment = 'mixed';
   315→    }
   316→
   317→    return {
   318→      primaryTopic,
   319→      domain,
   320→      subTopics,
   321→      entities,
   322→      temporalFocus,
   323→      sentiment,
   324→    };
   325→  }
   326→
   327→  // ─── Helper: planSearchStrategy() ──────────────────────
   328→
   329→  private planSearchStrategy(classification: RechercheClassification, depth: number): SearchStrategy {
   330→    const engines = ['duckduckgo', 'google'];
   331→    if (classification.domain === 'academic' || classification.domain === 'science') {
   332→      engines.push('scholar', 'semantic-scholar');
   333→    }
   334→    if (classification.domain === 'news' || classification.temporalFocus === 'recent' || classification.temporalFocus === 'current') {
   335→      engines.push('news');
   336→    }
   337→
   338→    const maxSources = 5 + depth * 5; // depth 1=10, 3=20, 5=30
   339→    const minRelevance = Math.max(0.3, 0.7 - depth * 0.08); // deeper research accepts lower relevance
   340→
   341→    // Generate keyword variations
   342→    const keywords = [
   343→      classification.primaryTopic,
   344→      ...classification.subTopics.slice(0, depth),
   345→      ...classification.entities.slice(0, 2).map(e => e.name),
   346→    ];
   347→
   348→    return {
   349→      keywords,
   350→      engines: Array.from(new Set(engines)),
   351→      depth,
   352→      maxSources,
   353→      minRelevance,
   354→      language: 'en',
   355→    };
   356→  }

   358→  // ─── Helper: generateSources() ─────────────────────────
   359→  // Simulates source discovery. In production, this would integrate
   360→  // with the ResearchWorker plugin's browser-based search.
   361→
   362→  private generateSources(query: string, strategy: SearchStrategy): RechercheSource[] {
   363→    const sources: RechercheSource[] = [];
   364→    const sourceCount = Math.min(strategy.maxSources, 8 + strategy.depth * 3);
   365→
   366→    const typeDistribution: RechercheSourceType[] = ['academic', 'news', 'blog', 'documentation', 'government', 'social', 'other'];
   367→    const credibilityOptions: RechercheSourceCredibility[] = ['high', 'medium', 'low'];
   368→
   369→    for (let i = 0; i < sourceCount; i++) {
   370→      const id = `src_${uuidv4().slice(0, 8)}`;
   371→      const type = typeDistribution[i % typeDistribution.length];
   372→      const credibility = credibilityOptions[i < sourceCount * 0.4 ? 0 : i < sourceCount * 0.8 ? 1 : 2];
   373→      const relevance = Math.max(strategy.minRelevance, 0.5 + Math.random() * 0.5);
   374→
   375→      sources.push({
   376→        id,
   377→        title: this.simulateSourceTitle(query, type, i),
   378→        url: this.simulateSourceUrl(query, type, i),
   379→        type,
   380→        credibility,
   381→        relevance: Math.round(relevance * 100) / 100,
   382→        snippet: this.simulateSnippet(query, type, i),
   383→        publishedDate: this.simulateDate(strategy.depth),
   384→        keyClaims: this.simulateKeyClaims(query, type, i),
   385→      });
   386→    }
   387→
   388→    // Sort by relevance descending
   389→    return sources.sort((a, b) => b.relevance - a.relevance);
   390→  }

   392→  // ─── Helper: extractFindings() ─────────────────────────
   393→
   394→  private extractFindings(sources: RechercheSource[]): RechercheFinding[] {
   395→    const findings: RechercheFinding[] = [];
   396→
   397→    for (const source of sources) {
   398→      for (let c = 0; c < source.keyClaims.length; c++) {
   399→        const claim = source.keyClaims[c];
   400→        const id = `find_${uuidv4().slice(0, 8)}`;
   401→
   402→        // Determine finding type from claim content
   403→        const type = this.inferFindingType(claim);
   404→
   405→        // Determine consensus based on source credibility and claim nature
   406→        const consensus = this.inferConsensus(source.credibility, claim);
   407→
   408→        // Confidence influenced by source credibility and relevance
   409→        const credibilityMultiplier = source.credibility === 'high' ? 0.85 : source.credibility === 'medium' ? 0.65 : 0.45;
   410→        const confidence = Math.min(1, Math.round((credibilityMultiplier * source.relevance + Math.random() * 0.15) * 100) / 100);
   411→
   412→        findings.push({
   413→          id,
   414→          statement: claim,
   415→          evidence: [`Supported by ${source.title}`, `Source relevance: ${(source.relevance * 100).toFixed(0)}%`],
   416→          confidence,
   417→          sourceIds: [source.id],
   418→          consensus,
   419→          type,
   420→        });
   421→      }
   422→    }
   423→
   424→    // Deduplicate similar findings (merge sourceIds for matching statements)
   425→    return this.deduplicateFindings(findings);
   426→  }

   428→  // ─── Helper: compareSources() ──────────────────────────
   429→
   430→  private compareSources(sources: RechercheSource[]): SourceComparison[] {
   431→    const comparisons: SourceComparison[] = [];
   432→
   433→    // Compare pairs of sources (limit to avoid combinatorial explosion)
   434→    const limit = Math.min(sources.length, 8);
   435→    for (let i = 0; i < limit; i++) {
   436→      for (let j = i + 1; j < limit; j++) {
   437→        const srcA = sources[i];
   438→        const srcB = sources[j];
   439→
   440→        const agreements = this.findAgreements(srcA, srcB);
   441→        const disagreements = this.findDisagreements(srcA, srcB);
   442→
   443→        // Similarity based on shared claims and overlapping topics
   444→        const totalClaims = srcA.keyClaims.length + srcB.keyClaims.length;
   445→        const sharedClaims = agreements.length;
   446→        const similarity = totalClaims > 0
   447→          ? Math.round((2 * sharedClaims / totalClaims) * 100) / 100
   448→          : 0;
   449→
   450→        if (agreements.length > 0 || disagreements.length > 0) {
   451→          comparisons.push({
   452→            sources: [srcA.id, srcB.id],
   453→            agreements,
   454→            disagreements,
   455→            similarity,
   456→          });
   457→        }
   458→      }
   459→    }
   460→
   461→    return comparisons;
   462→  }

   466→  private generateCitations(sources: RechercheSource[], format: string): RechercheReference[] {
   467→    const references: RechercheReference[] = [];
   468→    const formats: RechercheReferenceFormat[] = format === 'all'
   469→      ? ['apa', 'mla', 'chicago', 'bibtex']
   470→      : [format as RechercheReferenceFormat];
   471→
   472→    for (const source of sources) {
   473→      for (const fmt of formats) {
   474→        const id = `ref_${uuidv4().slice(0, 8)}`;
   475→        const citation = this.formatCitation(source, fmt);
   476→        references.push({
   477→          id,
   478→          sourceId: source.id,
   479→          format: fmt,
   480→          citation,
   481→        });
   482→      }
   483→    }
   484→
   485→    return references;
   486→  }

   488→  // ─── Helper: identifyGaps() ────────────────────────────
   489→
   490→  private identifyGaps(findings: RechercheFinding[], sources: RechercheSource[]): string[] {
   491→    const gaps: string[] = [];
   492→
   493→    // Gap: insufficient source diversity
   494→    const types = new Set(sources.map(s => s.type));
   495→    if (types.size < 3) {
   496→      gaps.push('Limited source type diversity — research would benefit from broader source categories (academic, government, news)');
   497→    }
   498→
   499→    // Gap: low-confidence findings
   500→    const lowConfidence = findings.filter(f => f.confidence < 0.5);
   501→    if (lowConfidence.length > findings.length * 0.3) {
   502→      gaps.push('Many findings have low confidence — additional verification from authoritative sources is needed');
   503→    }
   504→
   505→    // Gap: disputed findings
   506→    const disputed = findings.filter(f => f.consensus === 'disputed');
   507→    if (disputed.length > 0) {
   508→      gaps.push(`${disputed.length} finding(s) are disputed across sources — further investigation required to resolve contradictions`);
   509→    }
   510→
   511→    // Gap: temporal coverage
   512→    const hasRecent = sources.some(s => {
   513→      if (!s.publishedDate) return false;
   514→      const year = new Date(s.publishedDate).getFullYear();
   515→      return year >= 2024;
   516→    });
   517→    if (!hasRecent) {
   518→      gaps.push('No recent sources (2024+) found — current data may be outdated for the topic');
   519→    }
   520→
   521→    // Gap: missing academic/peer-reviewed sources
   522→    const hasAcademic = sources.some(s => s.type === 'academic');
   523→    if (!hasAcademic) {
   524→      gaps.push('No peer-reviewed academic sources — adding scholarly research would strengthen evidence');
   525→    }
   526→
   527→    // Gap: insufficient cross-referencing
   528→    const multiSource = findings.filter(f => f.sourceIds.length > 1);
   529→    if (multiSource.length < findings.length * 0.3) {
   530→      gaps.push('Most findings are supported by only a single source — cross-validation from additional sources is recommended');
   531→    }
   532→
   533→    // Gap: opinion-heavy research
   534→    const opinions = findings.filter(f => f.type === 'opinion');
   535→    if (opinions.length > findings.length * 0.5) {
   536→      gaps.push('Research is opinion-dominant — more factual and statistical evidence is needed for a balanced report');
   537→    }
   538→
   539→    // Always include a forward-looking gap
   540→    gaps.push('Further research is recommended to track evolving developments on this topic');
   541→
   542→    return gaps;
   543→  }

   545→  // ─── Helper: renderMarkdown() ──────────────────────────
   546→
   547→  private renderMarkdown(output: RechercheOutput): string {
   548→    const lines: string[] = [];
   549→
   550→    // Title
   551→    lines.push(`# ${output.title}`);
   552→    lines.push('');
   553→
   554→    // Metadata
   555→    lines.push(`> **Query**: ${output.query}`);
   556→    lines.push(`> **Domain**: ${output.classification.domain}`);
   557→    lines.push(`> **Confidence**: ${(output.metadata.confidenceLevel * 100).toFixed(0)}%`);
   558→    lines.push(`> **Sources**: ${output.metadata.sourceCount} | **Findings**: ${output.metadata.findingCount}`);
   559→    lines.push('');
   560→
   561→    // Executive Summary
   562→    lines.push('## Executive Summary');
   563→    lines.push('');
   564→    lines.push(output.executiveSummary);
   565→    lines.push('');
   566→
   567→    // Sections
   568→    for (const section of output.sections) {
   569→      lines.push(`## ${section.heading}`);
   570→      lines.push('');
   571→      lines.push(section.content);
   572→      lines.push('');
   573→
   574→      if (section.sourceIds.length > 0) {
   575→        const sourceRefs = section.sourceIds
   576→          .map(sid => {
   577→            const src = output.sources.find(s => s.id === sid);
   578→            return src ? `[${src.title}](${src.url})` : sid;
   579→          })
   580→          .join(', ');
   581→        lines.push(`*Sources: ${sourceRefs}*`);
   582→        lines.push('');
   583→      }
   584→
   585→      lines.push(`*Confidence: ${(section.confidence * 100).toFixed(0)}%*`);
   586→      lines.push('');
   587→    }
   588→
   589→    // Knowledge Gaps
   590→    if (output.gaps.length > 0) {
   591→      lines.push('## Knowledge Gaps');
   592→      lines.push('');
   593→      for (const gap of output.gaps) {
   594→        lines.push(`- ${gap}`);
   595→      }
   596→      lines.push('');
   597→    }
   598→
   599→    // Sources
   600→    lines.push('## Sources');
   601→    lines.push('');
   602→    for (const source of output.sources) {
   603→      const credibilityBadge = source.credibility === 'high' ? '✓' : source.credibility === 'medium' ? '~' : '!';
   604→      lines.push(`- ${credibilityBadge} **${source.title}** — ${source.url} _(${source.type}, relevance: ${(source.relevance * 100).toFixed(0)}%)_`);
   605→    }
   606→    lines.push('');
   607→
   608→    // References
   609→    const apaRefs = output.references.filter(r => r.format === 'apa');
   610→    if (apaRefs.length > 0) {
   611→      lines.push('## References (APA)');
   612→      lines.push('');

   614→        lines.push(`1. ${ref.citation}`);
   615→      }
   616→      lines.push('');

   619→    return lines.join('\n');

   659→  private scoreSourceQuality(output: RechercheOutput, suggestions: string[]): number {
   660→    if (output.sources.length === 0) {
   661→      suggestions.push('No sources found — research cannot be validated');
   662→      return 0;
   663→    }
   664→
   665→    let score = 0;
   666→
   667→    // Credibility distribution (high credibility should dominate)
   668→    const highCred = output.sources.filter(s => s.credibility === 'high').length;
   669→    const medCred = output.sources.filter(s => s.credibility === 'medium').length;
   670→    const lowCred = output.sources.filter(s => s.credibility === 'low').length;
   671→    const total = output.sources.length;
   672→
   673→    const highRatio = highCred / total;
   674→    const lowRatio = lowCred / total;
   675→
   676→    // Reward high-credibility sources, penalize low-credibility
   677→    score += Math.min(0.4, highRatio * 0.5);
   678→    score += Math.min(0.2, (medCred / total) * 0.25);
   679→    score -= Math.min(0.2, lowRatio * 0.3);
   680→
   681→    // Source type diversity
   682→const uniqueTypes = Array.from(new Set(output.sources.map(s => s.type))).length;
   683→    score += Math.min(0.2, uniqueTypes * 0.04);
   684→
   685→    // Average relevance
   686→    const avgRelevance = output.sources.reduce((sum, s) => sum + s.relevance, 0) / total;
   687→    score += Math.min(0.2, avgRelevance * 0.2);
   688→
   689→    if (highRatio < 0.2) {
   690→      suggestions.push('Increase the proportion of high-credibility sources (academic, government)');
   691→    }
   692→    if (uniqueTypes < 3) {
   693→      suggestions.push('Diversify source types for more balanced research');
   694→    }
   695→
   696→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   697→  }

   699→  private scoreComprehensiveness(output: RechercheOutput, suggestions: string[]): number {
   700→    let score = 0;
   701→
   702→    // Section type coverage
   703→    const requiredTypes: RechercheSection['type'][] = ['overview', 'findings', 'analysis', 'conclusion'];
   704→    const presentTypes = new Set(output.sections.map(s => s.type));
   705→    const coverageRatio = requiredTypes.filter(t => presentTypes.has(t)).length / requiredTypes.length;
   706→    score += coverageRatio * 0.4;
   707→
   708→    // Section depth (word count per section)
   709→    const avgWords = output.sections.reduce((sum, s) => sum + s.content.split(/\s+/).length, 0) / Math.max(output.sections.length, 1);
   710→    if (avgWords >= 150) {
   711→      score += 0.2;
   712→    } else if (avgWords >= 100) {
   713→      score += 0.1;
   714→      suggestions.push('Sections should contain at least 150 words for thorough coverage');
   715→    }
   716→
   717→    // Source count adequacy
   718→    if (output.metadata.sourceCount >= 8) {
   719→      score += 0.2;
   720→    } else if (output.metadata.sourceCount >= 4) {
   721→      score += 0.1;
   722→      suggestions.push('Consider adding more sources for comprehensive coverage');
   723→    }
   724→
   725→    // Finding count
   726→    if (output.metadata.findingCount >= 5) {
   727→      score += 0.1;
   728→    } else if (output.metadata.findingCount >= 2) {
   729→      score += 0.05;
   730→    }
   731→
   732→    // Gaps identified
   733→    if (output.gaps.length >= 2) {
   734→      score += 0.1;
   735→    }
   736→
   737→    if (coverageRatio < 1) {
   738→      const missing = requiredTypes.filter(t => !presentTypes.has(t));
   739→      suggestions.push(`Add missing section types: ${missing.join(', ')}`);
   740→    }
   741→
   742→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   743→  }

   745→  private scoreAccuracy(output: RechercheOutput, suggestions: string[]): number {
   746→    if (output.findings.length === 0) {
   747→      suggestions.push('No findings extracted — accuracy cannot be assessed');
   748→      return 0;
   749→    }
   750→
   751→    let score = 0;
   752→
   753→    // Average finding confidence
   754→    const avgConfidence = output.findings.reduce((sum, f) => sum + f.confidence, 0) / output.findings.length;
   755→    score += avgConfidence * 0.3;
   756→
   757→    // Evidence backing per finding
   758→    const avgEvidence = output.findings.reduce((sum, f) => sum + f.evidence.length, 0) / output.findings.length;
   759→    score += Math.min(0.25, avgEvidence * 0.08);
   760→
   761→    // Multi-source validation
   762→    const multiSourceFindings = output.findings.filter(f => f.sourceIds.length > 1).length;
   763→    const multiSourceRatio = multiSourceFindings / output.findings.length;
   764→    score += multiSourceRatio * 0.25;
   765→
   766→    // Disputed findings penalty
   767→    const disputedRatio = output.findings.filter(f => f.consensus === 'disputed').length / output.findings.length;
   768→    score -= disputedRatio * 0.15;
   769→
   770→    // Strong consensus bonus
   771→    const strongConsensusRatio = output.findings.filter(f => f.consensus === 'strong').length / output.findings.length;
   772→    score += strongConsensusRatio * 0.1;
   773→
   774→    if (avgConfidence < 0.5) {
   775→      suggestions.push('Average finding confidence is below 50% — verify sources and claims');
   776→    }
   777→    if (disputedRatio > 0.3) {
   778→      suggestions.push('High proportion of disputed findings — resolve contradictions or present them clearly');
   779→    }
   780→
   781→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   782→  }

   784→  private scoreSynthesis(output: RechercheOutput, suggestions: string[]): number {
   785→    let score = 0;
   786→
   787→    // Cross-references: sections that reference multiple sources
   788→    const multiRefSections = output.sections.filter(s => s.sourceIds.length > 1).length;
   789→    const sectionRatio = output.sections.length > 0 ? multiRefSections / output.sections.length : 0;
   790→    score += Math.min(0.3, sectionRatio * 0.4);
   791→
   792→    // Comparisons between sources
   793→    if (output.comparisons.length > 0) {
   794→      score += Math.min(0.3, output.comparisons.length * 0.1);
   795→    } else {
   796→      suggestions.push('No source comparisons generated — add cross-referencing between sources');
   797→    }
   798→
   799→    // Executive summary references findings
   800→    if (output.executiveSummary.length > 100) {
   801→      score += 0.2;
   802→    } else {
   803→      suggestions.push('Executive summary is too brief — synthesize key findings more thoroughly');
   804→    }
   805→
   806→    // Findings linked to sections
   807→    const findingsReferenced = output.sections.some(s =>
   808→      s.type === 'findings' || s.type === 'analysis'
   809→    );
   810→    if (findingsReferenced) {
   811→      score += 0.2;
   812→    } else {
   813→      suggestions.push('Add a dedicated findings or analysis section to synthesize research results');
   814→    }
   815→
   816→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   817→  }

   819→  private scorePresentation(output: RechercheOutput, prompt: string, suggestions: string[]): number {
   820→    let score = 0;
   821→
   822→    // Title quality
   823→    if (output.title && output.title.length > 5) {
   824→      score += 0.1;
   825→    }
   826→
   827→    // Structured sections
   828→    if (output.sections.length >= 3) {
   829→      score += 0.2;
   830→    } else {
   831→      suggestions.push('Report needs at least 3 sections for adequate structure');
   832→    }
   833→
   834→    // Section ordering (overview should come first, conclusion last)
   835→    const sectionTypes = output.sections.map(s => s.type);
   836→    if (sectionTypes[0] === 'overview') {
   837→      score += 0.1;
   838→    }
   839→    if (sectionTypes[sectionTypes.length - 1] === 'conclusion' || sectionTypes[sectionTypes.length - 1] === 'recommendations') {
   840→      score += 0.1;
   841→    }
   842→
   843→    // References present
   844→    if (output.references.length > 0) {
   845→      score += 0.15;
   846→    } else {
   847→      suggestions.push('No references formatted — add citations in standard academic styles');
   848→    }
   849→
   850→    // Gaps identified
   851→    if (output.gaps.length > 0) {
   852→      score += 0.1;
   853→    }
   854→
   855→    // Metadata completeness
   856→    const meta = output.metadata;
   857→    if (meta.sourceCount > 0 && meta.findingCount > 0 && meta.confidenceLevel > 0) {
   858→      score += 0.15;
   859→    } else {
   860→      suggestions.push('Complete all metadata fields (sourceCount, findingCount, confidenceLevel)');
   861→    }
   862→
   863→    // Classification present
   864→    if (output.classification.primaryTopic && output.classification.domain) {
   865→      score += 0.1;
   866→    }
   867→
   868→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   869→  }

   892→    }
   893→

   900→    parts.push(`Overall sentiment across sources is ${classification.sentiment}.`);
   901→
   902→    return parts.join(' ');

  1195→  private inferConsensus(credibility: RechercheSourceCredibility, claim: string): RechercheFindingConsensus {
  1196→    // High-credibility sources tend to have stronger consensus
  1197→    if (credibility === 'high') return 'strong';
  1198→    if (credibility === 'medium') {
  1199→      // Check for hedging language
  1200→      if (/\b(may|might|could|suggest|possibly|perhaps)\b/.test(claim.toLowerCase())) return 'moderate';
  1201→      return 'moderate';
  1202→    }
  1203→    // Low credibility tends toward weaker consensus
  1204→    if (/\b(controversial|disputed|debated|contradicts)\b/.test(claim.toLowerCase())) return 'disputed';
  1205→    return 'weak';
  1206→  }

  1208→  private deduplicateFindings(findings: RechercheFinding[]): RechercheFinding[] {
  1209→    const seen = new Map<string, RechercheFinding>();
  1210→
  1211→    for (const finding of findings) {
  1212→      const key = finding.statement.toLowerCase().trim();
  1213→      const existing = seen.get(key);
  1214→
  1215→      if (existing) {
  1216→        // Merge: combine source IDs and evidence, keep higher confidence
  1217→        existing.sourceIds = Array.from(new Set([...existing.sourceIds, ...finding.sourceIds]));
  1218→        existing.evidence = Array.from(new Set([...existing.evidence, ...finding.evidence]));
  1219→        existing.confidence = Math.max(existing.confidence, finding.confidence);
  1220→
  1221→        // Upgrade consensus if multiple sources agree
  1222→        const consensusOrder: RechercheFindingConsensus[] = ['weak', 'moderate', 'strong'];
  1223→        if (consensusOrder.indexOf(finding.consensus) > consensusOrder.indexOf(existing.consensus)) {
  1224→          existing.consensus = finding.consensus;
  1225→        }
  1226→      } else {
  1227→        seen.set(key, { ...finding, sourceIds: [...finding.sourceIds], evidence: [...finding.evidence] });
  1228→      }
  1229→    }
  1230→
  1231→    return Array.from(seen.values());
  1232→  }

  1234→  private findAgreements(srcA: RechercheSource, srcB: RechercheSource): string[] {
  1235→    const agreements: string[] = [];
  1236→
  1237→    // Simple keyword overlap on key claims
  1238→    const wordsA = new Set(srcA.keyClaims.flatMap(c => c.toLowerCase().split(/\s+/)));
  1239→    const wordsB = new Set(srcB.keyClaims.flatMap(c => c.toLowerCase().split(/\s+/)));
  1240→
  1241→    for (const claimA of srcA.keyClaims) {
  1242→      const claimAWords = new Set(claimA.toLowerCase().split(/\s+/));
  1243→      const overlap = Array.from(claimAWords).filter(w => w.length > 4 && wordsB.has(w)).length;
  1244→      if (overlap >= 2) {
  1245→        agreements.push(claimA);
  1246→      }
  1247→    }
  1248→
  1249→    return agreements.slice(0, 5);
  1250→  }

  1252→  private findDisagreements(srcA: RechercheSource, srcB: RechercheSource): string[] {
  1253→    const disagreements: string[] = [];
  1254→
  1255→    // Detect contradictory language patterns
  1256→    const contradictionPatterns = [
  1257→      /\bincrease\b/i, /\bdecrease\b/i,
  1258→      /\brising\b/i, /\bfalling\b/i,
  1259→      /\bpositive\b/i, /\bnegative\b/i,
  1260→      /\bsupports\b/i, /\bcontradicts\b/i,
  1261→      /\bbeneficial\b/i, /\bharmful\b/i,
  1262→    ];
  1263→
  1264→    for (const claimA of srcA.keyClaims) {
  1265→      for (let i = 0; i < contradictionPatterns.length; i += 2) {
  1266→        const patA = contradictionPatterns[i];
  1267→        const patB = contradictionPatterns[i + 1];
  1268→        if (patA && patB && patA.test(claimA)) {
  1269→          for (const claimB of srcB.keyClaims) {
  1270→            if (patB.test(claimB)) {
  1271→              disagreements.push(`${claimA} vs. ${claimB}`);
  1272→            }
  1273→          }
  1274→        }
  1275→      }
  1276→    }
  1277→
  1278→    return disagreements.slice(0, 3);
  1279→  }

  1306→  private computeConfidenceLevel(findings: RechercheFinding[], sources: RechercheSource[]): number {
  1307→    if (findings.length === 0) return 0;
  1308→
  1309→    // Weighted average of finding confidences, boosted by multi-source validation
  1310→    const totalConfidence = findings.reduce((sum, f) => {
  1311→      const multiSourceBonus = f.sourceIds.length > 1 ? 0.05 : 0;
  1312→      const consensusBonus = f.consensus === 'strong' ? 0.05 : f.consensus === 'moderate' ? 0.02 : 0;
  1313→      return sum + Math.min(1, f.confidence + multiSourceBonus + consensusBonus);
  1314→    }, 0);
  1315→
  1316→    const avgConfidence = totalConfidence / findings.length;
  1317→
  1318→    // Factor in source credibility
  1319→    const highCredRatio = sources.filter(s => s.credibility === 'high').length / Math.max(sources.length, 1);
  1320→    const credibilityBoost = highCredRatio * 0.1;
  1321→
  1322→    return Math.min(1, Math.round((avgConfidence + credibilityBoost) * 100) / 100);
  1323→  }

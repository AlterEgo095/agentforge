    48→function randomUserAgent(): string {
    49→  return USER_AGENTS[Math.floor(Math.random() * USER_AGENTS.length)];
    50→}
    51→
    52→function randomDelay(min = 50, max = 200): number {
    53→  return Math.floor(Math.random() * (max - min) + min);
    54→}

   129→  async launch(): Promise<ActionResult<boolean>> {
   130→    const start = Date.now();
   131→    try {
   132→      const launchOptions: Record<string, unknown> = {
   133→        headless: this.config.headless,
   134→      };
   135→      if (this.config.proxy) {
   136→        launchOptions.proxy = this.config.proxy;
   137→      }
   138→
   139→      this.browser = await chromium.launch(launchOptions);
   140→
   141→      const contextOptions: Record<string, unknown> = {
   142→        viewport: this.config.viewport,
   143→        userAgent: this.config.userAgent ?? (this.config.stealth ? randomUserAgent() : undefined),
   144→      };
   145→
   146→      this.context = await this.browser.newContext(contextOptions);
   147→
   148→      if (this.config.stealth) {
   149→        await this.applyStealthScripts();
   150→      }
   151→
   152→      const page = await this.context.newPage();
   153→      page.setDefaultTimeout(this.config.defaultTimeout);
   154→      page.setDefaultNavigationTimeout(this.config.navigationTimeout);
   155→      this.pages = [page];
   156→      this.activePageIndex = 0;
   157→
   158→      return result('success', true, Date.now() - start);
   159→    } catch (err) {
   160→      return errorResult(`Failed to launch browser: ${(err as Error).message}`, Date.now() - start);
   161→    }
   162→  }
   163→
   164→  async close(): Promise<ActionResult<boolean>> {
   165→    const start = Date.now();
   166→    try {
   167→      if (this.context) {
   168→        await this.context.close();
   169→      }
   170→      if (this.browser) {
   171→        await this.browser.close();
   172→      }
   173→      this.browser = null;
   174→      this.context = null;
   175→      this.pages = [];
   176→      this.activePageIndex = 0;
   177→      return result('success', true, Date.now() - start);
   178→    } catch (err) {
   179→      return errorResult(`Failed to close browser: ${(err as Error).message}`, Date.now() - start);
   180→    }
   181→  }

   191→  async open(url: string): Promise<ActionResult<NavigateResult>> {
   192→    const start = Date.now();
   193→    try {
   194→      const page = this.getActivePage();
   195→      const response = await page.goto(url, { waitUntil: 'domcontentloaded' });
   196→
   197→      const finalUrl = page.url();
   198→      const title = await page.title();
   199→      const redirected = finalUrl !== url;
   200→
   201→      const redirectChain: string[] = [url];
   202→      if (redirected) {
   203→        redirectChain.push(finalUrl);
   204→      }
   205→
   206→      const navResult: NavigateResult = {
   207→        url: finalUrl,
   208→        title,
   209→        redirected,
   210→        redirectChain,
   211→      };
   212→
   213→      // Save to history
   214→      this.memory.addToHistory(finalUrl, title);
   215→
   216→      return result('success', navResult, Date.now() - start, finalUrl);
   217→    } catch (err) {
   218→      const msg = (err as Error).message;
   219→      if (msg.includes('Timeout') || msg.includes('timeout')) {
   220→        return timeoutResult(msg, Date.now() - start, url);
   221→      }
   222→      return errorResult(msg, Date.now() - start, url);
   223→    }
   224→  }

   441→    const start = Date.now();

   443→      const page = this.getActivePage();
   444→      const values = await page.evaluate(
   445→        ([sel, attr]) => {

   447→          return elements.map((el) => {
   448→            if (attr) return (el as HTMLElement).getAttribute(attr) ?? '';
   449→            return el.textContent?.trim() ?? '';
   450→          });
   451→        },
   452→        [selector, attribute ?? null] as [string, string | null],
   453→      );
   454→      const extractResult: ExtractResult = { selector, values, count: values.length };
   455→      return result('success', extractResult, Date.now() - start, page.url());
   456→    } catch (err) {
   457→      return errorResult((err as Error).message, Date.now() - start);
   458→    }

   546→  async search(query: string, engine: SearchEngine = 'duckduckgo'): Promise<ActionResult<SearchResult>> {
   547→    const start = Date.now();
   548→    try {
   549→      if (!SEARCH_URL_BUILDERS[engine]) {
   550→        return errorResult(`Unknown search engine: ${engine}`, Date.now() - start);
   551→      }
   552→
   553→      const searchUrl = SEARCH_URL_BUILDERS[engine](query);
   554→      const page = this.getActivePage();
   555→      await page.goto(searchUrl, { waitUntil: 'domcontentloaded', timeout: this.config.navigationTimeout });
   556→
   557→      let items: SearchItem[] = [];
   558→
   559→      switch (engine) {
   560→        case 'google':
   561→          items = await this.extractGoogleResults();
   562→          break;
   563→        case 'github':
   564→          items = await this.extractGithubResults();
   565→          break;
   566→        case 'arxiv':
   567→          items = await this.extractArxivResults();
   568→          break;
   569→        default:
   570→          items = await this.extractGenericResults();
   571→          break;
   572→      }
   573→
   574→      const searchResult: SearchResult = { engine, query, items };
   575→      return result('success', searchResult, Date.now() - start, page.url());
   576→    } catch (err) {
   577→      return errorResult(`Search failed: ${(err as Error).message}`, Date.now() - start);
   578→    }
   579→  }

   581→  private async extractGoogleResults(): Promise<SearchItem[]> {
   582→    const page = this.getActivePage();
   583→    return page.evaluate(() => {
   584→      const results: SearchItem[] = [];
   585→      const elements = document.querySelectorAll('div.g');
   586→      elements.forEach((el) => {
   587→        const titleEl = el.querySelector('h3');
   588→        const linkEl = el.querySelector('a');
   589→        const snippetEl = el.querySelector('[data-sncf]');

   682→  async download(url: string, path?: string): Promise<ActionResult<DownloadResult>> {
   683→    const start = Date.now();
   684→    try {
   685→      const page = this.getActivePage();
   686→      const downloadPromise = page.waitForEvent('download', { timeout: this.config.defaultTimeout });
   687→      await page.goto(url, { waitUntil: 'domcontentloaded' });
   688→      // If we get here, the page loaded instead of triggering a download
   689→      // Try clicking any download link
   690→      const download = await downloadPromise.catch(() => null);
   691→      if (download) {
   692→        const dest = path ?? (this.config.downloadDir ? `${this.config.downloadDir}/${download.suggestedFilename()}` : `/tmp/${download.suggestedFilename()}`);
   693→        await download.saveAs(dest);
   694→        const downloadResult: DownloadResult = { url, path: dest, size: 0 };
   695→        return result('success', downloadResult, Date.now() - start, url);
   696→      }
   697→      return errorResult('No download triggered', Date.now() - start, url);
   698→    } catch (err) {
   699→      return errorResult((err as Error).message, Date.now() - start, url);
   700→    }
   701→  }

   819→        url: page.url(),

   850→  async closeTab(index?: number): Promise<ActionResult<boolean>> {
   851→    const start = Date.now();
   852→    try {
   853→      const closeIndex = index ?? this.activePageIndex;
   854→      if (closeIndex < 0 || closeIndex >= this.pages.length) {
   855→        return errorResult(`Tab index ${closeIndex} out of range`, Date.now() - start);
   856→      }
   857→      if (this.pages.length <= 1) {
   858→        return errorResult('Cannot close the last tab', Date.now() - start);
   859→      }
   860→      const page = this.pages[closeIndex];
   861→      await page.close();
   862→      this.pages.splice(closeIndex, 1);
   863→
   864→      if (this.activePageIndex >= this.pages.length) {
   865→        this.activePageIndex = this.pages.length - 1;
   866→      }
   867→
   868→      return result('success', true, Date.now() - start);
   869→    } catch (err) {
   870→      return errorResult((err as Error).message, Date.now() - start);
   871→    }
   872→  }

   978→  async fillForm(fields: FormField[]): Promise<ActionResult<boolean>> {
   979→    const start = Date.now();

   981→      const page = this.getActivePage();
   982→      for (const field of fields) {
   983→        if (this.config.stealth) await this.stealthDelay();
   984→        switch (field.type ?? 'text') {
   985→          case 'text':
   986→            await page.fill(field.selector, field.value);
   987→            break;
   988→          case 'select':
   989→            await page.selectOption(field.selector, field.value);
   990→            break;
   991→          case 'checkbox':
   992→            if (field.value === 'true' || field.value === 'checked') {
   993→              await page.check(field.selector);
   994→            } else {
   995→              await page.uncheck(field.selector);
   996→            }
   997→            break;
   998→          case 'radio':
   999→            await page.click(field.selector);
  1000→            break;
  1001→          case 'file':
  1002→            await page.setInputFiles(field.selector, field.value);
  1003→            break;
  1004→        }
  1005→      }
  1006→      return result('success', true, Date.now() - start, page.url());
  1007→    } catch (err) {
  1008→      return errorResult((err as Error).message, Date.now() - start);
  1009→    }

  1073→  private async applyStealthScripts(): Promise<void> {
  1074→    if (!this.context) return;

  1076→    // Override navigator.webdriver
  1077→    await this.context.addInitScript(() => {
  1078→      Object.defineProperty(navigator, 'webdriver', {
  1079→        get: () => false,

  1084→        get: () => [1, 2, 3, 4, 5],

  1086→
  1087→      // Override languages
  1088→      Object.defineProperty(navigator, 'languages', {
  1089→        get: () => ['en-US', 'en'],
  1090→      });
  1091→
  1092→      // Override Chrome runtime

  1096→    });

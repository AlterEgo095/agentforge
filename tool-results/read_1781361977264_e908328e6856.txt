   115→   115→
   116→   116→  protected async generate(
   117→   117→    prompt: string,
   118→   118→    context: Record<string, unknown>,
   119→   119→    sessionId: string,
   120→   120→  ): Promise<DocOutput> {
   121→   121→    // Phase 1: Parse document type from prompt
   122→   122→    const docType = this.parseDocumentType(prompt);
   123→   123→    const topic = this.extractTopic(prompt);
   124→   124→
   125→   125→    this.emitEvent('doc_outline', sessionId, { docType, topic });
   126→   126→
   127→   127→    // Phase 2: Build document outline with proper hierarchy
   128→   128→    const outline = this.buildOutline(docType, topic);
   129→   129→
   130→   130→    // Phase 3: Generate content for each section
   131→   131→    // (structured data generation — actual LLM content is provided by the orchestrator)
   132→   132→    const sections = this.generateSectionContent(outline, prompt, context);
   133→   133→
   134→   134→    this.emitEvent('doc_content', sessionId, {
   135→   135→      sectionCount: sections.length,
   136→   136→      totalWordEstimate: this.estimateSectionWords(sections),
   137→   137→    });
   138→   138→
   139→   139→    // Phase 4: Build table of contents from section hierarchy
   140→   140→    const tableOfContents = this.buildToc(sections);
   141→   141→
   142→   142→    // Phase 5: Add references if sources are mentioned or available
   143→   143→    const references = this.extractReferences(prompt, context);
   144→   144→
   145→   145→    // Phase 6: Compute metadata
   146→   146→    const totalWordCount = this.computeTotalWordCount(sections);
   147→   147→    const readingTimeMinutes = Math.max(1, Math.round(totalWordCount / 238)); // avg 238 wpm
   148→   148→
   149→   149→    this.emitEvent('doc_formatting', sessionId, {
   150→   150→      wordCount: totalWordCount,
   151→   151→      readingTimeMinutes,
   152→   152→      format: 'document',
   153→   153→    });
   154→   154→
   155→   155→    return {
   156→   156→      title: this.generateTitle(docType, topic),
   157→   157→      subtitle: this.generateSubtitle(docType, topic),
   158→   158→      author: context.author as string | undefined,
   159→   159→      sections,
   160→   160→      tableOfContents,
   161→   161→      references,
   162→   162→      metadata: {
   163→   163→        wordCount: totalWordCount,
   164→   164→        readingTimeMinutes,
   165→   165→        format: docType,
   166→   166→        language: 'en',
   167→   167→        sourceCount: references.length,
   168→   168→      },
   169→   169→    };
   170→   170→  }

   174→   174→  protected async validateOutput(output: DocOutput, _prompt: string): Promise<QualityReport> {
   175→   175→    const dimensions: Record<string, number> = {};
   176→   176→    const suggestions: string[] = [];
   177→   177→
   178→   178→    // 1. Content Depth (0–1)
   179→   179→    dimensions['Content Depth'] = this.scoreContentDepth(output, suggestions);
   180→   180→
   181→   181→    // 2. Structure & Organization (0–1)
   182→   182→    dimensions['Structure & Organization'] = this.scoreStructure(output, suggestions);
   183→   183→
   184→   184→    // 3. Formatting & Typography (0–1)
   185→   185→    dimensions['Formatting & Typography'] = this.scoreFormatting(output, suggestions);
   186→   186→
   187→   187→    // 4. Accuracy & Citations (0–1)
   188→   188→    dimensions['Accuracy & Citations'] = this.scoreCitations(output, suggestions);
   189→   189→
   190→   190→    // 5. Clarity & Readability (0–1)
   191→   191→    dimensions['Clarity & Readability'] = this.scoreReadability(output, suggestions);
   192→   192→
   193→   193→    const overall = this.computeWeightedScore(dimensions);
   194→   194→    const passed = this.isQualityPassing(overall);
   195→   195→
   196→   196→    return { overall, dimensions, passed, suggestions };
   197→   197→  }

   227→   227→  // ─── Helper: Document Type Parsing ─────────────────────
   228→   228→
   229→   229→  /**
   230→   230→   * Detect the document type from the user prompt by matching against
   231→   231→   * known keyword sets. Falls back to "report".
   232→   232→   */
   233→   233→  parseDocumentType(prompt: string): string {
   234→   234→    const lower = prompt.toLowerCase();
   235→   235→    let bestMatch = 'report';
   236→   236→    let bestScore = 0;
   237→   237→
   238→   238→    for (const [docType, keywords] of Object.entries(DOCUMENT_TYPE_KEYWORDS)) {
   239→   239→      let score = 0;
   240→   240→      for (const keyword of keywords) {
   241→   241→        if (lower.includes(keyword)) {
   242→   242→          score += keyword.length; // longer keyword matches are stronger signals
   243→   243→        }
   244→   244→      }
   245→   245→      if (score > bestScore) {
   246→   246→        bestScore = score;
   247→   247→        bestMatch = docType;
   248→   248→      }
   249→   249→    }
   250→   250→
   251→   251→    return bestMatch;
   252→   252→  }
   253→   253→
   254→   254→  /**
   255→   255→   * Extract the core topic / subject from the prompt.
   256→   256→   * Uses a simple heuristic: strips common command prefixes and returns
   257→   257→   * the remainder trimmed to a reasonable length.
   258→   258→   */
   259→   259→  private extractTopic(prompt: string): string {
   260→   260→    const prefixes = [
   261→   261→      /^(?:write|create|generate|draft|produce|build)\s+(?:a\s+)?(?:report|proposal|manual|whitepaper|memo|letter|article|specification|spec|plan|document)\s+(?:about|on|for|regarding)\s+/i,
   262→   262→      /^(?:write|create|generate|draft|produce|build)\s+(?:a\s+)?/i,
   263→   263→    ];
   264→   264→
   265→   265→    let topic = prompt.trim();
   266→   266→    for (const prefix of prefixes) {
   267→   267→      topic = topic.replace(prefix, '');
   268→   268→    }
   269→   269→
   270→   270→    // Truncate to first sentence or 120 characters whichever is shorter
   271→   271→    const firstSentence = topic.match(/^[^.!?\n]+/);
   272→   272→    if (firstSentence) {
   273→   273→      topic = firstSentence[0].trim();
   274→   274→    }
   275→   275→
   276→   276→    return topic.length > 120 ? topic.slice(0, 117) + '...' : topic;
   277→   277→  }
   278→   278→
   279→   279→  // ─── Helper: Outline Building ──────────────────────────
   280→   280→
   281→   281→  /**
   282→   282→   * Create a section hierarchy from a document-type template.
   283→   283→   * Each section gets a unique ID, placeholder content, and recursive
   284→   284→   * children based on the template.
   285→   285→   */
   286→   286→  buildOutline(docType: string, topic: string): DocSection[] {
   287→   287→    const templates = DOCUMENT_TYPE_TEMPLATES[docType] ?? DOCUMENT_TYPE_TEMPLATES['report'];
   288→   288→    return templates.map((tpl, index) => this.instantiateSection(tpl, topic, [index]));
   289→   289→  }
   290→   290→
   291→   291→  private instantiateSection(tpl: SectionTemplate, topic: string, path: number[]): DocSection {
   292→   292→    const id = `sec-${path.join('-')}`;
   293→   293→    const section: DocSection = {
   294→   294→      id,
   295→   295→      heading: tpl.heading,
   296→   296→      level: tpl.level,
   297→   297→      content: this.generatePlaceholderContent(tpl.heading, tpl.level, topic),
   298→   298→    };
   299→   299→
   300→   300→    if (tpl.children && tpl.children.length > 0) {
   301→   301→      section.children = tpl.children.map((child, childIndex) =>
   302→   302→        this.instantiateSection(child, topic, [...path, childIndex]),
   303→   303→      );
   304→   304→    }
   305→   305→
   306→   306→    return section;
   307→   307→  }
   308→   308→
   309→   309→  /**
   310→   310→   * Generate structured placeholder content for a section.
   311→   311→   * Real LLM-generated content would be injected by the orchestrator;
   312→   312→   * this provides a well-structured scaffold with appropriate depth cues.
   313→   313→   */
   314→   314→  private generatePlaceholderContent(heading: string, level: 1 | 2 | 3 | 4, topic: string): string {
   315→   315→    const minWords = level <= 2 ? 150 : 75;
   316→   316→    const base = `This section covers ${heading.toLowerCase()} in the context of ${topic}. `;
   317→   317→
   318→   318→    const elaborations = [
   319→   319→      `The following provides a comprehensive overview of ${heading.toLowerCase()}, including key considerations, analysis, and actionable insights relevant to ${topic}.`,
   320→   320→      `It is essential to understand the foundational aspects presented here, as they form the basis for subsequent sections and recommendations throughout this document.`,
   321→   321→      `The discussion that follows draws upon established practices and contemporary research, ensuring both rigor and relevance in addressing the core themes of ${topic}.`,
   322→   322→      `Stakeholders should consider the interdependencies between this area and the broader strategic context, as outlined in the executive summary and conclusion sections.`,
   323→   323→      `Further detail and supporting evidence can be found in the referenced sources appended to this document.`,
   324→   324→    ];
   325→   325→
   326→   326→    let content = base + elaborations.join(' ');
   327→   327→
   328→   328→    // Ensure minimum word count for level 1–2 sections
   329→   329→    while (this.countWords(content) < minWords) {
   330→   330→      content += ` Additional context and elaboration on ${heading.toLowerCase()} is provided to ensure comprehensive coverage of this important topic. The analysis considers multiple perspectives and draws on both qualitative and quantitative evidence where available.`;
   331→   331→    }
   332→   332→
   333→   333→    return content;
   334→   334→  }

   336→   336→  // ─── Helper: Section Content Generation ────────────────
   337→   337→
   338→   338→  /**
   339→   339→   * Enrich sections with contextual content, tables, images, or code
   340→   340→   * blocks based on the document type and prompt signals.
   341→   341→   */
   342→   342→  private generateSectionContent(
   343→   343→    outline: DocSection[],
   344→   344→    prompt: string,
   345→   345→    context: Record<string, unknown>,
   346→   346→  ): DocSection[] {
   347→   347→    return outline.map((section) => {
   348→   348→      const enriched = { ...section };
   349→   349→
   350→   350→      // If the orchestrator has provided previous output feedback, incorporate it
   351→   351→      if (context.previousOutput && typeof context.previousOutput === 'object') {
   352→   352→        const prev = context.previousOutput as Record<string, unknown>;
   353→   353→        if (prev.sections && Array.isArray(prev.sections)) {
   354→   354→          const matching = (prev.sections as DocSection[]).find((s) => s.heading === section.heading);
   355→   355→          if (matching && matching.content) {
   356→   356→            enriched.content = matching.content;
   357→   357→          }
   358→   358→        }
   359→   359→      }
   360→   360→
   361→   361→      // Recursively enrich children
   362→   362→      if (enriched.children && enriched.children.length > 0) {
   363→   363→        enriched.children = this.generateSectionContent(enriched.children, prompt, context);
   364→   364→      }
   365→   365→
   366→   366→      // Add structural elements based on heading semantics
   367→   367→      enriched.tables = this.suggestTables(enriched.heading, prompt);
   368→   368→      enriched.images = this.suggestImages(enriched.heading, prompt);
   369→   369→      enriched.codeBlocks = this.suggestCodeBlocks(enriched.heading, prompt);
   370→   370→
   371→   371→      return enriched;
   372→   372→    });
   373→   373→  }

   388→   388→          ['Item 2', 'Description of item 2', 'Additional detail', 'Pending'],

   391→   391→        caption: `${heading} — Overview Table`,
   392→   392→      },
   393→   393→    ];
   394→   394→  }

   431→   431→  /**
   432→   432→   * Generate a recursive Table of Contents from the section hierarchy.
   433→   433→   */
   434→   434→  buildToc(sections: DocSection[]): TocEntry[] {
   435→   435→    return sections.map((section) => {

   437→   437→        id: section.id,
   438→   438→        title: section.heading,
   439→   439→        level: section.level,
   440→   440→      };
   441→   441→
   442→   442→      if (section.children && section.children.length > 0) {
   443→   443→        entry.children = this.buildToc(section.children);
   444→   444→      }
   445→   445→
   446→   446→      return entry;
   447→   447→    });
   448→   448→  }

   451→   451→
   452→   452→  /**
   453→   453→   * Extract references from prompt and context. In production this
   454→   454→   * would be powered by the RECHERCHE agent; here we provide a
   455→   455→   * scaffold that detects explicit source mentions.
   456→   456→   */
   457→   457→  private extractReferences(prompt: string, context: Record<string, unknown>): DocReference[] {
   458→   458→    const references: DocReference[] = [];
   459→   459→    const sourcePattern = /(?:source|ref|cite|according to|referenced?)(?:\s+(?:to|from))?\s+["']([^"']+)["']/gi;
   460→   460→    let match: RegExpExecArray | null;
   461→   461→
   462→   462→    while ((match = sourcePattern.exec(prompt)) !== null) {
   463→   463→      references.push({
   464→   464→        id: `ref-${references.length + 1}`,
   465→   465→        type: 'other',
   466→   466→        title: match[1],
   467→   467→      });
   468→   468→    }
   469→   469→
   470→   470→    // Also include any references passed through context (e.g. from RECHERCHE agent)
   471→   471→    if (context.references && Array.isArray(context.references)) {
   472→   472→      for (const ref of context.references as DocReference[]) {
   473→   473→        if (ref.id && ref.title) {
   474→   474→          references.push(ref);
   475→   475→        }
   476→   476→      }
   477→   477→    }
   478→   478→
   479→   479→    return references;
   480→   480→  }

   550→   550→  private scoreContentDepth(output: DocOutput, suggestions: string[]): number {
   551→   551→    let totalSections = 0;
   552→   552→    let deepSections = 0;
   553→   553→
   554→   554→    const walk = (sections: DocSection[]): void => {
   555→   555→      for (const section of sections) {
   556→   556→        totalSections++;
   557→   557→        const wordCount = this.countWords(section.content);
   558→   558→        // Level 1–2 sections should have at least 150 words; levels 3–4 at least 75
   559→   559→        const minimum = section.level <= 2 ? 150 : 75;
   560→   560→        if (wordCount >= minimum) {
   561→   561→          deepSections++;
   562→   562→        } else {
   563→   563→          suggestions.push(
   564→   564→            `Section "${section.heading}" has only ${wordCount} words (minimum: ${minimum}). Consider expanding the content.`,
   565→   565→          );
   566→   566→        }
   567→   567→        if (section.children) walk(section.children);
   568→   568→      }
   569→   569→    };
   570→   570→
   571→   571→    walk(output.sections);
   572→   572→
   573→   573→    if (totalSections === 0) return 0;
   574→   574→    return deepSections / totalSections;
   575→   575→  }

   577→   577→  private scoreStructure(output: DocOutput, suggestions: string[]): number {
   578→   578→    let score = 1.0;
   579→   579→
   580→   580→    // Must have at least 2 top-level sections
   581→   581→    if (output.sections.length < 2) {
   582→   582→      score -= 0.3;
   583→   583→      suggestions.push('Document should have at least 2 top-level sections.');
   584→   584→    }
   585→   585→
   586→   586→    // Validate heading hierarchy — each section's level should not skip (e.g. 1 → 3)
   587→   587→    let prevLevel = 0;
   588→   588→    const checkHierarchy = (sections: DocSection[]): void => {
   589→   589→      for (const section of sections) {
   590→   590→        if (prevLevel > 0 && section.level > prevLevel + 1) {
   591→   591→          score -= 0.1;
   592→   592→          suggestions.push(
   593→   593→            `Heading hierarchy skip detected: level ${prevLevel} → ${section.level} at "${section.heading}".`,
   594→   594→          );
   595→   595→        }
   596→   596→        prevLevel = section.level;
   597→   597→        if (section.children) checkHierarchy(section.children);
   598→   598→      }
   599→   599→    };
   600→   600→    checkHierarchy(output.sections);
   601→   601→
   602→   602→    // TOC should match sections
   603→   603→    if (output.tableOfContents.length !== output.sections.length) {
   604→   604→      score -= 0.15;
   605→   605→      suggestions.push('Table of contents entry count does not match top-level sections.');
   606→   606→    }
   607→   607→
   608→   608→    return Math.max(0, Math.min(1, score));
   609→   609→  }

   611→   611→  private scoreFormatting(output: DocOutput, suggestions: string[]): number {
   612→   612→    let score = 1.0;
   613→   613→    const headingSet = new Set<string>();
   614→   614→    let duplicateHeadings = 0;
   615→   615→
   616→   616→    const walk = (sections: DocSection[]): void => {
   617→   617→      for (const section of sections) {
   618→   618→        if (headingSet.has(section.heading.toLowerCase())) {
   619→   619→          duplicateHeadings++;
   620→   620→        }
   621→   621→        headingSet.add(section.heading.toLowerCase());
   622→   622→
   623→   623→        // Check for empty content
   624→   624→        if (!section.content || section.content.trim().length === 0) {
   625→   625→          score -= 0.15;
   626→   626→          suggestions.push(`Section "${section.heading}" has empty content.`);
   627→   627→        }
   628→   628→
   629→   629→        // Validate tables have rows
   630→   630→        if (section.tables) {
   631→   631→          for (let i = 0; i < section.tables.length; i++) {
   632→   632→            const table = section.tables[i];
   633→   633→            if (table.rows.length === 0) {
   634→   634→              score -= 0.05;
   635→   635→              suggestions.push(`Table ${i + 1} in "${section.heading}" has no rows.`);
   636→   636→            }
   637→   637→            if (table.headers.length === 0) {
   638→   638→              score -= 0.05;
   639→   639→              suggestions.push(`Table ${i + 1} in "${section.heading}" has no headers.`);
   640→   640→            }
   641→   641→          }
   642→   642→        }
   643→   643→
   644→   644→        // Validate code blocks have code
   645→   645→        if (section.codeBlocks) {
   646→   646→          for (let i = 0; i < section.codeBlocks.length; i++) {
   647→   647→            if (!section.codeBlocks[i].code || section.codeBlocks[i].code.trim().length === 0) {
   648→   648→              score -= 0.05;
   649→   649→              suggestions.push(`Code block ${i + 1} in "${section.heading}" is empty.`);
   650→   650→            }
   651→   651→          }
   652→   652→        }
   653→   653→
   654→   654→        if (section.children) walk(section.children);
   655→   655→      }
   656→   656→    };
   657→   657→
   658→   658→    walk(output.sections);
   659→   659→
   660→   660→    if (duplicateHeadings > 0) {
   661→   661→      score -= 0.1 * duplicateHeadings;
   662→   662→      suggestions.push(`${duplicateHeadings} duplicate heading(s) detected. Use unique headings for each section.`);
   663→   663→    }
   664→   664→
   665→   665→    return Math.max(0, Math.min(1, score));
   666→   666→  }
   667→   667→
   668→   668→  private scoreCitations(output: DocOutput, suggestions: string[]): number {
   669→   669→    const refCount = output.references.length;
   670→   670→
   671→   671→    if (refCount === 0) {
   672→   672→      // Not necessarily a failure for shorter document types
   673→   673→      const docTypesRequiringRefs = ['whitepaper', 'report', 'specification', 'proposal'];
   674→   674→      if (docTypesRequiringRefs.includes(output.metadata.format)) {
   675→   675→        suggestions.push(`${output.metadata.format} documents typically require references. Consider adding sources.`);
   676→   676→        return 0.4;
   677→   677→      }
   678→   678→      return 0.8; // acceptable for memos, letters, etc.
   679→   679→    }
   680→   680→
   681→   681→    // Check reference completeness
   682→   682→    let incomplete = 0;
   683→   683→    for (const ref of output.references) {
   684→   684→      if (!ref.title || ref.title.trim().length === 0) incomplete++;
   685→   685→    }
   686→   686→
   687→   687→    if (incomplete > 0) {
   688→   688→      suggestions.push(`${incomplete} reference(s) have incomplete information.`);
   689→   689→    }
   690→   690→
   691→   691→    // Score based on completeness ratio
   692→   692→    const completeness = 1 - incomplete / refCount;
   693→   693→    // Bonus for having multiple well-formed references
   694→   694→    const countBonus = Math.min(refCount / 5, 1) * 0.2;
   695→   695→    return Math.min(1, completeness * 0.8 + countBonus);
   696→   696→  }

   698→   698→  private scoreReadability(output: DocOutput, suggestions: string[]): number {
   699→   699→    let totalWords = 0;
   700→   700→    let totalSentences = 0;
   701→   701→    let totalSyllables = 0;
   702→   702→
   703→   703→    const walk = (sections: DocSection[]): void => {
   704→   704→      for (const section of sections) {
   705→   705→        const text = section.content;
   706→   706→        totalWords += this.countWords(text);
   707→   707→        totalSentences += this.countSentences(text);
   708→   708→        totalSyllables += this.estimateSyllables(text);
   709→   709→
   710→   710→        if (section.children) walk(section.children);
   711→   711→      }
   712→   712→    };
   713→   713→
   714→   714→    walk(output.sections);
   715→   715→
   716→   716→    if (totalSentences === 0 || totalWords === 0) return 0;
   717→   717→
   718→   718→    // Flesch-Kincaid Grade Level (approximate)
   719→   719→    const fkGrade = 0.39 * (totalWords / totalSentences) + 11.8 * (totalSyllables / totalWords) - 15.59;
   720→   720→
   721→   721→    // Ideal grade level for professional documents: 8–12
   722→   722→    // Score 1.0 for grade 10, decreasing as we move away
   723→   723→    const idealGrade = 10;
   724→   724→    const deviation = Math.abs(fkGrade - idealGrade);
   725→   725→    const score = Math.max(0, 1 - deviation / 10);
   726→   726→
   727→   727→    if (fkGrade > 14) {
   728→   728→      suggestions.push('Document may be too complex. Consider simplifying sentence structures.');
   729→   729→    } else if (fkGrade < 6) {
   730→   730→      suggestions.push('Document readability is very low grade-level. Consider if the tone is appropriate for the audience.');
   731→   731→    }
   732→   732→
   733→   733→    return Math.min(1, Math.max(0, score));
   734→   734→  }
   735→   735→
   736→   736→  private countSentences(text: string): number {
   737→   737→    const sentences = text.split(/[.!?]+/).filter((s) => s.trim().length > 0);
   738→   738→    return Math.max(1, sentences.length);
   739→   739→  }
   740→   740→
   741→   741→  private estimateSyllables(text: string): number {
   742→   742→    // Simple heuristic: count vowel groups per word
   743→   743→    const words = text.split(/\s+/).filter((w) => w.length > 0);
   744→   744→    let total = 0;
   745→   745→    for (const word of words) {
   746→   746→      const vowelGroups = word.toLowerCase().replace(/[^a-z]/g, '').match(/[aeiouy]+/g);
   747→   747→      total += vowelGroups ? Math.max(1, vowelGroups.length) : 1;
   748→   748→    }
   749→   749→    return total;
   750→   750→  }

   802→   802→  private renderSectionMarkdown(section: DocSection): string {
   803→   803→    const lines: string[] = [];
   804→   804→    const headingPrefix = '#'.repeat(section.level + 1); // level 1 → ##, level 2 → ###, etc.
   805→   805→    lines.push(`${headingPrefix} ${section.heading}`);
   806→   806→    lines.push('');

   808→   808→    lines.push('');
   809→   809→
   810→   810→    // Tables
   811→   811→    if (section.tables) {
   812→   812→      for (const table of section.tables) {
   813→   813→        lines.push(this.renderTableMarkdown(table));
   814→   814→        lines.push('');
   815→   815→      }
   816→   816→    }
   817→   817→
   818→   818→    // Images
   819→   819→    if (section.images) {
   820→   820→      for (const image of section.images) {
   821→   821→        const alt = image.alt || 'image';
   822→   822→        const md = image.url ? `![${alt}](${image.url})` : `![${alt}]`;
   823→   823→        lines.push(md);
   824→   824→        if (image.caption) lines.push(`*${image.caption}*`);
   825→   825→        lines.push('');
   826→   826→      }
   827→   827→    }
   828→   828→
   829→   829→    // Code blocks
   830→   830→    if (section.codeBlocks) {
   831→   831→      for (const block of section.codeBlocks) {
   832→   832→        lines.push('```' + block.language);
   833→   833→        lines.push(block.code);
   834→   834→        lines.push('```');
   835→   835→        if (block.caption) lines.push(`*${block.caption}*`);
   836→   836→        lines.push('');
   837→   837→      }
   838→   838→    }
   839→   839→
   840→   840→    // Children
   841→   841→    if (section.children) {
   842→   842→      for (const child of section.children) {
   843→   843→        lines.push(this.renderSectionMarkdown(child));
   844→   844→      }
   845→   845→    }
   846→   846→
   847→   847→    return lines.join('\n');
   848→   848→  }

   907→   907→    lines.push('  <nav class="toc">');
   908→   908→    lines.push('    <h2>Table of Contents</h2>');
   909→   909→    lines.push(this.renderTocHtml(output.tableOfContents, 1));
   910→   910→    lines.push('  </nav>');
   911→   911→
   912→   912→    // Sections
   913→   913→    for (const section of output.sections) {
   914→   914→      lines.push(this.renderSectionHtml(section));
   915→   915→    }
   916→   916→
   917→   917→    // References
   918→   918→    if (output.references.length > 0) {
   919→   919→      lines.push('  <section>');
   920→   920→      lines.push('    <h2>References</h2>');
   921→   921→      lines.push('    <ol>');
   922→   922→      for (const ref of output.references) {
   923→   923→        lines.push(`      <li>${this.renderReferenceHtml(ref)}</li>`);
   924→   924→      }
   925→   925→      lines.push('    </ol>');
   926→   926→      lines.push('  </section>');
   927→   927→    }
   928→   928→
   929→   929→    // Metadata footer
   930→   930→    lines.push(`  <p class="metadata">Word count: ${output.metadata.wordCount} | Reading time: ~${output.metadata.readingTimeMinutes} min</p>`);
   931→   931→
   932→   932→    lines.push('</body>');
   933→   933→    lines.push('</html>');
   934→   934→
   935→   935→    return lines.join('\n');
   936→   936→  }

   938→   938→  private renderTocHtml(entries: TocEntry[], depth: number): string {
   939→   939→    const indent = '    '.repeat(depth);
   940→   940→    const lines: string[] = [];
   941→   941→    lines.push(`${indent}<ul>`);
   942→   942→    for (const entry of entries) {
   943→   943→      lines.push(`${indent}  <li><a href="#${entry.id}">${this.escapeHtml(entry.title)}</a>`);
   944→   944→      if (entry.children && entry.children.length > 0) {
   945→   945→        lines.push(this.renderTocHtml(entry.children, depth + 1));
   946→   946→      }
   947→   947→      lines.push(`${indent}  </li>`);
   948→   948→    }
   949→   949→    lines.push(`${indent}</ul>`);
   950→   950→    return lines.join('\n');
   951→   951→  }

   953→   953→  private renderSectionHtml(section: DocSection): string {
   954→   954→    const tag = `h${section.level + 1}`; // level 1 → h2, level 2 → h3, etc.
   955→   955→    const lines: string[] = [];
   956→   956→    lines.push(`  <section id="${section.id}">`);
   957→   957→    lines.push(`    <${tag}>${this.escapeHtml(section.heading)}</${tag}>`);
   958→   958→    lines.push(`    <p>${this.escapeHtml(section.content)}</p>`);
   959→   959→
   960→   960→    // Tables
   961→   961→    if (section.tables) {
   962→   962→      for (const table of section.tables) {
   963→   963→        lines.push(this.renderTableHtml(table));
   964→   964→      }
   965→   965→    }
   966→   966→
   967→   967→    // Images
   968→   968→    if (section.images) {
   969→   969→      for (const image of section.images) {
   970→   970→        const src = image.url ?? '';
   971→   971→        const alt = this.escapeHtml(image.alt || 'image');
   972→   972→        lines.push(`    <figure>`);
   973→   973→        lines.push(`      <img src="${src}" alt="${alt}" />`);
   974→   974→        if (image.caption) {
   975→   975→          lines.push(`      <figcaption class="caption">${this.escapeHtml(image.caption)}</figcaption>`);
   976→   976→        }
   977→   977→        lines.push(`    </figure>`);
   978→   978→      }
   979→   979→    }
   980→   980→
   981→   981→    // Code blocks
   982→   982→    if (section.codeBlocks) {
   983→   983→      for (const block of section.codeBlocks) {
   984→   984→        lines.push(`    <pre><code class="language-${this.escapeHtml(block.language)}">${this.escapeHtml(block.code)}</code></pre>`);
   985→   985→        if (block.caption) {
   986→   986→          lines.push(`    <p class="caption">${this.escapeHtml(block.caption)}</p>`);
   987→   987→        }
   988→   988→      }
   989→   989→    }
   990→   990→
   991→   991→    // Children
   992→   992→    if (section.children) {
   993→   993→      for (const child of section.children) {
   994→   994→        lines.push(this.renderSectionHtml(child));
   995→   995→      }
   996→   996→    }
   997→   997→
   998→   998→    lines.push('  </section>');
   999→   999→    return lines.join('\n');
  1000→  1000→  }

  1002→  1002→  private renderTableHtml(table: DocTable): string {
  1003→  1003→    const lines: string[] = [];
  1004→  1004→    lines.push('    <table>');
  1005→  1005→    lines.push('      <thead>');
  1006→  1006→    lines.push('        <tr>' + table.headers.map((h: string) => `<th>${this.escapeHtml(h)}</th>`).join('') + '</tr>');
  1007→  1007→    lines.push('      </thead>');
  1008→  1008→    lines.push('      <tbody>');
  1009→  1009→    for (const row of table.rows) {
  1010→  1010→      lines.push('        <tr>' + row.map((cell: string) => `<td>${this.escapeHtml(cell)}</td>`).join('') + '</tr>');
  1011→  1011→    }
  1012→  1012→    lines.push('      </tbody>');
  1013→  1013→    lines.push('    </table>');
  1014→  1014→    if (table.caption) {
  1015→  1015→      lines.push(`    <p class="caption">${this.escapeHtml(table.caption)}</p>`);
  1016→  1016→    }
  1017→  1017→    return lines.join('\n');
  1018→  1018→  }

  1020→  1020→  private renderReferenceHtml(ref: DocReference): string {
  1021→  1021→    const parts: string[] = [];
  1022→  1022→    if (ref.authors && ref.authors.length > 0) parts.push(this.escapeHtml(ref.authors.join(', ')));
  1023→  1023→    if (ref.year) parts.push(`(${ref.year})`);
  1024→  1024→    parts.push(`<em>${this.escapeHtml(ref.title)}</em>`);
  1025→  1025→    if (ref.url) parts.push(`<a href="${this.escapeHtml(ref.url)}">Link</a>`);
  1026→  1026→    return parts.join(' ');
  1027→  1027→  }

  1029→  1029→  private escapeHtml(text: string): string {

  1031→  1031→      .replace(/&/g, '&amp;')
  1032→  1032→      .replace(/</g, '&lt;')
  1033→  1033→      .replace(/>/g, '&gt;')
  1034→  1034→      .replace(/"/g, '&quot;')
  1035→  1035→      .replace(/'/g, '&#039;');
  1036→  1036→  }

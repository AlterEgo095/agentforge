     2→// AgentForge - DATA Agent Worker
     3→// Handles data analysis, visualization, and spreadsheet generation.
     4→// Extends BaseAgentWorker with a pipeline that ingests data,
     5→// performs statistical analysis, detects anomalies, generates
     6→// chart specifications, and produces actionable insights.
     7→// ============================================================
     8→
     9→import { BaseAgentWorker, type QualityReport } from './BaseAgentWorker';
    10→import { v4 as uuidv4 } from 'uuid';

    12→// ─── Type Definitions ──────────────────────────────────────
    13→
    14→export interface DataSummary {
    15→  overview: string;
    16→  keyMetrics: Record<string, number>;
    17→  anomalies: string[];
    18→  recommendations: string[];
    19→}
    20→
    21→export interface ColumnDef {
    22→  name: string;
    23→  type: 'string' | 'number' | 'date' | 'boolean' | 'object';
    24→  nullable: boolean;
    25→  description?: string;
    26→}
    27→
    28→export interface Dataset {
    29→  id: string;
    30→  name: string;
    31→  columns: ColumnDef[];
    32→  rows: Record<string, unknown>[];
    33→  rowCount: number;
    34→}
    35→
    36→export interface ChartSeries {
    37→  name: string;
    38→  data: (number | string)[];
    39→  type?: string;
    40→  color?: string;
    41→}
    42→
    43→export interface ChartSpec {
    44→  id: string;
    45→  type: 'bar' | 'line' | 'pie' | 'scatter' | 'area' | 'heatmap' | 'histogram' | 'boxplot' | 'radar';
    46→  title: string;
    47→  xAxis?: { label: string; field: string };
    48→  yAxis?: { label: string; field: string };
    49→  series: ChartSeries[];
    50→  options: Record<string, unknown>;
    51→  insight: string;
    52→}
    53→
    54→export type DataInsightType = 'trend' | 'anomaly' | 'correlation' | 'outlier' | 'pattern' | 'forecast';
    55→
    56→export interface DataInsight {
    57→  id: string;
    58→  type: DataInsightType;
    59→  title: string;
    60→  description: string;
    61→  confidence: number;
    62→  affectedColumns: string[];
    63→  evidence: string;
    64→}
    65→
    66→export interface StatisticalSummary {
    67→  column: string;
    68→  count: number;
    69→  mean?: number;
    70→  median?: number;
    71→  stdDev?: number;
    72→  min?: number | string;

    74→  uniqueValues?: number;
    75→  topValues?: { value: string; count: number }[];
    76→}
    77→
    78→export interface DataOutput {

    80→  summary: DataSummary;

    86→    rowCount: number;
    87→    columnCount: number;
    88→    dataTypes: Record<string, string>;
    89→    qualityScore: number;

   100→  // ─── Core Pipeline ──────────────────────────────────────
   101→
   102→  protected async generate(
   103→    prompt: string,
   104→    context: Record<string, unknown>,
   105→    sessionId: string,
   106→  ): Promise<DataOutput> {
   107→    // Phase 1: Data Ingestion
   108→    this.emitEvent('data_ingest', sessionId, { status: 'started' });
   109→    const dataset = this.parseDataSource(prompt, context);
   110→    this.emitEvent('data_ingest', sessionId, {
   111→      status: 'completed',
   112→      rowCount: dataset.rowCount,
   113→      columnCount: dataset.columns.length,
   114→    });
   115→
   116→    // Phase 2: Statistical Analysis
   117→    this.emitEvent('data_analyze', sessionId, { status: 'started' });
   118→    const statistics = this.computeStatistics(dataset);
   119→    const anomalies = this.detectAnomalies(dataset, statistics);
   120→    this.emitEvent('data_analyze', sessionId, {
   121→      status: 'completed',
   122→      columnsAnalyzed: statistics.length,
   123→      anomaliesFound: anomalies.length,
   124→    });
   125→
   126→    // Phase 3: Visualization
   127→    this.emitEvent('data_visualize', sessionId, { status: 'started' });
   128→    const charts = this.recommendCharts(dataset, statistics);
   129→    this.emitEvent('data_visualize', sessionId, {
   130→      status: 'completed',
   131→      chartsGenerated: charts.length,
   132→    });
   133→
   134→    // Phase 4: Insights Generation
   135→    this.emitEvent('data_insights', sessionId, { status: 'started' });
   136→    const insights = this.generateInsights(dataset, statistics, charts);
   137→    this.emitEvent('data_insights', sessionId, {
   138→      status: 'completed',
   139→      insightsCount: insights.length,
   140→    });
   141→
   142→    // Assemble output
   143→    const dataTypes: Record<string, string> = {};
   144→    for (const col of dataset.columns) {
   145→      dataTypes[col.name] = col.type;
   146→    }
   147→
   148→    const qualityScore = this.assessDataQuality(dataset, statistics);
   149→
   150→    const summary: DataSummary = {
   151→      overview: this.buildOverview(dataset, statistics, insights),
   152→      keyMetrics: this.extractKeyMetrics(statistics),
   153→      anomalies: anomalies,
   154→      recommendations: this.buildRecommendations(dataset, statistics, anomalies),
   155→    };
   156→
   157→    return {
   158→      title: this.inferTitle(prompt),
   159→      summary,
   160→      datasets: [dataset],
   161→      charts,
   162→      insights,
   163→      statistics,
   164→      metadata: {
   165→        rowCount: dataset.rowCount,
   166→        columnCount: dataset.columns.length,
   167→        dataTypes,
   168→        qualityScore,
   169→      },
   170→    };
   171→  }

   173→  // ─── Validation ─────────────────────────────────────────
   174→
   175→  protected async validateOutput(output: DataOutput, prompt: string): Promise<QualityReport> {
   176→    const dimensions: Record<string, number> = {};
   177→    const suggestions: string[] = [];
   178→
   179→    // Accuracy (0-1): Check statistical calculations are consistent
   180→    dimensions.Accuracy = this.scoreAccuracy(output, suggestions);
   181→
   182→    // Visualization (0-1): Validate chart specs match data types
   183→    dimensions.Visualization = this.scoreVisualization(output, suggestions);
   184→
   185→    // Insight Depth (0-1): Validate insights are substantive
   186→    dimensions['Insight Depth'] = this.scoreInsightDepth(output, suggestions);
   187→
   188→    // Code Quality (0-1): Internal pipeline quality
   189→    dimensions['Code Quality'] = this.scoreCodeQuality(output, suggestions);
   190→
   191→    // Documentation (0-1): Check column descriptions, chart insights
   192→    dimensions.Documentation = this.scoreDocumentation(output, suggestions);
   193→
   194→    const overall = this.computeWeightedScore(dimensions);
   195→    const passed = this.isQualityPassing(overall);
   196→
   197→    return { overall, dimensions, passed, suggestions };
   198→  }

   202→  protected async renderOutput(output: DataOutput, format?: string): Promise<unknown> {
   203→    switch (format) {
   204→      case 'charts':
   205→        return output.charts;

   225→  // ─── Helper: Data Source Parsing ────────────────────────
   226→
   227→  private parseDataSource(prompt: string, context?: Record<string, unknown>): Dataset {
   228→    // Check if context provides a dataset directly
   229→    if (context?.dataset && typeof context.dataset === 'object') {
   230→      const ds = context.dataset as Dataset;
   231→      if (ds.id && ds.columns && ds.rows) {
   232→        return ds;
   233→      }
   234→    }
   235→
   236→    // Attempt to parse inline CSV from the prompt
   237→    const csvMatch = prompt.match(/```csv\n([\s\S]*?)```/);
   238→    if (csvMatch) {
   239→      return this.parseCsv(csvMatch[1]);
   240→    }
   241→
   242→    // Attempt to parse inline JSON from the prompt
   243→    const jsonMatch = prompt.match(/```json\n([\s\S]*?)```/);
   244→    if (jsonMatch) {
   245→      try {
   246→        const parsed = JSON.parse(jsonMatch[1]);
   247→        if (Array.isArray(parsed)) {
   248→          return this.parseJsonArray(parsed);
   249→        }
   250→      } catch {
   251→        // Fall through to sample generation
   252→      }
   253→    }
   254→
   255→    // Attempt to parse raw JSON array in the prompt
   256→    const jsonArrayMatch = prompt.match(/\[[\s\S]*\]/);
   257→    if (jsonArrayMatch) {
   258→      try {
   259→        const parsed = JSON.parse(jsonArrayMatch[0]);
   260→        if (Array.isArray(parsed) && parsed.length > 0 && typeof parsed[0] === 'object') {
   261→          return this.parseJsonArray(parsed);
   262→        }
   263→      } catch {
   264→        // Fall through to sample generation
   265→      }
   266→    }
   267→
   268→    // Generate a sample dataset from the prompt description
   269→    return this.generateSampleDataset(prompt);
   270→  }
   271→
   272→  private parseCsv(csvText: string): Dataset {
   273→    const lines = csvText.trim().split('\n').map(line => line.trim()).filter(Boolean);
   274→    if (lines.length === 0) {
   275→      return this.generateSampleDataset('empty CSV');
   276→    }
   277→
   278→    const headers = lines[0].split(',').map(h => h.trim().replace(/^"|"$/g, ''));
   279→    const rows: Record<string, unknown>[] = [];
   280→
   281→    for (let i = 1; i < lines.length; i++) {
   282→      const values = this.splitCsvLine(lines[i]);
   283→      const row: Record<string, unknown> = {};
   284→      for (let j = 0; j < headers.length; j++) {
   285→        const raw = j < values.length ? values[j] : null;
   286→        row[headers[j]] = this.castValue(raw);
   287→      }
   288→      rows.push(row);
   289→    }
   290→
   291→    const columns = headers.map(name => {
   292→      const values = rows.map(r => r[name]);
   293→      return {
   294→        name,
   295→        type: this.inferColumnType(values) as ColumnDef['type'],
   296→        nullable: values.some(v => v === null || v === undefined || v === ''),
   297→        description: undefined,
   298→      };
   299→    });
   300→
   301→    return {
   302→      id: uuidv4(),
   303→      name: 'Parsed CSV',
   304→      columns,
   305→      rows,
   306→      rowCount: rows.length,
   307→    };
   308→  }
   309→
   310→  private parseJsonArray(data: Record<string, unknown>[]): Dataset {
   311→    if (data.length === 0) {
   312→      return this.generateSampleDataset('empty JSON array');
   313→    }
   314→
   315→    const keys = Object.keys(data[0]);
   316→    const rows = data.map(item => {
   317→      const row: Record<string, unknown> = {};
   318→      for (const key of keys) {
   319→        row[key] = item[key] ?? null;
   320→      }
   321→      return row;
   322→    });
   323→
   324→    const columns = keys.map(name => {
   325→      const values = rows.map(r => r[name]);
   326→      return {
   327→        name,
   328→        type: this.inferColumnType(values) as ColumnDef['type'],
   329→        nullable: values.some(v => v === null || v === undefined),
   330→        description: undefined,
   331→      };
   332→    });
   333→
   334→    return {
   335→      id: uuidv4(),
   336→      name: 'Parsed JSON',
   337→      columns,
   338→      rows,
   339→      rowCount: rows.length,
   340→    };
   341→  }

   400→  // ─── Helper: Statistical Analysis ───────────────────────
   401→
   402→  private computeStatistics(dataset: Dataset): StatisticalSummary[] {
   403→    return dataset.columns.map(col => {
   404→      const values = dataset.rows
   405→        .map(r => r[col.name])
   406→        .filter(v => v !== null && v !== undefined && v !== '');
   407→
   408→      if (col.type === 'number') {
   409→        const numValues = values
   410→          .map(v => Number(v))
   411→          .filter(n => !isNaN(n));
   412→
   413→        const count = numValues.length;
   414→        const mean = count > 0 ? numValues.reduce((a, b) => a + b, 0) / count : undefined;
   415→        const sorted = [...numValues].sort((a, b) => a - b);
   416→        const median = count > 0
   417→          ? count % 2 === 0
   418→            ? (sorted[count / 2 - 1] + sorted[count / 2]) / 2
   419→            : sorted[Math.floor(count / 2)]
   420→          : undefined;
   421→        const variance = count > 1
   422→          ? numValues.reduce((sum, v) => sum + Math.pow(v - (mean ?? 0), 2), 0) / (count - 1)
   423→          : undefined;
   424→        const stdDev = variance !== undefined ? Math.sqrt(variance) : undefined;
   425→
   426→        return {
   427→          column: col.name,
   428→          count,
   429→          mean,
   430→          median,
   431→          stdDev,
   432→          min: count > 0 ? sorted[0] : undefined,
   433→          max: count > 0 ? sorted[sorted.length - 1] : undefined,
   434→        };
   435→      }
   436→
   437→      // Categorical / other columns
   438→      const strValues = values.map(String);
   439→      const freqMap = new Map<string, number>();
   440→      for (const v of strValues) {
   441→        freqMap.set(v, (freqMap.get(v) ?? 0) + 1);
   442→      }
   443→
   444→      const topValues = Array.from(freqMap.entries())
   445→        .sort((a, b) => b[1] - a[1])
   446→        .slice(0, 5)
   447→        .map(([value, count]) => ({ value, count }));
   448→
   449→      return {
   450→        column: col.name,
   451→        count: strValues.length,
   452→        uniqueValues: freqMap.size,
   453→        topValues,
   454→        min: topValues.length > 0 ? topValues[topValues.length - 1].value : undefined,
   455→        max: topValues.length > 0 ? topValues[0].value : undefined,
   456→      };
   457→    });
   458→  }

   460→  // ─── Helper: Anomaly Detection ──────────────────────────
   461→
   462→  private detectAnomalies(dataset: Dataset, stats: StatisticalSummary[]): string[] {
   463→    const anomalies: string[] = [];
   464→
   465→    for (const stat of stats) {
   466→      // Z-score based outlier detection for numeric columns
   467→      if (stat.mean !== undefined && stat.stdDev !== undefined && stat.stdDev > 0) {
   468→        const col = dataset.columns.find(c => c.name === stat.column);
   469→        if (!col || col.type !== 'number') continue;
   470→
   471→        const outlierRows: number[] = [];
   472→        dataset.rows.forEach((row, idx) => {
   473→          const val = Number(row[col.name]);
   474→          if (!isNaN(val)) {
   475→            const zScore = Math.abs(val - stat.mean!) / stat.stdDev!;
   476→            if (zScore > 3) {
   477→              outlierRows.push(idx + 1);
   478→            }
   479→          }
   480→        });
   481→
   482→        if (outlierRows.length > 0) {
   483→          anomalies.push(
   484→            `Column "${stat.column}" has ${outlierRows.length} outlier(s) at row(s) ${outlierRows.slice(0, 10).join(', ')} (z-score > 3)`,
   485→          );
   486→        }
   487→      }
   488→
   489→      // High null-rate detection
   490→      const col = dataset.columns.find(c => c.name === stat.column);
   491→      if (col && dataset.rowCount > 0) {
   492→        const missingCount = dataset.rowCount - stat.count;
   493→        const missingRate = missingCount / dataset.rowCount;
   494→        if (missingRate > 0.2) {
   495→          anomalies.push(
   496→            `Column "${stat.column}" has ${(missingRate * 100).toFixed(1)}% missing values (${missingCount}/${dataset.rowCount})`,
   497→          );
   498→        }
   499→      }
   500→
   501→      // Low cardinality detection for expected-high-cardinality columns
   502→      if (col?.type === 'string' && stat.uniqueValues !== undefined && dataset.rowCount > 10) {
   503→        const cardinalityRatio = stat.uniqueValues / dataset.rowCount;
   504→        if (cardinalityRatio < 0.05 && stat.uniqueValues <= 2 && stat.uniqueValues > 0) {
   505→          anomalies.push(
   506→            `Column "${stat.column}" has very low cardinality (${stat.uniqueValues} unique values across ${dataset.rowCount} rows)`,
   507→          );
   508→        }
   509→      }
   510→    }
   511→
   512→    return anomalies;
   513→  }

   515→  // ─── Helper: Chart Recommendation ────────────────────────
   516→
   517→  private recommendCharts(dataset: Dataset, stats: StatisticalSummary[]): ChartSpec[] {
   518→    const charts: ChartSpec[] = [];
   519→    const numericCols = dataset.columns.filter(c => c.type === 'number');
   520→    const stringCols = dataset.columns.filter(c => c.type === 'string');
   521→
   522→    // Bar chart: first string column vs first numeric column
   523→    if (stringCols.length >= 1 && numericCols.length >= 1) {
   524→      const xCol = stringCols[0];
   525→      const yCol = numericCols[0];
   526→      const seriesData = dataset.rows.map(r => Number(r[yCol.name]) || 0);
   527→      const categories = dataset.rows.map(r => String(r[xCol.name]));
   528→
   529→      charts.push({
   530→        id: uuidv4(),
   531→        type: 'bar',
   532→        title: `${yCol.name} by ${xCol.name}`,
   533→        xAxis: { label: xCol.name, field: xCol.name },
   534→        yAxis: { label: yCol.name, field: yCol.name },
   535→        series: [{ name: yCol.name, data: seriesData }],
   536→        options: {
   537→          color: ['#F59E0B'],
   538→          tooltip: { trigger: 'axis' },
   539→          xAxis: { type: 'category', data: categories },
   540→          yAxis: { type: 'value' },
   541→        },
   542→        insight: `Shows the distribution of ${yCol.name} across ${xCol.name} categories`,
   543→      });
   544→    }
   545→
   546→    // Line chart: if there are multiple numeric columns (time series feel)
   547→    if (numericCols.length >= 2) {
   548→      const xCol = stringCols[0] ?? numericCols[0];
   549→      const series: ChartSeries[] = numericCols.slice(0, 4).map((col, idx) => ({
   550→        name: col.name,
   551→        data: dataset.rows.map(r => Number(r[col.name]) || 0),
   552→        color: this.getChartColor(idx),
   553→      }));
   554→      const categories = dataset.rows.map(r => String(r[xCol.name]));
   555→
   556→      charts.push({
   557→        id: uuidv4(),
   558→        type: 'line',
   559→        title: `Trends: ${numericCols.slice(0, 4).map(c => c.name).join(', ')}`,
   560→        xAxis: { label: xCol.name, field: xCol.name },
   561→        yAxis: { label: 'Value', field: 'value' },
   562→        series,
   563→        options: {
   564→          tooltip: { trigger: 'axis' },
   565→          legend: { data: series.map(s => s.name) },
   566→          xAxis: { type: 'category', data: categories },
   567→          yAxis: { type: 'value' },
   568→        },
   569→        insight: `Tracks trends across ${numericCols.slice(0, 4).map(c => c.name).join(', ')} over ${xCol.name}`,
   570→      });
   571→    }
   572→
   573→    // Pie chart: distribution of first string column
   574→    if (stringCols.length >= 1) {
   575→      const col = stringCols[0];
   576→      const freqMap = new Map<string, number>();
   577→      for (const row of dataset.rows) {
   578→        const val = String(row[col.name]);
   579→        freqMap.set(val, (freqMap.get(val) ?? 0) + 1);
   580→      }
   581→      const pieData = Array.from(freqMap.entries()).map(([name, value]) => ({ name, value }));
   582→      const topPie = pieData.sort((a, b) => b.value - a.value).slice(0, 10);
   583→
   584→      charts.push({
   585→        id: uuidv4(),
   586→        type: 'pie',
   587→        title: `Distribution of ${col.name}`,
   588→        series: [{
   589→          name: col.name,
   590→          data: topPie.map(p => p.value),
   591→        }],
   592→        options: {
   593→          tooltip: { trigger: 'item' },
   594→          legend: { orient: 'vertical', left: 'left' },
   595→          series: [{
   596→            type: 'pie',
   597→            radius: '60%',
   598→            data: topPie.map(p => ({ name: p.name, value: p.value })),
   599→          }],
   600→        },
   601→        insight: `Shows the proportional distribution of ${col.name} values`,
   602→      });
   603→    }
   604→
   605→    // Scatter chart: if 2+ numeric columns exist
   606→    if (numericCols.length >= 2) {
   607→      const xCol = numericCols[0];
   608→      const yCol = numericCols[1];
   609→
   610→      charts.push({
   611→        id: uuidv4(),
   612→        type: 'scatter',
   613→        title: `${yCol.name} vs ${xCol.name}`,
   614→        xAxis: { label: xCol.name, field: xCol.name },
   615→        yAxis: { label: yCol.name, field: yCol.name },
   616→        series: [{
   617→          name: `${xCol.name} vs ${yCol.name}`,
   618→          data: dataset.rows.map(r => [Number(r[xCol.name]) || 0, Number(r[yCol.name]) || 0]).flat() as (number | string)[],
   619→        }],
   620→        options: {
   621→          tooltip: { trigger: 'item' },
   622→          xAxis: { type: 'value', name: xCol.name },
   623→          yAxis: { type: 'value', name: yCol.name },
   624→          series: [{
   625→            type: 'scatter',
   626→            data: dataset.rows.map(r => [Number(r[xCol.name]) || 0, Number(r[yCol.name]) || 0]),
   627→          }],
   628→        },
   629→        insight: `Reveals correlation patterns between ${xCol.name} and ${yCol.name}`,
   630→      });
   631→    }
   632→
   633→    // Histogram for each numeric column
   634→    for (const col of numericCols.slice(0, 2)) {
   635→      const numValues = dataset.rows
   636→        .map(r => Number(r[col.name]))
   637→        .filter(n => !isNaN(n));
   638→
   639→      if (numValues.length < 3) continue;
   640→
   641→      const min = Math.min(...numValues);
   642→      const max = Math.max(...numValues);
   643→      const binCount = Math.min(10, Math.ceil(Math.sqrt(numValues.length)));
   644→      const binWidth = (max - min) / binCount || 1;
   645→      const bins = Array(binCount).fill(0);
   646→      const binLabels: string[] = [];
   647→
   648→      for (let i = 0; i < binCount; i++) {
   649→        const low = min + i * binWidth;
   650→        const high = low + binWidth;
   651→        binLabels.push(`${low.toFixed(1)}-${high.toFixed(1)}`);
   652→      }
   653→
   654→      for (const v of numValues) {
   655→        let binIdx = Math.floor((v - min) / binWidth);
   656→        if (binIdx >= binCount) binIdx = binCount - 1;
   657→        bins[binIdx]++;
   658→      }
   659→
   660→      charts.push({
   661→        id: uuidv4(),
   662→        type: 'histogram',
   663→        title: `Distribution of ${col.name}`,
   664→        xAxis: { label: col.name, field: col.name },
   665→        yAxis: { label: 'Frequency', field: 'count' },
   666→        series: [{ name: col.name, data: bins }],
   667→        options: {
   668→          tooltip: { trigger: 'axis' },
   669→          xAxis: { type: 'category', data: binLabels },
   670→          yAxis: { type: 'value' },
   671→          series: [{ type: 'bar', data: bins, barWidth: '90%' }],
   672→        },
   673→        insight: `Shows the frequency distribution of ${col.name} values across ${binCount} bins`,
   674→      });
   675→    }
   676→
   677→    return charts;
   678→  }

   680→  // ─── Helper: Insight Generation ──────────────────────────

   682→  private generateInsights(
   683→    dataset: Dataset,
   684→    stats: StatisticalSummary[],
   685→    charts: ChartSpec[],
   686→  ): DataInsight[] {
   687→    const insights: DataInsight[] = [];
   688→    const numericStats = stats.filter(s => s.mean !== undefined);
   689→
   690→    // Trend insights
   691→    for (const stat of numericStats) {
   692→      const col = dataset.columns.find(c => c.name === stat.column);
   693→      if (!col || col.type !== 'number') continue;
   694→
   695→      const values = dataset.rows
   696→        .map(r => Number(r[col.name]))
   697→        .filter(n => !isNaN(n));
   698→
   699→      if (values.length < 3) continue;
   700→
   701→      // Simple linear trend
   702→      const firstHalf = values.slice(0, Math.floor(values.length / 2));
   703→      const secondHalf = values.slice(Math.floor(values.length / 2));
   704→      const firstAvg = firstHalf.reduce((a, b) => a + b, 0) / firstHalf.length;
   705→      const secondAvg = secondHalf.reduce((a, b) => a + b, 0) / secondHalf.length;
   706→
   707→      const percentChange = firstAvg !== 0 ? ((secondAvg - firstAvg) / Math.abs(firstAvg)) * 100 : 0;
   708→
   709→      if (Math.abs(percentChange) > 10) {
   710→        insights.push({
   711→          id: uuidv4(),
   712→          type: 'trend',
   713→          title: `${percentChange > 0 ? 'Upward' : 'Downward'} trend in ${stat.column}`,
   714→          description: `${stat.column} shows a ${Math.abs(percentChange).toFixed(1)}% ${percentChange > 0 ? 'increase' : 'decrease'} from the first half to the second half of the data (avg: ${firstAvg.toFixed(2)} → ${secondAvg.toFixed(2)})`,
   715→          confidence: Math.min(0.95, 0.6 + Math.abs(percentChange) / 200),
   716→          affectedColumns: [stat.column],
   717→          evidence: `First half avg: ${firstAvg.toFixed(2)}, Second half avg: ${secondAvg.toFixed(2)}, Change: ${percentChange.toFixed(1)}%`,
   718→        });
   719→      }
   720→    }
   721→
   722→    // Correlation insights (Pearson)
   723→    for (let i = 0; i < numericStats.length; i++) {
   724→      for (let j = i + 1; j < numericStats.length; j++) {
   725→        const colA = numericStats[i].column;
   726→        const colB = numericStats[j].column;
   727→
   728→        const correlation = this.pearsonCorrelation(dataset, colA, colB);
   729→        if (correlation !== null && Math.abs(correlation) > 0.7) {
   730→          insights.push({
   731→            id: uuidv4(),
   732→            type: 'correlation',
   733→            title: `Strong ${correlation > 0 ? 'positive' : 'negative'} correlation: ${colA} & ${colB}`,
   734→            description: `${colA} and ${colB} have a Pearson correlation of ${correlation.toFixed(3)}, indicating a strong ${correlation > 0 ? 'positive' : 'negative'} relationship`,
   735→            confidence: Math.min(0.95, 0.5 + Math.abs(correlation) / 2),
   736→            affectedColumns: [colA, colB],
   737→            evidence: `Pearson r = ${correlation.toFixed(3)}`,
   738→          });
   739→        }
   740→      }
   741→    }
   742→
   743→    // Outlier insights
   744→    for (const stat of numericStats) {
   745→      if (stat.stdDev === undefined || stat.stdDev === 0 || stat.mean === undefined) continue;
   746→
   747→      const col = dataset.columns.find(c => c.name === stat.column);
   748→      if (!col) continue;
   749→
   750→      const outlierCount = dataset.rows.filter(r => {
   751→        const val = Number(r[col.name]);
   752→        return !isNaN(val) && Math.abs(val - stat.mean!) / stat.stdDev! > 3;
   753→      }).length;
   754→
   755→      if (outlierCount > 0) {
   756→        insights.push({
   757→          id: uuidv4(),
   758→          type: 'outlier',
   759→          title: `Outliers detected in ${stat.column}`,
   760→          description: `${outlierCount} value(s) in ${stat.column} are more than 3 standard deviations from the mean (${stat.mean.toFixed(2)} ± ${stat.stdDev.toFixed(2)})`,
   761→          confidence: 0.85,
   762→          affectedColumns: [stat.column],
   763→          evidence: `Mean: ${stat.mean.toFixed(2)}, StdDev: ${stat.stdDev.toFixed(2)}, Outliers: ${outlierCount}`,
   764→        });
   765→      }
   766→    }
   767→
   768→    // Pattern insights — value distribution for categorical columns
   769→    const categoricalStats = stats.filter(s => s.topValues && s.topValues.length > 0);
   770→    for (const stat of categoricalStats) {
   771→      if (!stat.topValues || stat.uniqueValues === undefined) continue;
   772→
   773→      const topVal = stat.topValues[0];
   774→      const dominanceRatio = topVal.count / stat.count;
   775→
   776→      if (dominanceRatio > 0.5 && stat.uniqueValues > 2) {
   777→        insights.push({
   778→          id: uuidv4(),
   779→          type: 'pattern',
   780→          title: `Dominant value in ${stat.column}`,
   781→          description: `"${topVal.value}" accounts for ${(dominanceRatio * 100).toFixed(1)}% of ${stat.column} values (${topVal.count}/${stat.count}), potentially skewing analysis`,
   782→          confidence: 0.75,
   783→          affectedColumns: [stat.column],
   784→          evidence: `Top value: "${topVal.value}" (${topVal.count} occurrences), ${stat.uniqueValues} unique values total`,
   785→        });
   786→      }
   787→    }
   788→
   789→    return insights;
   790→  }

   794→  private inferColumnType(values: unknown[]): 'string' | 'number' | 'date' | 'boolean' | 'object' {
   795→    const nonNull = values.filter(v => v !== null && v !== undefined && v !== '');
   796→    if (nonNull.length === 0) return 'string';
   797→
   798→    // Check boolean
   799→    const boolStrings = new Set(['true', 'false', 'yes', 'no', '0', '1']);
   800→    if (nonNull.every(v => boolStrings.has(String(v).toLowerCase()))) {
   801→      return 'boolean';
   802→    }
   803→
   804→    // Check number
   805→    if (nonNull.every(v => !isNaN(Number(v)))) {
   806→      return 'number';
   807→    }
   808→
   809→    // Check date
   810→    const datePattern = /^\d{4}[-/]\d{1,2}[-/]\d{1,2}/;
   811→    if (nonNull.every(v => datePattern.test(String(v)))) {
   812→      return 'date';
   813→    }
   814→
   815→    // Check object
   816→    if (nonNull.every(v => typeof v === 'object' && !Array.isArray(v))) {
   817→      return 'object';
   818→    }
   819→
   820→    return 'string';
   821→  }

   825→  private scoreAccuracy(output: DataOutput, suggestions: string[]): number {
   826→    let score = 1.0;
   827→    let checks = 0;
   828→
   829→    // Check that statistical means fall between min and max
   830→    for (const stat of output.statistics) {
   831→      if (stat.mean !== undefined && stat.min !== undefined && stat.max !== undefined) {
   832→        checks++;
   833→        const mean = stat.mean;
   834→        const min = typeof stat.min === 'number' ? stat.min : -Infinity;
   835→        const max = typeof stat.max === 'number' ? stat.max : Infinity;
   836→        if (mean < min || mean > max) {
   837→          score -= 0.3;
   838→          suggestions.push(`Statistical inconsistency: mean of "${stat.column}" (${mean.toFixed(2)}) is outside min/max range [${min}, ${max}]`);
   839→        }
   840→      }
   841→
   842→      // Check stdDev is non-negative
   843→      if (stat.stdDev !== undefined) {
   844→        checks++;
   845→        if (stat.stdDev < 0) {
   846→          score -= 0.3;
   847→          suggestions.push(`Invalid stdDev for "${stat.column}": cannot be negative`);
   848→        }
   849→      }
   850→    }
   851→
   852→    // Check dataset rowCount matches actual rows
   853→    for (const ds of output.datasets) {
   854→      checks++;
   855→      if (ds.rowCount !== ds.rows.length) {
   856→        score -= 0.2;
   857→        suggestions.push(`Dataset "${ds.name}" rowCount (${ds.rowCount}) doesn't match actual rows (${ds.rows.length})`);
   858→      }
   859→    }
   860→
   861→    // Check metadata consistency
   862→    checks++;
   863→    const primaryDs = output.datasets[0];
   864→    if (primaryDs) {
   865→      if (output.metadata.rowCount !== primaryDs.rowCount) {
   866→        score -= 0.15;
   867→        suggestions.push('Metadata rowCount does not match primary dataset rowCount');
   868→      }
   869→      if (output.metadata.columnCount !== primaryDs.columns.length) {
   870→        score -= 0.15;
   871→        suggestions.push('Metadata columnCount does not match primary dataset column count');
   872→      }
   873→    }
   874→
   875→    return checks > 0 ? Math.max(0, Math.min(1, score)) : 0.5;
   876→  }

  1078→  private splitCsvLine(line: string): string[] {
  1079→    const result: string[] = [];
  1080→    let current = '';
  1081→    let inQuotes = false;
  1082→
  1083→    for (let i = 0; i < line.length; i++) {
  1084→      const ch = line[i];
  1085→      if (inQuotes) {
  1086→        if (ch === '"' && line[i + 1] === '"') {
  1087→          current += '"';
  1088→          i++;
  1089→        } else if (ch === '"') {
  1090→          inQuotes = false;
  1091→        } else {
  1092→          current += ch;
  1093→        }
  1094→      } else {
  1095→        if (ch === '"') {
  1096→          inQuotes = true;
  1097→        } else if (ch === ',') {
  1098→          result.push(current.trim());
  1099→          current = '';
  1100→        } else {
  1101→          current += ch;
  1102→        }
  1103→      }
  1104→    }
  1105→    result.push(current.trim());
  1106→    return result;
  1107→  }

  1118→  private pearsonCorrelation(dataset: Dataset, colA: string, colB: string): number | null {
  1119→    const pairs = dataset.rows
  1120→      .map(r => [Number(r[colA]), Number(r[colB])])
  1121→      .filter(([a, b]) => !isNaN(a) && !isNaN(b));
  1122→
  1123→    const n = pairs.length;
  1124→    if (n < 3) return null;
  1125→
  1126→    const sumA = pairs.reduce((s, [a]) => s + a, 0);
  1127→    const sumB = pairs.reduce((s, [, b]) => s + b, 0);
  1128→    const sumAB = pairs.reduce((s, [a, b]) => s + a * b, 0);
  1129→    const sumA2 = pairs.reduce((s, [a]) => s + a * a, 0);
  1130→    const sumB2 = pairs.reduce((s, [, b]) => s + b * b, 0);
  1131→
  1132→    const numerator = n * sumAB - sumA * sumB;
  1133→    const denominator = Math.sqrt((n * sumA2 - sumA * sumA) * (n * sumB2 - sumB * sumB));
  1134→
  1135→    if (denominator === 0) return null;
  1136→    return numerator / denominator;
  1137→  }

  1139→  private assessDataQuality(dataset: Dataset, stats: StatisticalSummary[]): number {
  1140→    let score = 1.0;
  1141→
  1142→    for (const col of dataset.columns) {
  1143→      const stat = stats.find(s => s.column === col.name);
  1144→      if (!stat) continue;
  1145→
  1146→      // Penalize missing data
  1147→      const completeness = stat.count / dataset.rowCount;
  1148→      score *= completeness;
  1149→    }
  1150→
  1151→    return Math.round(score * 100) / 100;
  1152→  }

  1186→  private buildRecommendations(
  1187→    dataset: Dataset,
  1188→    stats: StatisticalSummary[],
  1189→    anomalies: string[],
  1190→  ): string[] {
  1191→    const recs: string[] = [];
  1192→
  1193→    if (anomalies.length > 0) {
  1194→      recs.push(`Investigate ${anomalies.length} detected anomaly/anomalies before drawing conclusions`);
  1195→    }
  1196→
  1197→    const highNullCols = dataset.columns.filter(col => {
  1198→      const stat = stats.find(s => s.column === col.name);
  1199→      return stat && stat.count < dataset.rowCount * 0.8;
  1200→    });
  1201→
  1202→    if (highNullCols.length > 0) {
  1203→      recs.push(`Consider imputing or removing columns with high null rates: ${highNullCols.map(c => c.name).join(', ')}`);
  1204→    }
  1205→
  1206→    const numericCols = dataset.columns.filter(c => c.type === 'number');
  1207→    if (numericCols.length >= 3) {
  1208→      recs.push('With multiple numeric columns, consider multivariate analysis for deeper insights');
  1209→    }
  1210→
  1211→    if (dataset.rowCount < 30) {
  1212→      recs.push('Dataset is small — statistical significance may be limited; consider collecting more data');
  1213→    }
  1214→
  1215→    if (recs.length === 0) {
  1216→      recs.push('Data quality looks good — proceed with deeper domain-specific analysis');
  1217→    }
  1218→
  1219→    return recs;
  1220→  }

    30→// All admin routes now use RBAC requirePermission() instead of
    31→// the old tier-based adminAuth middleware.
    32→//
    33→// Old: admin.use('*', adminAuth)  →  tier === 'enterprise'

    44→// GET /admin/users - List all users
    45→admin.get('/users', requirePermission({ resource: 'users', action: 'read' }), async (c) => {
    46→  try {
    47→    const limit = parseInt(c.req.query('limit') || '50');
    48→    const offset = parseInt(c.req.query('offset') || '0');
    49→
    50→    const userList = await db
    51→      .select({
    52→        id: users.id,
    53→        email: users.email,
    54→        name: users.name,
    55→        tier: users.tier,
    56→        mfaEnabled: users.mfaEnabled,
    57→        createdAt: users.createdAt,
    58→        updatedAt: users.updatedAt,
    59→      })
    60→      .from(users)
    61→      .orderBy(desc(users.createdAt))
    62→      .limit(Math.min(limit, 100))
    63→      .offset(offset);
    64→
    65→    const total = await db.select({ count: count() }).from(users);
    66→
    67→    return c.json({
    68→      success: true,
    69→      data: {
    70→        users: userList,
    71→        total: total[0]?.count || 0,
    72→        limit,
    73→        offset,
    74→      },
    75→    });
    76→  } catch (error) {
    77→    console.error('[Admin] Failed to list users:', error);
    78→    return c.json({ success: false, error: 'Failed to list users' }, 500);
    79→  }
    80→});

    82→// PATCH /admin/users - Update user (e.g., change tier)
    83→const UpdateUserSchema = z.object({
    84→  tier: z.enum(['free', 'pro', 'enterprise']).optional(),
    85→  name: z.string().min(1).max(255).optional(),
    86→});
    87→
    88→admin.patch('/users', requirePermission({ resource: 'users', action: 'update' }), zValidator('json', UpdateUserSchema), async (c) => {
    89→  const data = c.req.valid('json');
    90→  const targetUserId = c.req.query('userId');
    91→
    92→  if (!targetUserId) {
    93→    return c.json({ success: false, error: 'userId query parameter required' }, 400);
    94→  }
    95→
    96→  try {
    97→    const result = await db
    98→      .update(users)
    99→      .set({ ...data, updatedAt: new Date() })
   100→      .where(eq(users.id, targetUserId))
   101→      .returning({
   102→        id: users.id,
   103→        email: users.email,
   104→        name: users.name,
   105→        tier: users.tier,
   106→      });
   107→
   108→    if (!result.length) {
   109→      return c.json({ success: false, error: 'User not found' }, 404);
   110→    }
   111→
   112→    // Audit: admin user update — P2.1.1: No secrets in audit
   113→    await auditTrailService.record({
   114→      action: 'admin.user_update',
   115→      actorId: c.get('userId'),
   116→      actorType: 'admin',
   117→      resourceType: 'user',
   118→      resourceId: targetUserId,
   119→      details: { updatedFields: Object.keys(data) },
   120→      ipAddress: c.req.header('X-Forwarded-For')?.split(',')[0]?.trim() || 'unknown',
   121→      userAgent: c.req.header('User-Agent') || 'unknown',
   122→      success: true,
   123→      riskLevel: 'medium',
   124→    });
   125→
   126→    return c.json({ success: true, data: result[0] });
   127→  } catch (error) {
   128→    console.error('[Admin] Failed to update user:', error);
   129→    return c.json({ success: false, error: 'Failed to update user' }, 500);
   130→  }
   131→});

   135→  const userId = c.req.param('id');
   136→
   137→  try {

   139→      .select({
   140→        id: users.id,

   151→      .limit(1);
   152→
   153→    if (!userRows.length) {
   154→      return c.json({ success: false, error: 'User not found' }, 404);
   155→    }
   156→
   157→    // Get user's session count and cost stats
   158→    const sessionCount = await db
   159→      .select({ count: count() })
   160→      .from(generationSessions)

   169→      .where(eq(costTracking.userId, userId));

   171→    return c.json({
   172→      success: true,
   173→      data: {
   174→        user: userRows[0],

   176→          sessions: sessionCount[0]?.count || 0,
   177→          totalCost: costStats[0]?.totalCost || '0',
   178→          totalTokens: costStats[0]?.totalTokens || '0',

   188→// DELETE /admin/users/:id - Delete user (P2.1.2: requires 'manage' permission — highest privilege)
   189→admin.delete('/users/:id', requirePermission({ resource: 'users', action: 'manage' }), async (c) => {
   190→  const userId = c.req.param('id');
   191→
   192→  try {
   193→    // Safety check: don't allow self-deletion
   194→    const actorId = c.get('userId');
   195→    if (actorId === userId) {
   196→      return c.json({ success: false, error: 'Cannot delete your own account' }, 400);
   197→    }
   198→
   199→    const result = await db.delete(users).where(eq(users.id, userId)).returning({ id: users.id });
   200→
   201→    if (!result.length) {
   202→      return c.json({ success: false, error: 'User not found' }, 404);
   203→    }
   204→
   205→    // Audit: admin user deletion
   206→    await auditTrailService.record({
   207→      action: 'admin.user_delete',
   208→      actorId,
   209→      actorType: 'admin',
   210→      resourceType: 'user',
   211→      resourceId: userId,
   212→      details: {},
   213→      ipAddress: c.req.header('X-Forwarded-For')?.split(',')[0]?.trim() || 'unknown',
   214→      userAgent: c.req.header('User-Agent') || 'unknown',
   215→      success: true,
   216→      riskLevel: 'high',
   217→    });
   218→
   219→    return c.json({ success: true, message: 'User deleted' });
   220→  } catch (error) {
   221→    console.error('[Admin] Failed to delete user:', error);
   222→    return c.json({ success: false, error: 'Failed to delete user' }, 500);
   223→  }
   224→});

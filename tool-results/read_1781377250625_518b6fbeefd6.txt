    85→const CAN_SPAM_RULES = [
    86→  { rule: 'has_unsubscribe', check: (html: string, text: string) => /unsubscribe/i.test(html) || /unsubscribe/i.test(text) },
    87→  { rule: 'has_physical_address', check: (html: string, text: string) => /{{physicalAddress}}/i.test(html) || /address/i.test(html) || /address/i.test(text) },
    88→  { rule: 'no_deceptive_subject', check: (_html: string, _text: string, subject: string) => !/^(re:|fwd:)/i.test(subject.trim()) },
    89→  { rule: 'has_sender_identity', check: (html: string, text: string) => /from|sent by|on behalf of/i.test(html) || /from|sent by/i.test(text) },
    90→  { rule: 'subject_under_50_chars', check: (_html: string, _text: string, subject: string) => subject.length <= 50 },
    91→  { rule: 'no_spam_trigger_words', check: (_html: string, _text: string, subject: string) => {
    92→    const spamTriggers = ['free', 'act now', 'limited time', 'congratulations', 'you\'re a winner', 'cash', 'bonus', 'no obligation', 'risk-free', 'click here now'];
    93→    const lowerSubject = subject.toLowerCase();
    94→    return !spamTriggers.some(trigger => lowerSubject.includes(trigger));
    95→  }},

   105→  // ─── Core Pipeline ─────────────────────────────────────
   106→
   107→  protected async generate(
   108→    prompt: string,
   109→    context: Record<string, unknown>,
   110→    sessionId: string,
   111→  ): Promise<EmailOutput> {
   112→    // Phase 1: Parse email purpose and audience from prompt
   113→    const purpose = this.parseEmailPurpose(prompt);
   114→    const tone = this.detectTone(prompt);
   115→    const topic = this.extractTopic(prompt);
   116→
   117→    this.emitEvent('email_context', sessionId, { purpose, tone, topic });
   118→
   119→    // Phase 2: Generate subject line with best practices
   120→    const subject = this.generateSubject(topic, purpose, tone);
   121→
   122→    // Phase 3: Create preheader text
   123→    const preheader = this.generatePreheader(topic, purpose, tone);
   124→
   125→    // Phase 4: Generate HTML body with email-safe CSS
   126→    const bodyHtml = this.generateHtmlBody(subject, topic, purpose, tone, context);
   127→
   128→    // Phase 5: Create plain text alternative
   129→    const bodyText = this.generatePlainText(bodyHtml);
   130→
   131→    const primaryDraft: EmailDraft = { subject, preheader, bodyHtml, bodyText };
   132→
   133→    this.emitEvent('email_draft', sessionId, {
   134→      subject,
   135→      preheader,
   136→      wordCount: this.countWords(bodyText),
   137→      purpose,
   138→      tone,
   139→    });
   140→
   141→    // Phase 6: Generate A/B variants
   142→    const variants = this.generateVariants(primaryDraft, purpose);
   143→
   144→    this.emitEvent('email_variants', sessionId, {
   145→      variantCount: variants.length,
   146→      variantLabels: variants.map(v => v.label),
   147→    });
   148→
   149→    // Phase 7: Extract merge fields from body content
   150→    const mergeFields = this.extractMergeFields(bodyHtml, bodyText);
   151→
   152→    // Phase 8: Run CAN-SPAM compliance checks
   153→    const complianceChecks = this.checkCompliance({
   154→      subject,
   155→      preheader,
   156→      bodyHtml,
   157→      bodyText,
   158→      variants,
   159→      mergeFields,
   160→      metadata: { wordCount: 0, readingTimeSeconds: 0, tone, purpose, complianceChecks: [] },
   161→    });
   162→
   163→    this.emitEvent('email_compliance', sessionId, {
   164→      totalChecks: complianceChecks.length,
   165→      passed: complianceChecks.filter(c => c.passed).length,
   166→      failed: complianceChecks.filter(c => !c.passed).length,
   167→    });
   168→
   169→    // Phase 9: Compute metadata
   170→    const wordCount = this.countWords(bodyText);
   171→    const readingTimeSeconds = Math.max(1, Math.round((wordCount / 238) * 60)); // 238 wpm → seconds
   172→
   173→    return {
   174→      subject,
   175→      preheader,
   176→      bodyHtml,
   177→      bodyText,
   178→      variants,
   179→      mergeFields,
   180→      metadata: {
   181→        wordCount,
   182→        readingTimeSeconds,
   183→        tone,
   184→        purpose,
   185→        complianceChecks,
   186→      },
   187→    };
   188→  }

   190→  // ─── Validation ────────────────────────────────────────
   191→
   192→  protected async validateOutput(output: EmailOutput, _prompt: string): Promise<QualityReport> {
   193→    const dimensions: Record<string, number> = {};
   194→    const suggestions: string[] = [];
   195→
   196→    // 1. Clarity (0–1): Subject clarity, body readability
   197→    dimensions['Clarity'] = this.scoreClarity(output, suggestions);
   198→
   199→    // 2. Persuasion (0–1): CTA presence, value proposition
   200→    dimensions['Persuasion'] = this.scorePersuasion(output, suggestions);
   201→
   202→    // 3. Tone Match (0–1): Consistent tone throughout
   203→    dimensions['Tone Match'] = this.scoreToneMatch(output, suggestions);
   204→
   205→    // 4. Actionability (0–1): Clear CTA, next steps
   206→    dimensions['Actionability'] = this.scoreActionability(output, suggestions);
   207→
   208→    // 5. Conciseness (0–1): Word count appropriateness
   209→    dimensions['Conciseness'] = this.scoreConciseness(output, suggestions);
   210→
   211→    const overall = this.computeWeightedScore(dimensions);
   212→    const passed = this.isQualityPassing(overall);
   213→
   214→    return { overall, dimensions, passed, suggestions };
   215→  }

   254→  // ─── Helper: Email Purpose Parsing ─────────────────────

   256→  /**
   257→   * Detect the email purpose/type from the user prompt by matching
   258→   * against known keyword sets. Falls back to "announcement".
   259→   */
   260→  parseEmailPurpose(prompt: string): string {
   261→    const lower = prompt.toLowerCase();
   262→    let bestMatch = 'announcement';
   263→    let bestScore = 0;
   264→
   265→    for (const [purpose, keywords] of Object.entries(EMAIL_PURPOSE_KEYWORDS)) {
   266→      let score = 0;
   267→      for (const keyword of keywords) {
   268→        if (lower.includes(keyword)) {
   269→          score += keyword.length; // longer keyword matches are stronger signals
   270→        }
   271→      }
   272→      if (score > bestScore) {
   273→        bestScore = score;
   274→        bestMatch = purpose;
   275→      }
   276→    }
   277→
   278→    return bestMatch;
   279→  }

   283→  /**
   284→   * Infer tone from the language in the prompt. Falls back to "professional".
   285→   */
   286→  detectTone(prompt: string): string {
   287→    const lower = prompt.toLowerCase();
   288→    let bestTone = 'professional';
   289→    let bestScore = 0;
   290→
   291→    for (const [tone, keywords] of Object.entries(TONE_KEYWORDS)) {
   292→      let score = 0;
   293→      for (const keyword of keywords) {
   294→        if (lower.includes(keyword)) {
   295→          score += keyword.length;
   296→        }
   297→      }
   298→      if (score > bestScore) {
   299→        bestScore = score;
   300→        bestTone = tone;
   301→      }
   302→    }
   303→
   304→    return bestTone;
   305→  }

   510→  /**
   511→   * Generate the body content paragraphs based on purpose.
   512→   */
   513→  private generateBodyContent(
   514→    topic: string,
   515→    purpose: string,
   516→    tone: string,
   517→    context: Record<string, unknown>,
   518→  ): string {
   519→    // If previous output feedback is available, incorporate it
   520→    if (context.previousOutput && typeof context.previousOutput === 'object') {
   521→      const prev = context.previousOutput as Record<string, unknown>;
   522→      if (prev.bodyText && typeof prev.bodyText === 'string') {
   523→        return this.paragraphsToHtml(prev.bodyText);
   524→      }
   525→    }
   526→
   527→    // If quality feedback is available, use it to improve
   528→    if (context.qualityFeedback && Array.isArray(context.qualityFeedback)) {
   529→      // Intentionally incorporate feedback signals into content
   530→      const feedbackStr = (context.qualityFeedback as string[]).join('. ');
   531→      return this.buildPurposeContent(topic, purpose, tone, feedbackStr);
   532→    }
   533→
   534→    return this.buildPurposeContent(topic, purpose, tone);
   535→  }
   536→
   537→  private buildPurposeContent(topic: string, purpose: string, tone: string, feedbackHint?: string): string {
   538→    const paragraphs: string[] = [];
   539→
   540→    switch (purpose) {
   541→      case 'sales':
   542→        paragraphs.push(`I wanted to reach out because I believe ${topic} could make a real impact for {{companyName}}.`);
   543→        paragraphs.push(`Many teams like yours have seen measurable results by leveraging ${topic} — from increased efficiency to better outcomes across the board.`);
   544→        paragraphs.push(`I'd love to share how ${topic} can specifically benefit your team and answer any questions you might have.`);
   545→        break;
   546→      case 'newsletter':
   547→        paragraphs.push(`Here's your latest roundup of updates and insights on ${topic}.`);

   595→  // ─── Helper: Plain Text Generation ─────────────────────
   596→
   597→  /**
   598→   * Strip HTML tags and decode entities to create a clean plain text
   599→   * alternative for email clients that don't render HTML.
   600→   */
   601→  generatePlainText(htmlBody: string): string {
   602→    let text = htmlBody;
   603→
   604→    // Remove style and head sections entirely
   605→    text = text.replace(/<style[^>]*>[\s\S]*?<\/style>/gi, '');
   606→    text = text.replace(/<head[^>]*>[\s\S]*?<\/head>/gi, '');
   607→
   608→    // Convert <br> and block elements to newlines
   609→    text = text.replace(/<br\s*\/?>/gi, '\n');
   610→    text = text.replace(/<\/p>/gi, '\n\n');
   611→    text = text.replace(/<\/tr>/gi, '\n');
   612→    text = text.replace(/<\/div>/gi, '\n');
   613→    text = text.replace(/<\/h[1-6]>/gi, '\n\n');
   614→
   615→    // Convert links: <a href="url">text</a> → text (url)
   616→    text = text.replace(/<a[^>]*href="([^"]*)"[^>]*>([^<]*)<\/a>/gi, '$2 ($1)');
   617→
   618→    // Convert horizontal rules
   619→    text = text.replace(/<hr[^>]*\/?>/gi, '---');
   620→
   621→    // Remove all remaining HTML tags
   622→    text = text.replace(/<[^>]+>/g, '');
   623→
   624→    // Decode common HTML entities
   625→    text = text.replace(/&amp;/g, '&');
   626→    text = text.replace(/&lt;/g, '<');
   627→    text = text.replace(/&gt;/g, '>');
   628→    text = text.replace(/&quot;/g, '"');
   629→    text = text.replace(/&#039;/g, "'");
   630→    text = text.replace(/&nbsp;/g, ' ');
   631→
   632→    // Collapse excessive whitespace
   633→    text = text.replace(/[ \t]+/g, ' ');
   634→    text = text.replace(/\n{3,}/g, '\n\n');
   635→
   636→    return text.trim();
   637→  }

   811→          : this.getComplianceFailMessage(rule.rule),
   812→      });

   814→
   815→    // Also check all variants

   824→          });
   825→        }
   826→      }
   827→    }
   828→
   829→    return checks;
   830→  }

   856→  // ─── Helper: Merge Field Extraction ────────────────────

   858→  /**
   859→   * Find all {{field}} placeholders in the HTML and text bodies
   860→   * and return them as MergeField definitions with defaults.
   861→   */
   862→  extractMergeFields(bodyHtml: string, bodyText: string): MergeField[] {
   863→    const fieldPattern = /\{\{(\w+)\}\}/g;
   864→    const foundFields = new Set<string>();
   865→    let match: RegExpExecArray | null;
   866→
   867→    // Scan HTML body
   868→    while ((match = fieldPattern.exec(bodyHtml)) !== null) {
   869→      foundFields.add(match[1]);
   870→    }
   871→
   872→    // Scan plain text body
   873→    fieldPattern.lastIndex = 0;
   874→    while ((match = fieldPattern.exec(bodyText)) !== null) {
   875→      foundFields.add(match[1]);
   876→    }
   877→
   878→    // Map to MergeField with defaults and descriptions
   879→    const knownFields: Record<string, { defaultValue: string; description: string }> = {
   880→      'firstName':        { defaultValue: 'Friend', description: 'Recipient\'s first name' },
   881→      'lastName':         { defaultValue: '', description: 'Recipient\'s last name' },
   882→      'companyName':      { defaultValue: 'your company', description: 'Recipient\'s company name' },
   883→      'senderName':       { defaultValue: 'Our Team', description: 'Name of the email sender' },
   884→      'physicalAddress':  { defaultValue: '123 Main St, City, State 00000', description: 'Physical mailing address (CAN-SPAM requirement)' },
   885→      'unsubscribeUrl':   { defaultValue: '#unsubscribe', description: 'URL for unsubscribe action' },
   886→      'preferencesUrl':   { defaultValue: '#preferences', description: 'URL for email preferences center' },
   887→      'ctaUrl':           { defaultValue: '#cta', description: 'URL for the primary call-to-action button' },
   888→      'lastPurchase':     { defaultValue: 'your recent order', description: 'Name or description of last purchase' },
   889→      'productName':      { defaultValue: 'our product', description: 'Name of the product being referenced' },
   890→      'eventDate':        { defaultValue: 'TBD', description: 'Date of the referenced event' },
   891→      'eventLocation':    { defaultValue: 'Online', description: 'Location of the referenced event' },
   892→      'discountCode':     { defaultValue: 'SAVE20', description: 'Promotional discount code' },
   893→      'expirationDate':   { defaultValue: 'soon', description: 'Offer expiration date' },
   894→    };
   895→
   896→    const mergeFields: MergeField[] = [];
   897→
   898→    const fieldList = Array.from(foundFields);
   899→    for (const field of fieldList) {
   900→      const known = knownFields[field];
   901→      mergeFields.push({
   902→        name: field,
   903→        defaultValue: known?.defaultValue ?? `[${field}]`,
   904→        description: known?.description ?? `Custom merge field: ${field}`,
   905→      });
   906→    }
   907→
   908→    // Ensure required compliance fields are always present
   909→    const requiredFields = ['unsubscribeUrl', 'preferencesUrl', 'physicalAddress', 'senderName', 'companyName'];
   910→    for (const required of requiredFields) {
   911→      if (!foundFields.has(required)) {
   912→        const known = knownFields[required];
   913→        mergeFields.push({
   914→          name: required,
   915→          defaultValue: known?.defaultValue ?? `[${required}]`,
   916→          description: known?.description ?? `Required field: ${required}`,
   917→        });
   918→      }
   919→    }
   920→
   921→    return mergeFields;
   922→  }

   990→  // ─── Quality Scoring Helpers ───────────────────────────
   991→
   992→  private scoreClarity(output: EmailOutput, suggestions: string[]): number {
   993→    let score = 1.0;
   994→
   995→    // Subject should be present and clear
   996→    if (!output.subject || output.subject.trim().length === 0) {
   997→      score -= 0.4;
   998→      suggestions.push('Email subject line is empty. A clear subject is essential.');
   999→    }
  1000→
  1001→    // Subject should not be too long (over 50 chars risks truncation)
  1002→    if (output.subject.length > 50) {
  1003→      score -= 0.15;
  1004→      suggestions.push(`Subject line is ${output.subject.length} characters. Keep it under 50 for optimal display.`);
  1005→    }
  1006→
  1007→    // Preheader should be present
  1008→    if (!output.preheader || output.preheader.trim().length === 0) {
  1009→      score -= 0.2;
  1010→      suggestions.push('Preheader text is empty. This preview text impacts open rates.');
  1011→    }
  1012→
  1013→    // Body should have sufficient content (at least 20 words)
  1014→    const wordCount = this.countWords(output.bodyText);
  1015→    if (wordCount < 20) {
  1016→      score -= 0.3;
  1017→      suggestions.push(`Email body has only ${wordCount} words. Consider adding more content.`);
  1018→    }
  1019→
  1020→    // Body should not be excessively long (over 500 words for most emails)
  1021→    if (wordCount > 500) {
  1022→      score -= 0.1;

  1024→    }

  1026→    return Math.max(0, Math.min(1, score));

  1029→  private scorePersuasion(output: EmailOutput, suggestions: string[]): number {

  1071→  private scoreToneMatch(output: EmailOutput, suggestions: string[]): number {
  1072→    let score = 1.0;
  1073→    const declaredTone = output.metadata.tone;
  1074→
  1075→    const toneSignals: Record<string, { present: RegExp[]; absent: RegExp[] }> = {
  1076→      'formal': {
  1077→        present: [/dear\s+/i, /respectfully/i, /sincerely/i, /regards/i],
  1078→        absent: [/hey/i, /cheers/i, /lol/i, /btw/i],
  1079→      },
  1080→      'friendly': {
  1081→        present: [/hey/i, /hi/i, /cheers/i, /hope\s+you/i, /excited/i],
  1082→        absent: [/pursuant/i, /hereby/i, /notwithstanding/i],
  1083→      },
  1084→      'urgent': {
  1085→        present: [/urgent/i, /immediately/i, /asap/i, /act\s+now/i, /time[\s-]sensitive/i],
  1086→        absent: [/whenever/i, /at\s+your\s+convenience/i, /no\s+rush/i],
  1087→      },
  1088→      'casual': {
  1089→        present: [/hi\s+/i, /hey/i, /thanks/i, /btw/i, /quick/i],
  1090→        absent: [/dear\s+/i, /pursuant/i, /hereby/i, /regards/i],
  1091→      },
  1092→      'professional': {
  1093→        present: [/hello/i, /best\s+regards/i, /please/i, /thank\s+you/i],
  1094→        absent: [/hey/i, /lol/i, /omg/i, /btw/i],
  1095→      },
  1096→    };
  1097→
  1098→    const signals = toneSignals[declaredTone];
  1099→    if (!signals) {
  1100→      // Unknown tone — give a moderate score
  1101→      return 0.6;
  1102→    }
  1103→
  1104→    const bodyText = output.bodyText;
  1105→
  1106→    // Check for expected tone signals
  1107→    let presentCount = 0;
  1108→    for (const pattern of signals.present) {
  1109→      if (pattern.test(bodyText)) presentCount++;
  1110→    }
  1111→    const presenceRatio = signals.present.length > 0 ? presentCount / signals.present.length : 0;
  1112→    if (presenceRatio < 0.25) {
  1113→      score -= 0.2;
  1114→      suggestions.push(`Email tone is declared as "${declaredTone}" but few ${declaredTone} language signals are present.`);
  1115→    }
  1116→
  1117→    // Check for tone-inappropriate signals
  1118→    let absentViolations = 0;
  1119→    for (const pattern of signals.absent) {
  1120→      if (pattern.test(bodyText)) absentViolations++;
  1121→    }
  1122→    if (absentViolations > 0) {
  1123→      score -= 0.15 * absentViolations;
  1124→      suggestions.push(`Some language in the email conflicts with the "${declaredTone}" tone.`);
  1125→    }
  1126→
  1127→    // Check consistency between subject and body
  1128→    const subjectTone = this.detectTone(output.subject);
  1129→    if (subjectTone !== declaredTone && declaredTone !== 'professional') {
  1130→      score -= 0.1;
  1131→      suggestions.push(`Subject tone (${subjectTone}) doesn't match the declared body tone (${declaredTone}).`);
  1132→    }
  1133→
  1134→    return Math.max(0, Math.min(1, score));
  1135→  }

  1137→  private scoreActionability(output: EmailOutput, suggestions: string[]): number {
  1138→    let score = 0.3; // Base score
  1139→
  1140→    const bodyLower = output.bodyHtml.toLowerCase();

  1144→    if (hasCtaButton) {

  1178→  private scoreConciseness(output: EmailOutput, suggestions: string[]): number {
  1179→    const wordCount = this.countWords(output.bodyText);
  1180→
  1181→    // Ideal email length: 50–200 words (sweet spot for engagement)
  1182→    // Acceptable: 30–300
  1183→    if (wordCount < 30) {
  1184→      suggestions.push(`Email is very short (${wordCount} words). May lack sufficient context.`);
  1185→      return 0.4;
  1186→    }
  1187→
  1188→    if (wordCount >= 50 && wordCount <= 200) {
  1189→      return 1.0; // Ideal range
  1190→    }
  1191→
  1192→    if (wordCount >= 30 && wordCount < 50) {
  1193→      const score = 0.6 + 0.4 * ((wordCount - 30) / 20);
  1194→      suggestions.push('Email is slightly short. Consider adding a bit more detail.');
  1195→      return Math.min(1, score);
  1196→    }
  1197→
  1198→    if (wordCount > 200 && wordCount <= 300) {
  1199→      const score = 0.6 + 0.4 * ((300 - wordCount) / 100);
  1200→      suggestions.push('Email is on the longer side. Consider trimming for better engagement.');
  1201→      return Math.max(0, score);
  1202→    }
  1203→
  1204→    if (wordCount > 300) {
  1205→      suggestions.push(`Email is very long (${wordCount} words). Most effective emails are 50–200 words.`);
  1206→      return Math.max(0, 0.5 - (wordCount - 300) / 1000);
  1207→    }
  1208→
  1209→    return 0.5;
  1210→  }

  1214→  private paragraphsToHtml(plainText: string): string {

    15→// We access the pluginLoader lazily via the global singleton set
    16→// by PluginBootstrap at server startup. This avoids a hard import
    17→// that would fail at test time when plugin packages aren't available.
    18→
    19→function getPluginLoader(): { getCapability<T = unknown>(name: string): T | undefined } | null {
    20→  try {
    21→    // eslint-disable-next-line @typescript-eslint/no-require-imports
    22→    const { pluginLoader } = require('../../services/PluginBootstrap');
    23→    return pluginLoader;
    24→  } catch {
    25→    // PluginBootstrap not available (e.g. in tests without plugin packages)
    26→    return null;
    27→  }
    28→}

   137→  // ─── Core Pipeline: generate() ─────────────────────────
   138→
   139→  protected async generate(
   140→    prompt: string,
   141→    context: Record<string, unknown>,
   142→    sessionId: string,
   143→  ): Promise<RechercheOutput> {
   144→    // ── Try ResearchWorker plugin delegation first ──
   145→    try {
   146→      const loader = getPluginLoader();
   147→      const researchCap = loader?.getCapability<IResearchCapability>('research');
   148→      if (researchCap && typeof researchCap.research === 'function') {
   149→        console.log(`[RechercheAgentWorker] Delegating to ResearchWorker plugin (session: ${sessionId})`);
   150→        this.emitEvent('research_plugin_delegation', sessionId, { plugin: 'research-worker' });
   151→
   152→        const researchConfig = adaptRechercheConfigToResearchConfig(context);
   153→        const researchSession = await researchCap.research(prompt, researchConfig);
   154→        const rechercheOutput = adaptResearchSessionToRechercheOutput(researchSession);
   155→
   156→        this.emitEvent('research_report', sessionId, {
   157→          title: rechercheOutput.title,
   158→          sectionCount: rechercheOutput.sections.length,
   159→          sourceCount: rechercheOutput.sources.length,
   160→          findingCount: rechercheOutput.findings.length,
   161→          confidenceLevel: rechercheOutput.metadata.confidenceLevel,
   162→        });
   163→
   164→        console.log(`[RechercheAgentWorker] ResearchWorker plugin succeeded (session: ${sessionId}, sources: ${rechercheOutput.sources.length}, findings: ${rechercheOutput.findings.length})`);
   165→        return rechercheOutput;
   166→      }
   167→    } catch (err) {
   168→      console.warn(
   169→        `[RechercheAgentWorker] ResearchWorker plugin failed, falling back to internal pipeline (session: ${sessionId}):`,
   170→        err instanceof Error ? err.message : err,
   171→      );
   172→    }
   173→
   174→    // ── Internal Pipeline (fallback) ──
   175→    console.log(`[RechercheAgentWorker] Using internal pipeline (session: ${sessionId})`);
   176→
   177→    // ── Step 1: Parse & Classify the Query ──
   178→    this.emitEvent('research_query', sessionId, { query: prompt });
   179→    const classification = this.classifyQuery(prompt);
   180→
   181→    // ── Step 2: Plan Search Strategy ──
   182→    const depth = typeof context.depth === 'number' ? context.depth : 3;
   183→    const strategy = this.planSearchStrategy(classification, depth);
   184→
   185→    // ── Step 3: Simulate Web Search (generate structured search results) ──
   186→    this.emitEvent('research_search', sessionId, {
   187→      engines: strategy.engines,
   188→      keywords: strategy.keywords,
   189→      depth: strategy.depth,
   190→    });
   191→    const sources = this.generateSources(prompt, strategy);
   192→
   193→    // ── Step 4: Extract Findings from Sources ──
   194→    this.emitEvent('research_extract', sessionId, {
   195→      sourceCount: sources.length,
   196→    });
   197→    const findings = this.extractFindings(sources);
   198→
   199→    // ── Step 5: Cross-Reference & Compare Sources ──
   200→    const comparisons = this.compareSources(sources);
   201→
   202→    // ── Step 6: Synthesize — Executive Summary & Sections ──
   203→    this.emitEvent('research_synthesize', sessionId, {
   204→      findingCount: findings.length,
   205→      comparisonCount: comparisons.length,
   206→    });
   207→    const executiveSummary = this.synthesizeSummary(prompt, classification, findings, comparisons);
   208→    const sections = this.synthesizeSections(prompt, classification, sources, findings, comparisons);
   209→
   210→    // ── Step 7: Identify Knowledge Gaps ──
   211→    const gaps = this.identifyGaps(findings, sources);
   212→
   213→    // ── Step 8: Format References in Multiple Citation Styles ──
   214→    const references = this.generateCitations(sources, 'all');
   215→
   216→    // ── Step 9: Assemble Output & Emit Report ──
   217→    const confidenceLevel = this.computeConfidenceLevel(findings, sources);
   218→
   219→    const output: RechercheOutput = {
   220→      title: this.generateTitle(prompt, classification),
   221→      query: prompt,
   222→      executiveSummary,
   223→      sections,
   224→      sources,
   225→      findings,
   226→      comparisons,
   227→      references,
   228→      classification,
   229→      gaps,
   230→      metadata: {
   231→        sourceCount: sources.length,
   232→        findingCount: findings.length,
   233→        confidenceLevel,
   234→        depthLevel: strategy.depth,
   235→        searchEnginesUsed: strategy.engines,
   236→      },
   237→    };
   238→
   239→    this.emitEvent('research_report', sessionId, {
   240→      title: output.title,
   241→      sectionCount: sections.length,
   242→      sourceCount: sources.length,
   243→      findingCount: findings.length,
   244→      confidenceLevel,
   245→    });
   246→
   247→    return output;
   248→  }

   250→  // ─── Quality Validation: validateOutput() ──────────────
   251→
   252→  protected async validateOutput(output: RechercheOutput, prompt: string): Promise<QualityReport> {
   253→    const suggestions: string[] = [];
   254→
   255→    // 1. Source Quality (0-1): Checks credibility distribution
   256→    const sourceQuality = this.scoreSourceQuality(output, suggestions);
   257→
   258→    // 2. Comprehensiveness (0-1): Checks section coverage
   259→    const comprehensiveness = this.scoreComprehensiveness(output, suggestions);
   260→
   261→    // 3. Accuracy (0-1): Checks confidence levels and evidence
   262→    const accuracy = this.scoreAccuracy(output, suggestions);
   263→
   264→    // 4. Synthesis (0-1): Validates cross-references between sources
   265→    const synthesis = this.scoreSynthesis(output, suggestions);
   266→
   267→    // 5. Presentation (0-1): Validates structure and formatting
   268→    const presentation = this.scorePresentation(output, prompt, suggestions);
   269→
   270→    const dimensions: Record<string, number> = {
   271→      'Source Quality': sourceQuality,
   272→      'Comprehensiveness': comprehensiveness,
   273→      'Accuracy': accuracy,
   274→      'Synthesis': synthesis,
   275→      'Presentation': presentation,
   276→    };
   277→
   278→    const overall = this.computeWeightedScore(dimensions);
   279→    const passed = this.isQualityPassing(overall);
   280→
   281→    return { overall, dimensions, passed, suggestions };
   282→  }

   311→  // ─── Helper: classifyQuery() ───────────────────────────
   312→
   313→  private classifyQuery(query: string): RechercheClassification {
   314→    const queryLower = query.toLowerCase();
   315→
   316→    // Domain detection
   317→    const domainMap: Record<string, string> = {
   318→      'ai|machine learning|neural|model|gpt|llm': 'artificial-intelligence',
   319→      'climate|carbon|emission|environment|sustainability': 'environment',
   320→      'health|medical|disease|treatment|clinical': 'healthcare',
   321→      'finance|stock|market|investment|crypto|bitcoin': 'finance',
   322→      'education|learning|school|university|curriculum': 'education',
   323→      'law|legal|regulation|compliance|policy': 'legal',
   324→      'tech|software|hardware|startup|silicon': 'technology',
   325→      'science|research|study|experiment|peer-review': 'science',
   326→      'politics|government|election|democracy|policy': 'politics',
   327→      'business|company|corporate|strategy|management': 'business',
   328→    };
   329→
   330→    let domain = 'general';
   331→    for (const [pattern, d] of Object.entries(domainMap)) {
   332→      if (new RegExp(pattern, 'i').test(queryLower)) {
   333→        domain = d;
   334→        break;
   335→      }
   336→    }
   337→
   338→    // Primary topic: use the first meaningful noun phrase from the query
   339→    const primaryTopic = query.replace(/[?!.]/g, '').trim();
   340→
   341→    // Sub-topics: extract potential sub-areas from the query
   342→    const subTopics = this.extractSubTopics(query);
   343→
   344→    // Entity extraction (basic pattern matching)
   345→    const entities = this.extractEntities(query);
   346→
   347→    // Temporal focus detection
   348→    let temporalFocus: RechercheTemporalFocus = 'current';
   349→    if (/\b(history|historical|origin|past|evolution of)\b/i.test(queryLower)) {
   350→      temporalFocus = 'historical';
   351→    } else if (/\b(recent|latest|new|2024|2025|this year)\b/i.test(queryLower)) {
   352→      temporalFocus = 'recent';
   353→    } else if (/\b(future|predict|forecast|outlook|trend|2030|2040)\b/i.test(queryLower)) {
   354→      temporalFocus = 'future';
   355→    }
   356→
   357→    // Sentiment detection
   358→    let sentiment: RechercheSentiment = 'neutral';
   359→    if (/\b(benefit|improve|advance|progress|opportunity|positive|growth)\b/i.test(queryLower)) {
   360→      sentiment = 'positive';
   361→    } else if (/\b(risk|danger|threat|crisis|negative|decline|problem)\b/i.test(queryLower)) {
   362→      sentiment = 'negative';
   363→    } else if (/\b(vs|versus|debate|controversy|compare|pros.*cons)\b/i.test(queryLower)) {
   364→      sentiment = 'mixed';
   365→    }
   366→
   367→    return {
   368→      primaryTopic,
   369→      domain,
   370→      subTopics,
   371→      entities,
   372→      temporalFocus,
   373→      sentiment,
   374→    };
   375→  }

   377→  // ─── Helper: planSearchStrategy() ──────────────────────
   378→
   379→  private planSearchStrategy(classification: RechercheClassification, depth: number): SearchStrategy {
   380→    const engines = ['duckduckgo', 'google'];
   381→    if (classification.domain === 'academic' || classification.domain === 'science') {
   382→      engines.push('scholar', 'semantic-scholar');
   383→    }
   384→    if (classification.domain === 'news' || classification.temporalFocus === 'recent' || classification.temporalFocus === 'current') {
   385→      engines.push('news');
   386→    }
   387→
   388→    const maxSources = 5 + depth * 5; // depth 1=10, 3=20, 5=30
   389→    const minRelevance = Math.max(0.3, 0.7 - depth * 0.08); // deeper research accepts lower relevance
   390→
   391→    // Generate keyword variations
   392→    const keywords = [
   393→      classification.primaryTopic,
   394→      ...classification.subTopics.slice(0, depth),
   395→      ...classification.entities.slice(0, 2).map(e => e.name),
   396→    ];
   397→
   398→    return {
   399→      keywords,
   400→      engines: Array.from(new Set(engines)),
   401→      depth,
   402→      maxSources,
   403→      minRelevance,
   404→      language: 'en',
   405→    };
   406→  }

   408→  // ─── Helper: generateSources() ─────────────────────────
   409→  // Simulates source discovery. In production, this would integrate
   410→  // with the ResearchWorker plugin's browser-based search.
   411→
   412→  private generateSources(query: string, strategy: SearchStrategy): RechercheSource[] {
   413→    const sources: RechercheSource[] = [];
   414→    const sourceCount = Math.min(strategy.maxSources, 8 + strategy.depth * 3);
   415→
   416→    const typeDistribution: RechercheSourceType[] = ['academic', 'news', 'blog', 'documentation', 'government', 'social', 'other'];
   417→    const credibilityOptions: RechercheSourceCredibility[] = ['high', 'medium', 'low'];
   418→
   419→    for (let i = 0; i < sourceCount; i++) {
   420→      const id = `src_${uuidv4().slice(0, 8)}`;
   421→      const type = typeDistribution[i % typeDistribution.length];
   422→      const credibility = credibilityOptions[i < sourceCount * 0.4 ? 0 : i < sourceCount * 0.8 ? 1 : 2];
   423→      const relevance = Math.max(strategy.minRelevance, 0.5 + Math.random() * 0.5);
   424→
   425→      sources.push({
   426→        id,
   427→        title: this.simulateSourceTitle(query, type, i),
   428→        url: this.simulateSourceUrl(query, type, i),
   429→        type,
   430→        credibility,
   431→        relevance: Math.round(relevance * 100) / 100,
   432→        snippet: this.simulateSnippet(query, type, i),
   433→        publishedDate: this.simulateDate(strategy.depth),
   434→        keyClaims: this.simulateKeyClaims(query, type, i),
   435→      });
   436→    }
   437→
   438→    // Sort by relevance descending
   439→    return sources.sort((a, b) => b.relevance - a.relevance);
   440→  }

   442→  // ─── Helper: extractFindings() ─────────────────────────
   443→
   444→  private extractFindings(sources: RechercheSource[]): RechercheFinding[] {
   445→    const findings: RechercheFinding[] = [];
   446→
   447→    for (const source of sources) {
   448→      for (let c = 0; c < source.keyClaims.length; c++) {
   449→        const claim = source.keyClaims[c];
   450→        const id = `find_${uuidv4().slice(0, 8)}`;
   451→
   452→        // Determine finding type from claim content
   453→        const type = this.inferFindingType(claim);
   454→
   455→        // Determine consensus based on source credibility and claim nature
   456→        const consensus = this.inferConsensus(source.credibility, claim);
   457→
   458→        // Confidence influenced by source credibility and relevance
   459→        const credibilityMultiplier = source.credibility === 'high' ? 0.85 : source.credibility === 'medium' ? 0.65 : 0.45;
   460→        const confidence = Math.min(1, Math.round((credibilityMultiplier * source.relevance + Math.random() * 0.15) * 100) / 100);
   461→
   462→        findings.push({
   463→          id,
   464→          statement: claim,
   465→          evidence: [`Supported by ${source.title}`, `Source relevance: ${(source.relevance * 100).toFixed(0)}%`],
   466→          confidence,
   467→          sourceIds: [source.id],
   468→          consensus,
   469→          type,
   470→        });
   471→      }
   472→    }
   473→
   474→    // Deduplicate similar findings (merge sourceIds for matching statements)
   475→    return this.deduplicateFindings(findings);
   476→  }

   478→  // ─── Helper: compareSources() ──────────────────────────
   479→
   480→  private compareSources(sources: RechercheSource[]): SourceComparison[] {
   481→    const comparisons: SourceComparison[] = [];
   482→
   483→    // Compare pairs of sources (limit to avoid combinatorial explosion)
   484→    const limit = Math.min(sources.length, 8);
   485→    for (let i = 0; i < limit; i++) {
   486→      for (let j = i + 1; j < limit; j++) {
   487→        const srcA = sources[i];
   488→        const srcB = sources[j];
   489→
   490→        const agreements = this.findAgreements(srcA, srcB);
   491→        const disagreements = this.findDisagreements(srcA, srcB);
   492→
   493→        // Similarity based on shared claims and overlapping topics
   494→        const totalClaims = srcA.keyClaims.length + srcB.keyClaims.length;
   495→        const sharedClaims = agreements.length;
   496→        const similarity = totalClaims > 0
   497→          ? Math.round((2 * sharedClaims / totalClaims) * 100) / 100
   498→          : 0;
   499→
   500→        if (agreements.length > 0 || disagreements.length > 0) {
   501→          comparisons.push({
   502→            sources: [srcA.id, srcB.id],
   503→            agreements,
   504→            disagreements,
   505→            similarity,
   506→          });
   507→        }
   508→      }
   509→    }
   510→
   511→    return comparisons;
   512→  }

   516→  private generateCitations(sources: RechercheSource[], format: string): RechercheReference[] {
   517→    const references: RechercheReference[] = [];
   518→    const formats: RechercheReferenceFormat[] = format === 'all'
   519→      ? ['apa', 'mla', 'chicago', 'bibtex']
   520→      : [format as RechercheReferenceFormat];
   521→
   522→    for (const source of sources) {
   523→      for (const fmt of formats) {
   524→        const id = `ref_${uuidv4().slice(0, 8)}`;
   525→        const citation = this.formatCitation(source, fmt);
   526→        references.push({
   527→          id,
   528→          sourceId: source.id,
   529→          format: fmt,
   530→          citation,
   531→        });
   532→      }
   533→    }
   534→
   535→    return references;
   536→  }

   538→  // ─── Helper: identifyGaps() ────────────────────────────
   539→
   540→  private identifyGaps(findings: RechercheFinding[], sources: RechercheSource[]): string[] {
   541→    const gaps: string[] = [];
   542→
   543→    // Gap: insufficient source diversity
   544→    const types = new Set(sources.map(s => s.type));
   545→    if (types.size < 3) {
   546→      gaps.push('Limited source type diversity — research would benefit from broader source categories (academic, government, news)');
   547→    }
   548→
   549→    // Gap: low-confidence findings
   550→    const lowConfidence = findings.filter(f => f.confidence < 0.5);
   551→    if (lowConfidence.length > findings.length * 0.3) {
   552→      gaps.push('Many findings have low confidence — additional verification from authoritative sources is needed');
   553→    }
   554→
   555→    // Gap: disputed findings
   556→    const disputed = findings.filter(f => f.consensus === 'disputed');
   557→    if (disputed.length > 0) {
   558→      gaps.push(`${disputed.length} finding(s) are disputed across sources — further investigation required to resolve contradictions`);
   559→    }
   560→
   561→    // Gap: temporal coverage
   562→    const hasRecent = sources.some(s => {
   563→      if (!s.publishedDate) return false;
   564→      const year = new Date(s.publishedDate).getFullYear();
   565→      return year >= 2024;
   566→    });
   567→    if (!hasRecent) {
   568→      gaps.push('No recent sources (2024+) found — current data may be outdated for the topic');
   569→    }
   570→
   571→    // Gap: missing academic/peer-reviewed sources
   572→    const hasAcademic = sources.some(s => s.type === 'academic');
   573→    if (!hasAcademic) {
   574→      gaps.push('No peer-reviewed academic sources — adding scholarly research would strengthen evidence');
   575→    }
   576→
   577→    // Gap: insufficient cross-referencing
   578→    const multiSource = findings.filter(f => f.sourceIds.length > 1);
   579→    if (multiSource.length < findings.length * 0.3) {
   580→      gaps.push('Most findings are supported by only a single source — cross-validation from additional sources is recommended');
   581→    }
   582→
   583→    // Gap: opinion-heavy research
   584→    const opinions = findings.filter(f => f.type === 'opinion');
   585→    if (opinions.length > findings.length * 0.5) {
   586→      gaps.push('Research is opinion-dominant — more factual and statistical evidence is needed for a balanced report');
   587→    }
   588→
   589→    // Always include a forward-looking gap
   590→    gaps.push('Further research is recommended to track evolving developments on this topic');
   591→
   592→    return gaps;
   593→  }

   597→  private renderMarkdown(output: RechercheOutput): string {
   598→    const lines: string[] = [];
   599→
   600→    // Title
   601→    lines.push(`# ${output.title}`);
   602→    lines.push('');
   603→
   604→    // Metadata
   605→    lines.push(`> **Query**: ${output.query}`);
   606→    lines.push(`> **Domain**: ${output.classification.domain}`);
   607→    lines.push(`> **Confidence**: ${(output.metadata.confidenceLevel * 100).toFixed(0)}%`);
   608→    lines.push(`> **Sources**: ${output.metadata.sourceCount} | **Findings**: ${output.metadata.findingCount}`);
   609→    lines.push('');
   610→
   611→    // Executive Summary
   612→    lines.push('## Executive Summary');
   613→    lines.push('');
   614→    lines.push(output.executiveSummary);
   615→    lines.push('');
   616→
   617→    // Sections
   618→    for (const section of output.sections) {
   619→      lines.push(`## ${section.heading}`);
   620→      lines.push('');
   621→      lines.push(section.content);
   622→      lines.push('');
   623→
   624→      if (section.sourceIds.length > 0) {
   625→        const sourceRefs = section.sourceIds
   626→          .map(sid => {
   627→            const src = output.sources.find(s => s.id === sid);
   628→            return src ? `[${src.title}](${src.url})` : sid;
   629→          })
   630→          .join(', ');
   631→        lines.push(`*Sources: ${sourceRefs}*`);
   632→        lines.push('');
   633→      }
   634→
   635→      lines.push(`*Confidence: ${(section.confidence * 100).toFixed(0)}%*`);
   636→      lines.push('');
   637→    }
   638→
   639→    // Knowledge Gaps
   640→    if (output.gaps.length > 0) {
   641→      lines.push('## Knowledge Gaps');
   642→      lines.push('');
   643→      for (const gap of output.gaps) {
   644→        lines.push(`- ${gap}`);
   645→      }
   646→      lines.push('');
   647→    }
   648→
   649→    // Sources
   650→    lines.push('## Sources');
   651→    lines.push('');
   652→    for (const source of output.sources) {
   653→      const credibilityBadge = source.credibility === 'high' ? '✓' : source.credibility === 'medium' ? '~' : '!';
   654→      lines.push(`- ${credibilityBadge} **${source.title}** — ${source.url} _(${source.type}, relevance: ${(source.relevance * 100).toFixed(0)}%)_`);
   655→    }
   656→    lines.push('');
   657→
   658→    // References
   659→    const apaRefs = output.references.filter(r => r.format === 'apa');
   660→    if (apaRefs.length > 0) {
   661→      lines.push('## References (APA)');
   662→      lines.push('');
   663→      for (const ref of apaRefs) {
   664→        lines.push(`1. ${ref.citation}`);
   665→      }
   666→      lines.push('');
   667→    }
   668→
   669→    return lines.join('\n');
   670→  }

   709→  private scoreSourceQuality(output: RechercheOutput, suggestions: string[]): number {
   710→    if (output.sources.length === 0) {
   711→      suggestions.push('No sources found — research cannot be validated');
   712→      return 0;
   713→    }
   714→
   715→    let score = 0;
   716→
   717→    // Credibility distribution (high credibility should dominate)
   718→    const highCred = output.sources.filter(s => s.credibility === 'high').length;
   719→    const medCred = output.sources.filter(s => s.credibility === 'medium').length;
   720→    const lowCred = output.sources.filter(s => s.credibility === 'low').length;
   721→    const total = output.sources.length;
   722→
   723→    const highRatio = highCred / total;
   724→    const lowRatio = lowCred / total;
   725→
   726→    // Reward high-credibility sources, penalize low-credibility
   727→    score += Math.min(0.4, highRatio * 0.5);
   728→    score += Math.min(0.2, (medCred / total) * 0.25);
   729→    score -= Math.min(0.2, lowRatio * 0.3);
   730→
   731→    // Source type diversity
   732→const uniqueTypes = Array.from(new Set(output.sources.map(s => s.type))).length;
   733→    score += Math.min(0.2, uniqueTypes * 0.04);
   734→
   735→    // Average relevance
   736→    const avgRelevance = output.sources.reduce((sum, s) => sum + s.relevance, 0) / total;
   737→    score += Math.min(0.2, avgRelevance * 0.2);
   738→
   739→    if (highRatio < 0.2) {
   740→      suggestions.push('Increase the proportion of high-credibility sources (academic, government)');
   741→    }
   742→    if (uniqueTypes < 3) {
   743→      suggestions.push('Diversify source types for more balanced research');
   744→    }
   745→
   746→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   747→  }

   749→  private scoreComprehensiveness(output: RechercheOutput, suggestions: string[]): number {
   750→    let score = 0;
   751→
   752→    // Section type coverage
   753→    const requiredTypes: RechercheSection['type'][] = ['overview', 'findings', 'analysis', 'conclusion'];
   754→    const presentTypes = new Set(output.sections.map(s => s.type));
   755→    const coverageRatio = requiredTypes.filter(t => presentTypes.has(t)).length / requiredTypes.length;
   756→    score += coverageRatio * 0.4;
   757→
   758→    // Section depth (word count per section)
   759→    const avgWords = output.sections.reduce((sum, s) => sum + s.content.split(/\s+/).length, 0) / Math.max(output.sections.length, 1);
   760→    if (avgWords >= 150) {
   761→      score += 0.2;
   762→    } else if (avgWords >= 100) {
   763→      score += 0.1;
   764→      suggestions.push('Sections should contain at least 150 words for thorough coverage');
   765→    }
   766→
   767→    // Source count adequacy
   768→    if (output.metadata.sourceCount >= 8) {
   769→      score += 0.2;
   770→    } else if (output.metadata.sourceCount >= 4) {
   771→      score += 0.1;
   772→      suggestions.push('Consider adding more sources for comprehensive coverage');
   773→    }
   774→
   775→    // Finding count
   776→    if (output.metadata.findingCount >= 5) {
   777→      score += 0.1;
   778→    } else if (output.metadata.findingCount >= 2) {
   779→      score += 0.05;
   780→    }
   781→
   782→    // Gaps identified
   783→    if (output.gaps.length >= 2) {
   784→      score += 0.1;
   785→    }
   786→
   787→    if (coverageRatio < 1) {
   788→      const missing = requiredTypes.filter(t => !presentTypes.has(t));
   789→      suggestions.push(`Add missing section types: ${missing.join(', ')}`);
   790→    }
   791→
   792→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   793→  }

   795→  private scoreAccuracy(output: RechercheOutput, suggestions: string[]): number {
   796→    if (output.findings.length === 0) {
   797→      suggestions.push('No findings extracted — accuracy cannot be assessed');
   798→      return 0;
   799→    }
   800→
   801→    let score = 0;
   802→
   803→    // Average finding confidence
   804→    const avgConfidence = output.findings.reduce((sum, f) => sum + f.confidence, 0) / output.findings.length;
   805→    score += avgConfidence * 0.3;
   806→
   807→    // Evidence backing per finding
   808→    const avgEvidence = output.findings.reduce((sum, f) => sum + f.evidence.length, 0) / output.findings.length;
   809→    score += Math.min(0.25, avgEvidence * 0.08);
   810→
   811→    // Multi-source validation
   812→    const multiSourceFindings = output.findings.filter(f => f.sourceIds.length > 1).length;
   813→    const multiSourceRatio = multiSourceFindings / output.findings.length;
   814→    score += multiSourceRatio * 0.25;
   815→
   816→    // Disputed findings penalty
   817→    const disputedRatio = output.findings.filter(f => f.consensus === 'disputed').length / output.findings.length;
   818→    score -= disputedRatio * 0.15;
   819→
   820→    // Strong consensus bonus
   821→    const strongConsensusRatio = output.findings.filter(f => f.consensus === 'strong').length / output.findings.length;
   822→    score += strongConsensusRatio * 0.1;
   823→
   824→    if (avgConfidence < 0.5) {
   825→      suggestions.push('Average finding confidence is below 50% — verify sources and claims');
   826→    }
   827→    if (disputedRatio > 0.3) {
   828→      suggestions.push('High proportion of disputed findings — resolve contradictions or present them clearly');
   829→    }
   830→
   831→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   832→  }

   834→  private scoreSynthesis(output: RechercheOutput, suggestions: string[]): number {
   835→    let score = 0;
   836→
   837→    // Cross-references: sections that reference multiple sources
   838→    const multiRefSections = output.sections.filter(s => s.sourceIds.length > 1).length;
   839→    const sectionRatio = output.sections.length > 0 ? multiRefSections / output.sections.length : 0;
   840→    score += Math.min(0.3, sectionRatio * 0.4);
   841→
   842→    // Comparisons between sources
   843→    if (output.comparisons.length > 0) {
   844→      score += Math.min(0.3, output.comparisons.length * 0.1);
   845→    } else {
   846→      suggestions.push('No source comparisons generated — add cross-referencing between sources');
   847→    }
   848→
   849→    // Executive summary references findings
   850→    if (output.executiveSummary.length > 100) {
   851→      score += 0.2;
   852→    } else {
   853→      suggestions.push('Executive summary is too brief — synthesize key findings more thoroughly');
   854→    }
   855→
   856→    // Findings linked to sections
   857→    const findingsReferenced = output.sections.some(s =>
   858→      s.type === 'findings' || s.type === 'analysis'
   859→    );
   860→    if (findingsReferenced) {
   861→      score += 0.2;
   862→    } else {
   863→      suggestions.push('Add a dedicated findings or analysis section to synthesize research results');
   864→    }
   865→
   866→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   867→  }

   869→  private scorePresentation(output: RechercheOutput, prompt: string, suggestions: string[]): number {
   870→    let score = 0;
   871→
   872→    // Title quality
   873→    if (output.title && output.title.length > 5) {
   874→      score += 0.1;
   875→    }
   876→
   877→    // Structured sections
   878→    if (output.sections.length >= 3) {
   879→      score += 0.2;
   880→    } else {
   881→      suggestions.push('Report needs at least 3 sections for adequate structure');
   882→    }
   883→
   884→    // Section ordering (overview should come first, conclusion last)
   885→    const sectionTypes = output.sections.map(s => s.type);
   886→    if (sectionTypes[0] === 'overview') {
   887→      score += 0.1;
   888→    }
   889→    if (sectionTypes[sectionTypes.length - 1] === 'conclusion' || sectionTypes[sectionTypes.length - 1] === 'recommendations') {
   890→      score += 0.1;
   891→    }
   892→
   893→    // References present
   894→    if (output.references.length > 0) {
   895→      score += 0.15;
   896→    } else {
   897→      suggestions.push('No references formatted — add citations in standard academic styles');
   898→    }
   899→
   900→    // Gaps identified
   901→    if (output.gaps.length > 0) {
   902→      score += 0.1;
   903→    }
   904→
   905→    // Metadata completeness
   906→    const meta = output.metadata;
   907→    if (meta.sourceCount > 0 && meta.findingCount > 0 && meta.confidenceLevel > 0) {
   908→      score += 0.15;
   909→    } else {
   910→      suggestions.push('Complete all metadata fields (sourceCount, findingCount, confidenceLevel)');
   911→    }
   912→
   913→    // Classification present
   914→    if (output.classification.primaryTopic && output.classification.domain) {
   915→      score += 0.1;
   916→    }
   917→
   918→    return Math.max(0, Math.min(1, Math.round(score * 100) / 100));
   919→  }

  1236→  private inferFindingType(claim: string): RechercheFindingType {
  1237→    const lower = claim.toLowerCase();
  1238→    if (/\b(percent|%|statistics|data|survey|poll|rate|number|figure|approximately)\b/.test(lower)) return 'statistic';
  1239→    if (/\b(trend|growth|decline|increase|decrease|shift|evolution)\b/.test(lower)) return 'trend';

  1245→  private inferConsensus(credibility: RechercheSourceCredibility, claim: string): RechercheFindingConsensus {
  1246→    // High-credibility sources tend to have stronger consensus
  1247→    if (credibility === 'high') return 'strong';
  1248→    if (credibility === 'medium') {
  1249→      // Check for hedging language
  1250→      if (/\b(may|might|could|suggest|possibly|perhaps)\b/.test(claim.toLowerCase())) return 'moderate';
  1251→      return 'moderate';
  1252→    }
  1253→    // Low credibility tends toward weaker consensus
  1254→    if (/\b(controversial|disputed|debated|contradicts)\b/.test(claim.toLowerCase())) return 'disputed';
  1255→    return 'weak';
  1256→  }

  1258→  private deduplicateFindings(findings: RechercheFinding[]): RechercheFinding[] {
  1259→    const seen = new Map<string, RechercheFinding>();
  1260→
  1261→    for (const finding of findings) {
  1262→      const key = finding.statement.toLowerCase().trim();
  1263→      const existing = seen.get(key);
  1264→
  1265→      if (existing) {
  1266→        // Merge: combine source IDs and evidence, keep higher confidence
  1267→        existing.sourceIds = Array.from(new Set([...existing.sourceIds, ...finding.sourceIds]));
  1268→        existing.evidence = Array.from(new Set([...existing.evidence, ...finding.evidence]));
  1269→        existing.confidence = Math.max(existing.confidence, finding.confidence);
  1270→
  1271→        // Upgrade consensus if multiple sources agree
  1272→        const consensusOrder: RechercheFindingConsensus[] = ['weak', 'moderate', 'strong'];
  1273→        if (consensusOrder.indexOf(finding.consensus) > consensusOrder.indexOf(existing.consensus)) {
  1274→          existing.consensus = finding.consensus;
  1275→        }
  1276→      } else {
  1277→        seen.set(key, { ...finding, sourceIds: [...finding.sourceIds], evidence: [...finding.evidence] });
  1278→      }
  1279→    }
  1280→
  1281→    return Array.from(seen.values());
  1282→  }

  1284→  private findAgreements(srcA: RechercheSource, srcB: RechercheSource): string[] {
  1285→    const agreements: string[] = [];
  1286→
  1287→    // Simple keyword overlap on key claims
  1288→    const wordsA = new Set(srcA.keyClaims.flatMap(c => c.toLowerCase().split(/\s+/)));
  1289→    const wordsB = new Set(srcB.keyClaims.flatMap(c => c.toLowerCase().split(/\s+/)));
  1290→
  1291→    for (const claimA of srcA.keyClaims) {
  1292→      const claimAWords = new Set(claimA.toLowerCase().split(/\s+/));
  1293→      const overlap = Array.from(claimAWords).filter(w => w.length > 4 && wordsB.has(w)).length;
  1294→      if (overlap >= 2) {
  1295→        agreements.push(claimA);
  1296→      }
  1297→    }
  1298→
  1299→    return agreements.slice(0, 5);
  1300→  }

  1302→  private findDisagreements(srcA: RechercheSource, srcB: RechercheSource): string[] {
  1303→    const disagreements: string[] = [];
  1304→
  1305→    // Detect contradictory language patterns
  1306→    const contradictionPatterns = [
  1307→      /\bincrease\b/i, /\bdecrease\b/i,
  1308→      /\brising\b/i, /\bfalling\b/i,
  1309→      /\bpositive\b/i, /\bnegative\b/i,
  1310→      /\bsupports\b/i, /\bcontradicts\b/i,
  1311→      /\bbeneficial\b/i, /\bharmful\b/i,
  1312→    ];
  1313→
  1314→    for (const claimA of srcA.keyClaims) {
  1315→      for (let i = 0; i < contradictionPatterns.length; i += 2) {
  1316→        const patA = contradictionPatterns[i];
  1317→        const patB = contradictionPatterns[i + 1];
  1318→        if (patA && patB && patA.test(claimA)) {
  1319→          for (const claimB of srcB.keyClaims) {
  1320→            if (patB.test(claimB)) {
  1321→              disagreements.push(`${claimA} vs. ${claimB}`);
  1322→            }
  1323→          }
  1324→        }
  1325→      }
  1326→    }
  1327→
  1328→    return disagreements.slice(0, 3);
  1329→  }

  1331→  private formatCitation(source: RechercheSource, format: RechercheReferenceFormat): string {
  1332→    const year = source.publishedDate ? new Date(source.publishedDate).getFullYear() : 'n.d.';
  1333→    const title = source.title;

  1341→      case 'mla':

  1356→  private computeConfidenceLevel(findings: RechercheFinding[], sources: RechercheSource[]): number {
  1357→    if (findings.length === 0) return 0;
  1358→
  1359→    // Weighted average of finding confidences, boosted by multi-source validation
  1360→    const totalConfidence = findings.reduce((sum, f) => {
  1361→      const multiSourceBonus = f.sourceIds.length > 1 ? 0.05 : 0;
  1362→      const consensusBonus = f.consensus === 'strong' ? 0.05 : f.consensus === 'moderate' ? 0.02 : 0;
  1363→      return sum + Math.min(1, f.confidence + multiSourceBonus + consensusBonus);
  1364→    }, 0);
  1365→
  1366→    const avgConfidence = totalConfidence / findings.length;
  1367→
  1368→    // Factor in source credibility
  1369→    const highCredRatio = sources.filter(s => s.credibility === 'high').length / Math.max(sources.length, 1);
  1370→    const credibilityBoost = highCredRatio * 0.1;
  1371→
  1372→    return Math.min(1, Math.round((avgConfidence + credibilityBoost) * 100) / 100);
  1373→  }

   321→/** Get current memory usage in MB */
   322→function getMemoryUsageMB(): { heapUsed: number; heapTotal: number; rss: number } {
   323→  const usage = process.memoryUsage();
   324→  return {
   325→    heapUsed: Math.round(usage.heapUsed / 1024 / 1024),
   326→    heapTotal: Math.round(usage.heapTotal / 1024 / 1024),
   327→    rss: Math.round(usage.rss / 1024 / 1024),
   328→  };
   329→}

   359→      simulateAllDown();
   360→
   361→      // None of these should throw

   364→      await expect(jwtBlacklist.isBlacklisted(jti)).resolves.toBe(true);
   365→
   366→      const session = await sessionManager.createSession(
   367→        'user-cascade',
   368→        'fp-cascade',
   369→        '10.0.0.1',
   370→        'CascadingAgent'
   371→      );
   372→      expect(session).toBeDefined();
   373→      expect(session.userId).toBe('user-cascade');
   374→
   375→      await expect(
   376→        cacheManager.set('agent-cascade', 'prompt-cascade', { output: 'degraded' })
   377→      ).resolves.toBeUndefined();
   378→
   379→      const auditId = await auditTrailService.record({
   380→        action: 'security.threat_detected',
   381→        actorId: 'system',
   382→        actorType: 'system',
   383→        resourceType: 'infrastructure',

   387→        userAgent: 'ChaosTest',

   522→  describe('Memory leak detection under sustained failure', () => {
   523→    it('should not leak memory under sustained failure conditions', async () => {
   524→      simulateAllDown();

   526→      // Force garbage collection if available
   527→      if (global.gc) global.gc();
   528→
   529→      const memoryBefore = getMemoryUsageMB();
   530→
   531→      // Perform many operations under failure conditions
   532→      for (let i = 0; i < 100; i++) {
   533→        await jwtBlacklist.blacklist(`leak-test-jti-${i}`, 3600);
   534→        await jwtBlacklist.isBlacklisted(`leak-test-jti-${i}`);
   535→
   536→        // Also create and destroy sessions
   537→        const session = await sessionManager.createSession(
   538→          `user-leak-${i}`,
   539→          `fp-leak-${i}`,
   540→          '10.0.0.1',
   541→          'LeakAgent'
   542→        );
   543→        await sessionManager.destroySession(session.id);
   544→      }
   545→
   546→      // Force garbage collection if available
   547→      if (global.gc) global.gc();
   548→
   549→      const memoryAfter = getMemoryUsageMB();
   550→
   551→      // Memory growth should be reasonable (less than 50MB increase)
   552→      const heapGrowth = memoryAfter.heapUsed - memoryBefore.heapUsed;
   553→      expect(heapGrowth).toBeLessThan(50);

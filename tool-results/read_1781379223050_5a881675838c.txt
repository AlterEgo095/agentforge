   122→/** Create a fetch mock that succeeds for all providers with simulated latency */
   123→function createSuccessFetch(baseLatencyMs: number = 20) {
   124→  return vi.fn().mockImplementation(async (url: string) => {
   125→    await new Promise(resolve => setTimeout(resolve, baseLatencyMs));
   126→    if (url.includes('anthropic.com')) {
   127→      return {
   128→        ok: true,
   129→        json: async () => ({
   130→          content: [{ text: 'Simulated Anthropic response' }],
   131→          usage: { input_tokens: 100, output_tokens: 50 },
   132→        }),
   133→      };
   134→    }
   135→    if (url.includes('generativelanguage')) {
   136→      return {
   137→        ok: true,
   138→        json: async () => ({
   139→          candidates: [{ content: { parts: [{ text: 'Simulated Google response' }] } }],
   140→          usageMetadata: { promptTokenCount: 100, candidatesTokenCount: 50 },
   141→        }),
   142→      };
   143→    }
   144→    return {
   145→      ok: true,
   146→      json: async () => ({
   147→        choices: [{ message: { content: 'Simulated OpenAI-compatible response' } }],
   148→        usage: { prompt_tokens: 100, completion_tokens: 50 },

   154→/** Create a fetch mock that fails for specific providers */
   155→function createFailingFetch(
   156→  failPatterns: RegExp[],
   157→  failStatus: number = 500,
   158→  successLatencyMs: number = 20,
   159→) {
   160→  return vi.fn().mockImplementation(async (url: string) => {
   161→    for (const pattern of failPatterns) {
   162→      if (pattern.test(url)) {
   163→        return { ok: false, status: failStatus, text: async () => `Error ${failStatus}` };
   164→      }
   165→    }
   166→    await new Promise(resolve => setTimeout(resolve, successLatencyMs));
   167→    if (url.includes('anthropic.com')) {
   168→      return {
   169→        ok: true,
   170→        json: async () => ({
   171→          content: [{ text: 'Failover response' }],
   172→          usage: { input_tokens: 100, output_tokens: 50 },
   173→        }),
   174→      };
   175→    }
   176→    if (url.includes('generativelanguage')) {
   177→      return {
   178→        ok: true,
   179→        json: async () => ({
   180→          candidates: [{ content: { parts: [{ text: 'Failover response' }] } }],
   181→          usageMetadata: { promptTokenCount: 100, candidatesTokenCount: 50 },
   182→        }),
   183→      };
   184→    }
   185→    return {
   186→      ok: true,
   187→      json: async () => ({
   188→        choices: [{ message: { content: 'Failover response' } }],
   189→        usage: { prompt_tokens: 100, completion_tokens: 50 },
   190→      }),
   191→    };
   192→  });
   193→}

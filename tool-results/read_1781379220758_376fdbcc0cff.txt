   197→/** Box-Muller transform to generate normally distributed random numbers */
   198→function gaussianRandom(mean: number, std: number): number {
   199→  const u1 = Math.random();
   200→  const u2 = Math.random();
   201→  const z = Math.sqrt(-2 * Math.log(Math.max(u1, 1e-10))) * Math.cos(2 * Math.PI * u2);
   202→  return mean + z * std;
   203→}
   204→
   205→/** Generate a realistic simulated latency for a provider (log-normal) */
   206→function simulateLatency(provider: LLMId): number {
   207→  const params = PROVIDER_LATENCY_PARAMS[provider];
   208→  // Log-normal: ensures non-negative latencies with right skew
   209→  const logMean = Math.log(params.meanMs);
   210→  const logStd = params.stdMs / params.meanMs; // coefficient of variation
   211→  const lat = Math.exp(gaussianRandom(logMean, logStd));
   212→  return Math.max(50, Math.round(lat)); // minimum 50ms
   213→}
   214→
   215→/** Determine if a simulated call should fail based on provider error rate */
   216→function shouldSimulateError(provider: LLMId): boolean {
   217→  const errorRate = PROVIDER_LATENCY_PARAMS[provider].errorRate;
   218→  return Math.random() < errorRate;
   219→}

   263→/** Calculate percentile from sorted array */
   264→function percentile(sorted: number[], p: number): number {
   265→  if (sorted.length === 0) return 0;
   266→  const idx = (p / 100) * (sorted.length - 1);
   267→  const lower = Math.floor(idx);
   268→  const upper = Math.ceil(idx);
   269→  if (lower === upper) return sorted[lower];
   270→  return sorted[lower] + (sorted[upper] - sorted[lower]) * (idx - lower);
   271→}
   272→
   273→/** Build metrics from raw call results */
   274→function buildMetrics(
   275→  provider: LLMId,
   276→  latencies: number[],
   277→  tokens: number[],
   278→  costs: number[],
   279→  errors: number,
   280→  failovers: number,
   281→  cacheHits: number,
   282→  cacheMisses: number,
   283→): ProviderMetrics {
   284→  const totalCalls = latencies.length + errors;
   285→  const successCalls = latencies.length;
   286→  const sorted = [...latencies].sort((a, b) => a - b);
   287→  const totalTokens = tokens.reduce((s, t) => s + t, 0);
   288→  const totalCost = costs.reduce((s, c) => s + c, 0);
   289→  const totalLatencyMs = latencies.reduce((s, l) => s + l, 0);
   290→
   291→  return {
   292→    provider,
   293→    totalCalls,
   294→    successCalls,
   295→    failedCalls: errors,
   296→    avgLatency: successCalls > 0 ? Math.round(totalLatencyMs / successCalls) : 0,
   297→    p50Latency: percentile(sorted, 50),
   298→    p95Latency: percentile(sorted, 95),
   299→    p99Latency: percentile(sorted, 99),
   300→    minLatency: sorted[0] || 0,
   301→    maxLatency: sorted[sorted.length - 1] || 0,
   302→    totalTokens,
   303→    throughputTokensPerSec: totalLatencyMs > 0 ? Math.round(totalTokens / (totalLatencyMs / 1000)) : 0,
   304→    costPer1kTokens: totalTokens > 0 ? (totalCost / totalTokens) * 1000 : 0,
   305→    totalCost,
   306→    errorRate: totalCalls > 0 ? (errors / totalCalls) * 100 : 0,
   307→    cacheHitRatio: (cacheHits + cacheMisses) > 0 ? cacheHits / (cacheHits + cacheMisses) : 0,
   308→    fallbackRate: totalCalls > 0 ? (failovers / totalCalls) * 100 : 0,
   309→    failoverCount: failovers,
   310→    cacheHits,
   311→    cacheMisses,
   312→  };
   313→}

   319→function createBenchmarkFetch(
   320→  failProviders: Set<LLMId> = new Set(),
   321→  cacheEnabled: boolean = false,
   322→) {
   323→  const cache = new Map<string, { output: string; promptTokens: number; completionTokens: number }>();
   324→
   325→  return vi.fn().mockImplementation(async (url: string, _opts?: any) => {
   326→    // Determine provider from URL
   327→    let provider: LLMId | null = null;
   328→    if (url.includes('anthropic.com')) provider = 'claude-3.7-sonnet';
   329→    else if (url.includes('generativelanguage')) {
   330→      // Distinguish gemini-2.5-pro vs gemini-2.0-flash from URL
   331→      if (url.includes('gemini-2.5-pro')) provider = 'gemini-2.5-pro';
   332→      else provider = 'gemini-2.0-flash';
   333→    }
   334→    else if (url.includes('deepseek.com')) provider = 'deepseek-r1';
   335→    else if (url.includes('together.xyz')) {
   336→      // Could be llama or qwen — use opts body to distinguish
   337→      provider = 'llama-3.3'; // Default, will be overridden
   338→    }
   339→    else if (url.includes('dashscope')) provider = 'qwen-2.5';
   340→    else if (url.includes('mistral.ai')) provider = 'mistral-large';
   341→    else if (url.includes('openai.com')) {
   342→      // Could be gpt-4o or gpt-o1
   343→      provider = 'gpt-4o'; // Default
   344→    }
   345→
   346→    // Try to determine specific model from request body
   347→    if (_opts?.body) {
   348→      try {
   349→        const body = JSON.parse(_opts.body as string);
   350→        if (body.model) {
   351→          if (body.model.includes('o1') || body.model === 'o1-2024-12-17') provider = 'gpt-o1';
   352→          else if (body.model.includes('gpt-4o')) provider = 'gpt-4o';
   353→          else if (body.model.includes('Llama')) provider = 'llama-3.3';
   354→          else if (body.model.includes('Qwen') || body.model.includes('qwen')) provider = 'qwen-2.5';
   355→          else if (body.model.includes('deepseek')) provider = 'deepseek-r1';
   356→          else if (body.model.includes('mistral')) provider = 'mistral-large';
   357→        }
   358→      } catch {
   359→        // ignore parse errors
   360→      }
   361→    }
   362→
   363→    // If provider still null, default to gpt-4o
   364→    if (!provider) provider = 'gpt-4o';
   365→
   366→    // Check cache
   367→    const cacheKey = `${provider}:${url}`;
   368→    if (cacheEnabled && cache.has(cacheKey)) {
   369→      const cached = cache.get(cacheKey)!;
   370→      // Return cached result instantly (no simulated latency for cache hits)
   371→      if (url.includes('anthropic.com')) {
   372→        return { ok: true, json: async () => ({ content: [{ text: cached.output }], usage: { input_tokens: cached.promptTokens, output_tokens: cached.completionTokens } }) };
   373→      }
   374→      if (url.includes('generativelanguage')) {
   375→        return { ok: true, json: async () => ({ candidates: [{ content: { parts: [{ text: cached.output }] } }], usageMetadata: { promptTokenCount: cached.promptTokens, candidatesTokenCount: cached.completionTokens } }) };
   376→      }
   377→      return { ok: true, json: async () => ({ choices: [{ message: { content: cached.output } }], usage: { prompt_tokens: cached.promptTokens, completion_tokens: cached.completionTokens } }) };
   378→    }
   379→
   380→    // Simulate error for designated fail providers
   381→    if (failProviders.has(provider)) {
   382→      return { ok: false, status: 500, text: async () => 'Simulated provider failure' };
   383→    }
   384→
   385→    // Simulate random error based on provider error rate
   386→    if (shouldSimulateError(provider)) {
   387→      return { ok: false, status: 500, text: async () => 'Simulated random error' };
   388→    }
   389→
   390→    // Simulate realistic latency
   391→    const latency = simulateLatency(provider);
   392→    await new Promise(resolve => setTimeout(resolve, Math.min(latency, 50))); // Cap sleep at 50ms for test speed
   393→
   394→    const promptTokens = 100 + Math.round(Math.random() * 200);
   395→    const completionTokens = 50 + Math.round(Math.random() * 300);
   396→    const output = `Simulated response from ${provider} (${latency}ms latency)`;
   397→
   398→    // Store in cache if enabled
   399→    if (cacheEnabled) {
   400→      cache.set(cacheKey, { output, promptTokens, completionTokens });
   401→    }
   402→
   403→    // Return provider-appropriate response format
   404→    if (url.includes('anthropic.com')) {
   405→      return {
   406→        ok: true,
   407→        json: async () => ({
   408→          content: [{ text: output }],
   409→          usage: { input_tokens: promptTokens, output_tokens: completionTokens },
   410→        }),
   411→      };
   412→    }
   413→    if (url.includes('generativelanguage')) {
   414→      return {
   415→        ok: true,
   416→        json: async () => ({
   417→          candidates: [{ content: { parts: [{ text: output }] } }],
   418→          usageMetadata: { promptTokenCount: promptTokens, candidatesTokenCount: completionTokens },
   419→        }),
   420→      };
   421→    }
   422→    // OpenAI-compatible (openai, deepseek, together, mistral, alibaba)
   423→    return {
   424→      ok: true,
   425→      json: async () => ({
   426→        choices: [{ message: { content: output } }],
   427→        usage: { prompt_tokens: promptTokens, completion_tokens: completionTokens },
   428→      }),
   429→    };
   430→  });
   431→}

   482→async function runBenchmark(
   483→  router: LLMRouter,
   484→  providers: LLMId[],
   485→  prompts: LLMMessage[][],
   486→  options: { enableFailover?: boolean; cacheEnabled?: boolean } = {},
   487→): Promise<Record<LLMId, RawCallResult[]>> {

   493→  for (const messages of prompts) {
   494→    for (const provider of providers) {
   495→      const start = Date.now();
   496→      try {
   497→        const result = await router.call(provider, messages, {
   498→          retryAttempts: 1,
   499→          retryDelayMs: 10,
   500→          enableFailover: options.enableFailover ?? false,
   501→        });
   502→        const elapsed = Date.now() - start;
   503→
   504→        results[provider].push({
   505→          latency: result.latencyMs || elapsed,
   506→          tokens: result.tokensUsed,
   507→          cost: result.cost,
   508→          failed: result.content === null,
   509→          failover: result.failoverUsed ?? false,
   510→          cacheHit: false, // Would need cache integration for real measurement
   511→        });
   512→      } catch {
   513→        results[provider].push({
   514→          latency: Date.now() - start,
   515→          tokens: 0,
   516→          cost: 0,
   517→          failed: true,
   518→          failover: false,
   519→          cacheHit: false,
   520→        });
   521→      }
   522→    }
   523→  }
   524→
   525→  return results as Record<LLMId, RawCallResult[]>;
   526→}
   527→
   528→function compileResults(
   529→  name: string,
   530→  promptCount: number,
   531→  durationMs: number,
   532→  rawResults: Record<LLMId, RawCallResult[]>,
   533→): BenchmarkResult {
   534→  const providerMetrics: Record<string, ProviderMetrics> = {};
   535→
   536→  for (const provider of ALL_PROVIDERS) {
   537→    const calls = rawResults[provider] || [];
   538→    const successes = calls.filter(c => !c.failed);
   539→    const failures = calls.filter(c => c.failed);
   540→    const failovers = calls.filter(c => c.failover);
   541→
   542→    providerMetrics[provider] = buildMetrics(
   543→      provider,
   544→      successes.map(c => c.latency),
   545→      successes.map(c => c.tokens),
   546→      successes.map(c => c.cost),
   547→      failures.length,
   548→      failovers.length,
   549→      successes.filter(c => c.cacheHit).length,
   550→      successes.filter(c => !c.cacheHit).length,
   551→    );
   552→  }
   553→
   554→  // Determine comparison winners
   555→  const providers = ALL_PROVIDERS.filter(p => providerMetrics[p].successCalls > 0);
   556→  const fastest = providers.reduce((a, b) => providerMetrics[a].avgLatency < providerMetrics[b].avgLatency ? a : b);
   557→  const cheapest = providers.reduce((a, b) => providerMetrics[a].costPer1kTokens < providerMetrics[b].costPer1kTokens ? a : b);
   558→  const mostReliable = providers.reduce((a, b) => providerMetrics[a].errorRate < providerMetrics[b].errorRate ? a : b);
   559→  const highestThroughput = providers.reduce((a, b) => providerMetrics[a].throughputTokensPerSec > providerMetrics[b].throughputTokensPerSec ? a : b);
   560→  const lowestP99 = providers.reduce((a, b) => providerMetrics[a].p99Latency < providerMetrics[b].p99Latency ? a : b);
   561→
   562→  return {
   563→    name,
   564→    timestamp: new Date().toISOString(),
   565→    promptCount,
   566→    totalDurationMs: durationMs,
   567→    providerMetrics: providerMetrics as Record<LLMId, ProviderMetrics>,
   568→    comparison: {
   569→      fastestProvider: fastest,
   570→      cheapestProvider: cheapest,
   571→      mostReliableProvider: mostReliable,
   572→      highestThroughput,
   573→      lowestP99,
   574→    },
   575→  };
   576→}

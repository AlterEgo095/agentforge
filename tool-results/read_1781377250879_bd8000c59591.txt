    18→function getPluginLoader(): { getCapability<T = unknown>(name: string): T | undefined } | null {
    19→  try {
    20→    // eslint-disable-next-line @typescript-eslint/no-require-imports
    21→    const { pluginLoader } = require('../../services/PluginBootstrap');
    22→    return pluginLoader;

    25→    return null;
    26→  }

   131→  // ─── Core Pipeline ─────────────────────────────────────
   132→
   133→  protected async generate(
   134→    prompt: string,
   135→    context: Record<string, unknown>,
   136→    sessionId: string,
   137→  ): Promise<DocOutput> {
   138→    // ── Try WriterWorker plugin delegation first ──
   139→    try {
   140→      const loader = getPluginLoader();
   141→      const writerCap = loader?.getCapability<IWriterCapability>('writing');
   142→      if (writerCap && typeof writerCap.write === 'function') {
   143→        console.log(`[DocAgentWorker] Delegating to WriterWorker plugin (session: ${sessionId})`);
   144→        this.emitEvent('doc_plugin_delegation', sessionId, { plugin: 'writer-worker' });
   145→
   146→        const researchInput = adaptDocInputToResearchInput(prompt, context);
   147→        const writerSession = await writerCap.write(researchInput, {
   148→          defaultContentType: this.mapDocTypeToContentType(this.parseDocumentType(prompt)),
   149→          enableQualityScoring: false, // DocAgentWorker has its own quality scoring
   150→          enableRefinement: false,
   151→        });
   152→
   153→        const docOutput = adaptWriterSessionToDocOutput(writerSession);
   154→
   155→        // Carry author from context if provided
   156→        if (context.author && typeof context.author === 'string') {
   157→          docOutput.author = context.author;
   158→        }
   159→
   160→        console.log(`[DocAgentWorker] WriterWorker plugin succeeded (session: ${sessionId}, sections: ${docOutput.sections.length})`);
   161→        return docOutput;
   162→      }
   163→    } catch (err) {
   164→      console.warn(
   165→        `[DocAgentWorker] WriterWorker plugin failed, falling back to internal pipeline (session: ${sessionId}):`,
   166→        err instanceof Error ? err.message : err,
   167→      );
   168→    }
   169→
   170→    // ── Internal Pipeline (fallback) ──
   171→    console.log(`[DocAgentWorker] Using internal pipeline (session: ${sessionId})`);
   172→
   173→    // Phase 1: Parse document type from prompt
   174→    const docType = this.parseDocumentType(prompt);
   175→    const topic = this.extractTopic(prompt);
   176→
   177→    this.emitEvent('doc_outline', sessionId, { docType, topic });
   178→
   179→    // Phase 2: Build document outline with proper hierarchy
   180→    const outline = this.buildOutline(docType, topic);
   181→
   182→    // Phase 3: Generate content for each section
   183→    // (structured data generation — actual LLM content is provided by the orchestrator)
   184→    const sections = this.generateSectionContent(outline, prompt, context);
   185→
   186→    this.emitEvent('doc_content', sessionId, {
   187→      sectionCount: sections.length,
   188→      totalWordEstimate: this.estimateSectionWords(sections),
   189→    });
   190→
   191→    // Phase 4: Build table of contents from section hierarchy
   192→    const tableOfContents = this.buildToc(sections);
   193→
   194→    // Phase 5: Add references if sources are mentioned or available
   195→    const references = this.extractReferences(prompt, context);
   196→
   197→    // Phase 6: Compute metadata
   198→    const totalWordCount = this.computeTotalWordCount(sections);
   199→    const readingTimeMinutes = Math.max(1, Math.round(totalWordCount / 238)); // avg 238 wpm
   200→
   201→    this.emitEvent('doc_formatting', sessionId, {
   202→      wordCount: totalWordCount,
   203→      readingTimeMinutes,
   204→      format: 'document',
   205→    });
   206→
   207→    return {
   208→      title: this.generateTitle(docType, topic),
   209→      subtitle: this.generateSubtitle(docType, topic),
   210→      author: context.author as string | undefined,
   211→      sections,
   212→      tableOfContents,
   213→      references,
   214→      metadata: {
   215→        wordCount: totalWordCount,
   216→        readingTimeMinutes,
   217→        format: docType,
   218→        language: 'en',
   219→        sourceCount: references.length,
   220→      },
   221→    };
   222→  }

   242→  // ─── Validation ────────────────────────────────────────
   243→
   244→  protected async validateOutput(output: DocOutput, _prompt: string): Promise<QualityReport> {
   245→    const dimensions: Record<string, number> = {};
   246→    const suggestions: string[] = [];
   247→
   248→    // 1. Content Depth (0–1)
   249→    dimensions['Content Depth'] = this.scoreContentDepth(output, suggestions);
   250→
   251→    // 2. Structure & Organization (0–1)
   252→    dimensions['Structure & Organization'] = this.scoreStructure(output, suggestions);
   253→
   254→    // 3. Formatting & Typography (0–1)
   255→    dimensions['Formatting & Typography'] = this.scoreFormatting(output, suggestions);
   256→
   257→    // 4. Accuracy & Citations (0–1)
   258→    dimensions['Accuracy & Citations'] = this.scoreCitations(output, suggestions);
   259→
   260→    // 5. Clarity & Readability (0–1)
   261→    dimensions['Clarity & Readability'] = this.scoreReadability(output, suggestions);
   262→
   263→    const overall = this.computeWeightedScore(dimensions);
   264→    const passed = this.isQualityPassing(overall);
   265→
   266→    return { overall, dimensions, passed, suggestions };
   267→  }

   297→  // ─── Helper: Document Type Parsing ─────────────────────
   298→
   299→  /**
   300→   * Detect the document type from the user prompt by matching against
   301→   * known keyword sets. Falls back to "report".
   302→   */
   303→  parseDocumentType(prompt: string): string {
   304→    const lower = prompt.toLowerCase();
   305→    let bestMatch = 'report';
   306→    let bestScore = 0;
   307→
   308→    for (const [docType, keywords] of Object.entries(DOCUMENT_TYPE_KEYWORDS)) {
   309→      let score = 0;
   310→      for (const keyword of keywords) {
   311→        if (lower.includes(keyword)) {
   312→          score += keyword.length; // longer keyword matches are stronger signals
   313→        }
   314→      }
   315→      if (score > bestScore) {
   316→        bestScore = score;
   317→        bestMatch = docType;
   318→      }
   319→    }
   320→
   321→    return bestMatch;
   322→  }
   323→
   324→  /**
   325→   * Extract the core topic / subject from the prompt.
   326→   * Uses a simple heuristic: strips common command prefixes and returns
   327→   * the remainder trimmed to a reasonable length.
   328→   */
   329→  private extractTopic(prompt: string): string {
   330→    const prefixes = [
   331→      /^(?:write|create|generate|draft|produce|build)\s+(?:a\s+)?(?:report|proposal|manual|whitepaper|memo|letter|article|specification|spec|plan|document)\s+(?:about|on|for|regarding)\s+/i,
   332→      /^(?:write|create|generate|draft|produce|build)\s+(?:a\s+)?/i,
   333→    ];
   334→
   335→    let topic = prompt.trim();
   336→    for (const prefix of prefixes) {
   337→      topic = topic.replace(prefix, '');
   338→    }
   339→
   340→    // Truncate to first sentence or 120 characters whichever is shorter
   341→    const firstSentence = topic.match(/^[^.!?\n]+/);
   342→    if (firstSentence) {
   343→      topic = firstSentence[0].trim();
   344→    }
   345→
   346→    return topic.length > 120 ? topic.slice(0, 117) + '...' : topic;
   347→  }

   349→  // ─── Helper: Outline Building ──────────────────────────
   350→
   351→  /**
   352→   * Create a section hierarchy from a document-type template.
   353→   * Each section gets a unique ID, placeholder content, and recursive
   354→   * children based on the template.
   355→   */
   356→  buildOutline(docType: string, topic: string): DocSection[] {
   357→    const templates = DOCUMENT_TYPE_TEMPLATES[docType] ?? DOCUMENT_TYPE_TEMPLATES['report'];
   358→    return templates.map((tpl, index) => this.instantiateSection(tpl, topic, [index]));
   359→  }
   360→
   361→  private instantiateSection(tpl: SectionTemplate, topic: string, path: number[]): DocSection {
   362→    const id = `sec-${path.join('-')}`;
   363→    const section: DocSection = {
   364→      id,
   365→      heading: tpl.heading,
   366→      level: tpl.level,
   367→      content: this.generatePlaceholderContent(tpl.heading, tpl.level, topic),
   368→    };
   369→
   370→    if (tpl.children && tpl.children.length > 0) {
   371→      section.children = tpl.children.map((child, childIndex) =>
   372→        this.instantiateSection(child, topic, [...path, childIndex]),
   373→      );
   374→    }
   375→
   376→    return section;
   377→  }
   378→
   379→  /**
   380→   * Generate structured placeholder content for a section.
   381→   * Real LLM-generated content would be injected by the orchestrator;
   382→   * this provides a well-structured scaffold with appropriate depth cues.
   383→   */
   384→  private generatePlaceholderContent(heading: string, level: 1 | 2 | 3 | 4, topic: string): string {
   385→    const minWords = level <= 2 ? 150 : 75;
   386→    const base = `This section covers ${heading.toLowerCase()} in the context of ${topic}. `;
   387→
   388→    const elaborations = [
   389→      `The following provides a comprehensive overview of ${heading.toLowerCase()}, including key considerations, analysis, and actionable insights relevant to ${topic}.`,
   390→      `It is essential to understand the foundational aspects presented here, as they form the basis for subsequent sections and recommendations throughout this document.`,
   391→      `The discussion that follows draws upon established practices and contemporary research, ensuring both rigor and relevance in addressing the core themes of ${topic}.`,
   392→      `Stakeholders should consider the interdependencies between this area and the broader strategic context, as outlined in the executive summary and conclusion sections.`,
   393→      `Further detail and supporting evidence can be found in the referenced sources appended to this document.`,
   394→    ];
   395→
   396→    let content = base + elaborations.join(' ');
   397→
   398→    // Ensure minimum word count for level 1–2 sections
   399→    while (this.countWords(content) < minWords) {
   400→      content += ` Additional context and elaboration on ${heading.toLowerCase()} is provided to ensure comprehensive coverage of this important topic. The analysis considers multiple perspectives and draws on both qualitative and quantitative evidence where available.`;
   401→    }
   402→
   403→    return content;
   404→  }

   406→  // ─── Helper: Section Content Generation ────────────────
   407→
   408→  /**
   409→   * Enrich sections with contextual content, tables, images, or code
   410→   * blocks based on the document type and prompt signals.
   411→   */
   412→  private generateSectionContent(
   413→    outline: DocSection[],
   414→    prompt: string,
   415→    context: Record<string, unknown>,
   416→  ): DocSection[] {
   417→    return outline.map((section) => {
   418→      const enriched = { ...section };
   419→
   420→      // If the orchestrator has provided previous output feedback, incorporate it
   421→      if (context.previousOutput && typeof context.previousOutput === 'object') {
   422→        const prev = context.previousOutput as Record<string, unknown>;
   423→        if (prev.sections && Array.isArray(prev.sections)) {
   424→          const matching = (prev.sections as DocSection[]).find((s) => s.heading === section.heading);
   425→          if (matching && matching.content) {
   426→            enriched.content = matching.content;
   427→          }
   428→        }
   429→      }
   430→
   431→      // Recursively enrich children
   432→      if (enriched.children && enriched.children.length > 0) {
   433→        enriched.children = this.generateSectionContent(enriched.children, prompt, context);
   434→      }
   435→
   436→      // Add structural elements based on heading semantics
   437→      enriched.tables = this.suggestTables(enriched.heading, prompt);
   438→      enriched.images = this.suggestImages(enriched.heading, prompt);
   439→      enriched.codeBlocks = this.suggestCodeBlocks(enriched.heading, prompt);
   440→
   441→      return enriched;
   442→    });
   443→  }

   458→          ['Item 2', 'Description of item 2', 'Additional detail', 'Pending'],
   459→          ['Item 3', 'Description of item 3', 'Additional detail', 'Pending'],
   460→        ],
   461→        caption: `${heading} — Overview Table`,
   462→      },
   463→    ];
   464→  }

   485→  private suggestCodeBlocks(heading: string, _prompt: string): DocCodeBlock[] | undefined {

   499→  // ─── Helper: Table of Contents ─────────────────────────
   500→
   501→  /**
   502→   * Generate a recursive Table of Contents from the section hierarchy.
   503→   */
   504→  buildToc(sections: DocSection[]): TocEntry[] {
   505→    return sections.map((section) => {
   506→      const entry: TocEntry = {
   507→        id: section.id,
   508→        title: section.heading,
   509→        level: section.level,
   510→      };
   511→
   512→      if (section.children && section.children.length > 0) {
   513→        entry.children = this.buildToc(section.children);
   514→      }
   515→
   516→      return entry;
   517→    });
   518→  }

   520→  // ─── Helper: References ────────────────────────────────
   521→
   522→  /**
   523→   * Extract references from prompt and context. In production this
   524→   * would be powered by the RECHERCHE agent; here we provide a
   525→   * scaffold that detects explicit source mentions.
   526→   */
   527→  private extractReferences(prompt: string, context: Record<string, unknown>): DocReference[] {
   528→    const references: DocReference[] = [];
   529→    const sourcePattern = /(?:source|ref|cite|according to|referenced?)(?:\s+(?:to|from))?\s+["']([^"']+)["']/gi;
   530→    let match: RegExpExecArray | null;
   531→
   532→    while ((match = sourcePattern.exec(prompt)) !== null) {
   533→      references.push({
   534→        id: `ref-${references.length + 1}`,
   535→        type: 'other',
   536→        title: match[1],
   537→      });
   538→    }
   539→
   540→    // Also include any references passed through context (e.g. from RECHERCHE agent)
   541→    if (context.references && Array.isArray(context.references)) {
   542→      for (const ref of context.references as DocReference[]) {
   543→        if (ref.id && ref.title) {
   544→          references.push(ref);
   545→        }
   546→      }

   610→    }

   620→  private scoreContentDepth(output: DocOutput, suggestions: string[]): number {
   621→    let totalSections = 0;
   622→    let deepSections = 0;
   623→
   624→    const walk = (sections: DocSection[]): void => {
   625→      for (const section of sections) {
   626→        totalSections++;
   627→        const wordCount = this.countWords(section.content);
   628→        // Level 1–2 sections should have at least 150 words; levels 3–4 at least 75
   629→        const minimum = section.level <= 2 ? 150 : 75;
   630→        if (wordCount >= minimum) {
   631→          deepSections++;
   632→        } else {
   633→          suggestions.push(
   634→            `Section "${section.heading}" has only ${wordCount} words (minimum: ${minimum}). Consider expanding the content.`,
   635→          );
   636→        }
   637→        if (section.children) walk(section.children);
   638→      }
   639→    };
   640→
   641→    walk(output.sections);
   642→
   643→    if (totalSections === 0) return 0;
   644→    return deepSections / totalSections;
   645→  }

   647→  private scoreStructure(output: DocOutput, suggestions: string[]): number {
   648→    let score = 1.0;
   649→
   650→    // Must have at least 2 top-level sections
   651→    if (output.sections.length < 2) {
   652→      score -= 0.3;
   653→      suggestions.push('Document should have at least 2 top-level sections.');
   654→    }
   655→
   656→    // Validate heading hierarchy — each section's level should not skip (e.g. 1 → 3)
   657→    let prevLevel = 0;
   658→    const checkHierarchy = (sections: DocSection[]): void => {
   659→      for (const section of sections) {
   660→        if (prevLevel > 0 && section.level > prevLevel + 1) {
   661→          score -= 0.1;
   662→          suggestions.push(
   663→            `Heading hierarchy skip detected: level ${prevLevel} → ${section.level} at "${section.heading}".`,
   664→          );
   665→        }
   666→        prevLevel = section.level;
   667→        if (section.children) checkHierarchy(section.children);
   668→      }
   669→    };
   670→    checkHierarchy(output.sections);
   671→
   672→    // TOC should match sections
   673→    if (output.tableOfContents.length !== output.sections.length) {
   674→      score -= 0.15;
   675→      suggestions.push('Table of contents entry count does not match top-level sections.');
   676→    }
   677→
   678→    return Math.max(0, Math.min(1, score));
   679→  }

   681→  private scoreFormatting(output: DocOutput, suggestions: string[]): number {
   682→    let score = 1.0;
   683→    const headingSet = new Set<string>();
   684→    let duplicateHeadings = 0;
   685→
   686→    const walk = (sections: DocSection[]): void => {
   687→      for (const section of sections) {
   688→        if (headingSet.has(section.heading.toLowerCase())) {
   689→          duplicateHeadings++;
   690→        }
   691→        headingSet.add(section.heading.toLowerCase());
   692→
   693→        // Check for empty content
   694→        if (!section.content || section.content.trim().length === 0) {
   695→          score -= 0.15;
   696→          suggestions.push(`Section "${section.heading}" has empty content.`);
   697→        }
   698→
   699→        // Validate tables have rows
   700→        if (section.tables) {
   701→          for (let i = 0; i < section.tables.length; i++) {
   702→            const table = section.tables[i];
   703→            if (table.rows.length === 0) {
   704→              score -= 0.05;
   705→              suggestions.push(`Table ${i + 1} in "${section.heading}" has no rows.`);
   706→            }
   707→            if (table.headers.length === 0) {
   708→              score -= 0.05;
   709→              suggestions.push(`Table ${i + 1} in "${section.heading}" has no headers.`);
   710→            }
   711→          }
   712→        }
   713→
   714→        // Validate code blocks have code
   715→        if (section.codeBlocks) {
   716→          for (let i = 0; i < section.codeBlocks.length; i++) {
   717→            if (!section.codeBlocks[i].code || section.codeBlocks[i].code.trim().length === 0) {
   718→              score -= 0.05;
   719→              suggestions.push(`Code block ${i + 1} in "${section.heading}" is empty.`);
   720→            }
   721→          }
   722→        }
   723→
   724→        if (section.children) walk(section.children);
   725→      }
   726→    };
   727→
   728→    walk(output.sections);
   729→
   730→    if (duplicateHeadings > 0) {
   731→      score -= 0.1 * duplicateHeadings;
   732→      suggestions.push(`${duplicateHeadings} duplicate heading(s) detected. Use unique headings for each section.`);
   733→    }
   734→
   735→    return Math.max(0, Math.min(1, score));
   736→  }

   738→  private scoreCitations(output: DocOutput, suggestions: string[]): number {
   739→    const refCount = output.references.length;
   740→
   741→    if (refCount === 0) {
   742→      // Not necessarily a failure for shorter document types
   743→      const docTypesRequiringRefs = ['whitepaper', 'report', 'specification', 'proposal'];
   744→      if (docTypesRequiringRefs.includes(output.metadata.format)) {
   745→        suggestions.push(`${output.metadata.format} documents typically require references. Consider adding sources.`);
   746→        return 0.4;
   747→      }
   748→      return 0.8; // acceptable for memos, letters, etc.
   749→    }
   750→
   751→    // Check reference completeness
   752→    let incomplete = 0;
   753→    for (const ref of output.references) {
   754→      if (!ref.title || ref.title.trim().length === 0) incomplete++;
   755→    }
   756→
   757→    if (incomplete > 0) {
   758→      suggestions.push(`${incomplete} reference(s) have incomplete information.`);
   759→    }
   760→
   761→    // Score based on completeness ratio
   762→    const completeness = 1 - incomplete / refCount;
   763→    // Bonus for having multiple well-formed references
   764→    const countBonus = Math.min(refCount / 5, 1) * 0.2;
   765→    return Math.min(1, completeness * 0.8 + countBonus);
   766→  }

   768→  private scoreReadability(output: DocOutput, suggestions: string[]): number {
   769→    let totalWords = 0;
   770→    let totalSentences = 0;
   771→    let totalSyllables = 0;
   772→
   773→    const walk = (sections: DocSection[]): void => {
   774→      for (const section of sections) {
   775→        const text = section.content;
   776→        totalWords += this.countWords(text);
   777→        totalSentences += this.countSentences(text);
   778→        totalSyllables += this.estimateSyllables(text);
   779→
   780→        if (section.children) walk(section.children);
   781→      }
   782→    };
   783→
   784→    walk(output.sections);
   785→
   786→    if (totalSentences === 0 || totalWords === 0) return 0;
   787→
   788→    // Flesch-Kincaid Grade Level (approximate)
   789→    const fkGrade = 0.39 * (totalWords / totalSentences) + 11.8 * (totalSyllables / totalWords) - 15.59;
   790→
   791→    // Ideal grade level for professional documents: 8–12
   792→    // Score 1.0 for grade 10, decreasing as we move away
   793→    const idealGrade = 10;
   794→    const deviation = Math.abs(fkGrade - idealGrade);
   795→    const score = Math.max(0, 1 - deviation / 10);
   796→
   797→    if (fkGrade > 14) {
   798→      suggestions.push('Document may be too complex. Consider simplifying sentence structures.');
   799→    } else if (fkGrade < 6) {
   800→      suggestions.push('Document readability is very low grade-level. Consider if the tone is appropriate for the audience.');
   801→    }
   802→
   803→    return Math.min(1, Math.max(0, score));
   804→  }
   805→
   806→  private countSentences(text: string): number {
   807→    const sentences = text.split(/[.!?]+/).filter((s) => s.trim().length > 0);
   808→    return Math.max(1, sentences.length);
   809→  }

   811→  private estimateSyllables(text: string): number {
   812→    // Simple heuristic: count vowel groups per word
   813→    const words = text.split(/\s+/).filter((w) => w.length > 0);
   814→    let total = 0;
   815→    for (const word of words) {
   816→      const vowelGroups = word.toLowerCase().replace(/[^a-z]/g, '').match(/[aeiouy]+/g);
   817→      total += vowelGroups ? Math.max(1, vowelGroups.length) : 1;
   818→    }
   819→    return total;
   820→  }

   824→  renderMarkdown(output: DocOutput): string {

   875→    lines.push(`${headingPrefix} ${section.heading}`);

   880→    // Tables

   888→    // Images

   898→

   900→    if (section.codeBlocks) {

   904→        lines.push('```');

   906→        lines.push('');
   907→      }
   908→    }
   909→
   910→    // Children
   911→    if (section.children) {
   912→      for (const child of section.children) {
   913→        lines.push(this.renderSectionMarkdown(child));
   914→      }
   915→    }
   916→
   917→    return lines.join('\n');
   918→  }

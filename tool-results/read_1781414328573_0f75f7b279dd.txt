     1→// ============================================================
     2→// AgentForge - Admin Dashboard
     3→// Comprehensive system admin with tabbed navigation
     4→// ============================================================
     5→
     6→import React, { useState, useEffect, useCallback } from 'react';
     7→import {
     8→  Shield, Users, Cpu, GitBranch, DollarSign, Lock,
     9→  Server, Activity, RefreshCw, X, Loader2, Check,
    10→  AlertTriangle, Clock, TrendingUp, TrendingDown, Wifi,
    11→  WifiOff, AlertCircle, Eye,
    12→} from 'lucide-react';
    13→import {
    14→  api,
    15→  type AdminStats, type AdminUser, type ProviderInfo, type ProviderHealth,
    16→  type OrchestrationStatus, type FinanceData, type SecurityEvent,
    17→  type InfrastructureData, type MonitoringData,
    18→} from '../lib/api';
    19→import { cn } from '../lib/cn';
    20→
    21→type TabId = 'overview' | 'users' | 'providers' | 'orchestration' | 'finances' | 'security' | 'infrastructure' | 'monitoring';
    22→
    23→const TABS: { id: TabId; label: string; icon: React.ElementType }[] = [
    24→  { id: 'overview', label: 'Overview', icon: Activity },
    25→  { id: 'users', label: 'Users', icon: Users },
    26→  { id: 'providers', label: 'Providers', icon: Cpu },
    27→  { id: 'orchestration', label: 'Orchestration', icon: GitBranch },
    28→  { id: 'finances', label: 'Finances', icon: DollarSign },
    29→  { id: 'security', label: 'Security', icon: Lock },
    30→  { id: 'infrastructure', label: 'Infrastructure', icon: Server },
    31→  { id: 'monitoring', label: 'Monitoring', icon: Activity },
    32→];
    33→
    34→export function Admin() {
    35→  const [activeTab, setActiveTab] = useState<TabId>('overview');
    36→  const [stats, setStats] = useState<AdminStats | null>(null);
    37→  const [users, setUsers] = useState<AdminUser[]>([]);
    38→  const [providers, setProviders] = useState<ProviderInfo[]>([]);
    39→  const [providerHealth, setProviderHealth] = useState<Record<string, ProviderHealth>>({});
    40→  const [orchestration, setOrchestration] = useState<OrchestrationStatus | null>(null);
    41→  const [finances, setFinances] = useState<FinanceData | null>(null);
    42→  const [securityEvents, setSecurityEvents] = useState<SecurityEvent[]>([]);
    43→  const [infrastructure, setInfrastructure] = useState<InfrastructureData | null>(null);
    44→  const [monitoring, setMonitoring] = useState<MonitoringData | null>(null);
    45→  const [loading, setLoading] = useState(true);
    46→  const [error, setError] = useState<string | null>(null);
    47→
    48→  const fetchData = useCallback(async () => {
    49→    setLoading(true);
    50→    setError(null);
    51→    try {
    52→      const [statsRes, usersRes, providersRes, orchRes, finRes, secRes, infraRes, monRes] = await Promise.allSettled([
    53→        api.getAdminStats(),
    54→        api.listUsers(),
    55→        api.getProviders(),
    56→        api.getOrchestration(),
    57→        api.getFinances(),
    58→        api.getSecurityEvents(),
    59→        api.getInfrastructure(),
    60→        api.getMonitoring(),
    61→      ]);
    62→      if (statsRes.status === 'fulfilled') setStats(statsRes.value.data);
    63→      if (usersRes.status === 'fulfilled') setUsers(usersRes.value.data || []);
    64→      if (providersRes.status === 'fulfilled') setProviders(providersRes.value.data || []);
    65→      if (orchRes.status === 'fulfilled') setOrchestration(orchRes.value.data);
    66→      if (finRes.status === 'fulfilled') setFinances(finRes.value.data);
    67→      if (secRes.status === 'fulfilled') setSecurityEvents(secRes.value.data || []);
    68→      if (infraRes.status === 'fulfilled') setInfrastructure(infraRes.value.data);
    69→      if (monRes.status === 'fulfilled') setMonitoring(monRes.value.data);
    70→    } catch (err: any) {
    71→      setError(err.message || 'Failed to load admin data');
    72→    } finally {
    73→      setLoading(false);
    74→    }
    75→  }, []);
    76→
    77→  useEffect(() => {
    78→    fetchData();
    79→  }, [fetchData]);
    80→
    81→  return (
    82→    <div className="p-6 lg:p-8 max-w-7xl mx-auto animate-fade-in">
    83→      {/* Header */}
    84→      <div className="flex items-center justify-between mb-6">
    85→        <div>
    86→          <div className="flex items-center gap-3 mb-1">
    87→            <Shield className="w-7 h-7 text-amber-400" />
    88→            <h1 className="text-3xl font-bold gradient-text">Admin</h1>
    89→          </div>
    90→          <p className="text-gray-500 text-sm">System administration and monitoring</p>
    91→        </div>
    92→        <button
    93→          onClick={fetchData}
    94→          className="p-2.5 rounded-lg bg-gray-800 hover:bg-gray-700 text-gray-400 hover:text-white transition-colors"
    95→        >
    96→          <RefreshCw className={cn('w-4 h-4', loading && 'animate-spin')} />
    97→        </button>
    98→      </div>
    99→
   100→      {/* Error */}
   101→      {error && (
   102→        <div className="mb-4 px-4 py-3 rounded-lg bg-red-500/10 border border-red-500/20 text-red-400 text-sm flex items-center justify-between">
   103→          <span>{error}</span>
   104→          <button onClick={() => setError(null)}><X className="w-4 h-4" /></button>
   105→        </div>
   106→      )}
   107→
   108→      {/* Tab Navigation */}
   109→      <div className="mb-6 overflow-x-auto">
   110→        <div className="flex gap-1 p-1 glass-card rounded-xl min-w-max">
   111→          {TABS.map(({ id, label, icon: Icon }) => (
   112→            <button
   113→              key={id}
   114→              onClick={() => setActiveTab(id)}
   115→              className={cn(
   116→                'flex items-center gap-2 px-4 py-2 rounded-lg text-sm font-medium transition-all whitespace-nowrap',
   117→                activeTab === id
   118→                  ? 'bg-blue-500/10 text-blue-400'
   119→                  : 'text-gray-400 hover:text-white hover:bg-gray-800/50'
   120→              )}
   121→            >
   122→              <Icon className="w-4 h-4" />
   123→              {label}
   124→            </button>
   125→          ))}
   126→        </div>
   127→      </div>
   128→
   129→      {/* Tab Content */}
   130→      {loading && !stats ? (
   131→        <div className="flex items-center justify-center py-20">
   132→          <Loader2 className="w-8 h-8 text-blue-400 animate-spin" />
   133→        </div>
   134→      ) : (
   135→        <>
   136→          {activeTab === 'overview' && <OverviewTab stats={stats} providers={providers} />}
   137→          {activeTab === 'users' && <UsersTab users={users} onUpdate={fetchData} />}
   138→          {activeTab === 'providers' && <ProvidersTab providers={providers} />}
   139→          {activeTab === 'orchestration' && <OrchestrationTab orchestration={orchestration} />}
   140→          {activeTab === 'finances' && <FinancesTab finances={finances} />}
   141→          {activeTab === 'security' && <SecurityTab events={securityEvents} />}
   142→          {activeTab === 'infrastructure' && <InfrastructureTab infrastructure={infrastructure} />}
   143→          {activeTab === 'monitoring' && <MonitoringTab monitoring={monitoring} />}
   144→        </>
   145→      )}
   146→    </div>
   147→  );
   148→}

   209→// ── Users Tab ─────────────────────────────────────────────
   210→
   211→function UsersTab({ users, onUpdate }: { users: AdminUser[]; onUpdate: () => void }) {
   212→  const [editingId, setEditingId] = useState<string | null>(null);
   213→  const [editData, setEditData] = useState<{ tier: string; role: string; status: string }>({ tier: '', role: '', status: '' });
   214→  const [saving, setSaving] = useState(false);
   215→
   216→  const startEdit = (u: AdminUser) => {
   217→    setEditingId(u.id);
   218→    setEditData({ tier: u.tier, role: u.role, status: u.status });
   219→  };
   220→
   221→  const saveEdit = async () => {
   222→    if (!editingId) return;
   223→    try {
   224→      setSaving(true);
   225→      await api.updateUser(editingId, editData as any);
   226→      setEditingId(null);
   227→      onUpdate();
   228→    } catch {
   229→      // silent
   230→    } finally {
   231→      setSaving(false);
   232→    }
   233→  };
   234→
   235→  return (
   236→    <div className="glass-card rounded-xl overflow-hidden animate-fade-in">
   237→      <table className="w-full text-left">
   238→        <thead>
   239→          <tr>
   240→            <th className="text-xs text-gray-500 uppercase tracking-wider py-3 px-4 border-b border-gray-800">User</th>
   241→            <th className="text-xs text-gray-500 uppercase tracking-wider py-3 px-4 border-b border-gray-800">Tier</th>
   242→            <th className="text-xs text-gray-500 uppercase tracking-wider py-3 px-4 border-b border-gray-800">Role</th>
   243→            <th className="text-xs text-gray-500 uppercase tracking-wider py-3 px-4 border-b border-gray-800">Status</th>
   244→            <th className="text-xs text-gray-500 uppercase tracking-wider py-3 px-4 border-b border-gray-800 text-right">Actions</th>
   245→          </tr>
   246→        </thead>
   247→        <tbody>
   248→          {users.map((u) => (
   249→            <tr key={u.id} className="border-b border-gray-800/50 hover:bg-gray-800/30">
   250→              <td className="py-3 px-4">
   251→                <div className="flex items-center gap-3">
   252→                  <div className="w-8 h-8 rounded-full bg-gradient-to-br from-blue-500 to-purple-600 flex items-center justify-center text-xs font-bold text-white">
   253→                    {u.name?.[0]?.toUpperCase() || '?'}
   254→                  </div>
   255→                  <div>
   256→                    <p className="text-sm font-medium text-white">{u.name}</p>
   257→                    <p className="text-xs text-gray-500">{u.email}</p>
   258→                  </div>
   259→                </div>
   260→              </td>
   261→              <td className="py-3 px-4">
   262→                {editingId === u.id ? (
   263→                  <select value={editData.tier} onChange={(e) => setEditData({ ...editData, tier: e.target.value })} className="bg-gray-800 border border-gray-700 text-xs rounded px-2 py-1 text-gray-300">
   264→                    {['free', 'starter', 'pro', 'enterprise'].map((t) => <option key={t} value={t}>{t}</option>)}
   265→                  </select>
   266→                ) : (
   267→                  <span className="px-2 py-0.5 rounded-full text-xs font-medium bg-purple-500/10 text-purple-400 capitalize">{u.tier}</span>
   268→                )}
   269→              </td>
   270→              <td className="py-3 px-4">
   271→                {editingId === u.id ? (
   272→                  <select value={editData.role} onChange={(e) => setEditData({ ...editData, role: e.target.value })} className="bg-gray-800 border border-gray-700 text-xs rounded px-2 py-1 text-gray-300">
   273→                    {['user', 'admin', 'superadmin'].map((r) => <option key={r} value={r}>{r}</option>)}
   274→                  </select>
   275→                ) : (
   276→                  <span className="px-2 py-0.5 rounded-full text-xs font-medium bg-blue-500/10 text-blue-400 capitalize">{u.role}</span>
   277→                )}
   278→              </td>
   279→              <td className="py-3 px-4">
   280→                {editingId === u.id ? (
   281→                  <select value={editData.status} onChange={(e) => setEditData({ ...editData, status: e.target.value })} className="bg-gray-800 border border-gray-700 text-xs rounded px-2 py-1 text-gray-300">
   282→                    {['active', 'inactive', 'suspended'].map((s) => <option key={s} value={s}>{s}</option>)}
   283→                  </select>
   284→                ) : (
   285→                  <span className={cn('px-2 py-0.5 rounded-full text-xs font-medium capitalize',
   286→                    u.status === 'active' ? 'bg-green-500/10 text-green-400' : u.status === 'suspended' ? 'bg-red-500/10 text-red-400' : 'bg-gray-500/10 text-gray-400'
   287→                  )}>{u.status}</span>
   288→                )}
   289→              </td>
   290→              <td className="py-3 px-4 text-right">
   291→                {editingId === u.id ? (
   292→                  <div className="flex items-center justify-end gap-2">
   293→                    <button onClick={saveEdit} disabled={saving} className="p-1.5 rounded text-green-400 hover:bg-green-500/10">
   294→                      {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Check className="w-4 h-4" />}
   295→                    </button>
   296→                    <button onClick={() => setEditingId(null)} className="p-1.5 rounded text-gray-500 hover:bg-gray-800">
   297→                      <X className="w-4 h-4" />
   298→                    </button>
   299→                  </div>
   300→                ) : (
   301→                  <button onClick={() => startEdit(u)} className="text-xs text-gray-500 hover:text-blue-400 transition-colors">
   302→                    Edit
   303→                  </button>
   304→                )}
   305→              </td>
   306→            </tr>
   307→          ))}
   308→          {users.length === 0 && (
   309→            <tr>
   310→              <td colSpan={5} className="py-12 text-center text-gray-600 text-sm">No users found</td>
   311→            </tr>
   312→          )}
   313→        </tbody>
   314→      </table>
   315→    </div>
   316→  );
   317→}

   445→function FinancesTab({ finances }: { finances: FinanceData | null }) {
   446→  if (!finances) {
   447→    return (
   448→      <div className="glass-card rounded-xl p-12 text-center text-gray-600 animate-fade-in">

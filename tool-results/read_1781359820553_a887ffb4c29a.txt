    61→    61→  constructor() {

    80→    80→  async execute(agentId: AgentId, prompt: string, userId: string, projectId: string): Promise<AgentExecution> {
    81→    81→    const config = AGENT_REGISTRY[agentId];
    82→    82→    if (!config) throw new Error(`Unknown agent: ${agentId}`);
    83→    83→
    84→    84→    // Validate DAG structure
    85→    85→    const dagValidation = validateDAG(config.dagTemplate);
    86→    86→    if (!dagValidation.valid) {
    87→    87→      throw new Error(`Invalid DAG for agent ${agentId}: ${dagValidation.errors.join(', ')}`);
    88→    88→    }
    89→    89→
    90→    90→    const execution: AgentExecution = {
    91→    91→      id: uuidv4(),
    92→    92→      agentId,
    93→    93→      userId,
    94→    94→      projectId,
    95→    95→      status: 'running',
    96→    96→      input: prompt,
    97→    97→      dagExecution: {
    98→    98→        currentLevel: 0,
    99→    99→        completedNodes: [],
   100→   100→        activeNodes: [],
   101→   101→        pendingNodes: config.dagTemplate.nodes.map(n => n.id),
   102→   102→        results: {},
   103→   103→      },
   104→   104→      reflectionScores: [],
   105→   105→      iterations: 0,
   106→   106→      totalTokensUsed: 0,
   107→   107→      totalCost: 0,
   108→   108→      createdAt: new Date(),
   109→   109→    };
   110→   110→
   111→   111→    // Execution timeline for observability
   112→   112→    const timeline: TimelineEntry[] = [];
   113→   113→    const addTimeline = (phase: TimelineEntry['phase'], detail: string, durationMs?: number) => {
   114→   114→      const entry: TimelineEntry = { timestamp: Date.now(), phase, detail, durationMs };
   115→   115→      timeline.push(entry);
   116→   116→      this.emitEvent('timeline_entry', execution.id, agentId, entry);
   117→   117→    };
   118→   118→
   119→   119→    // Register abort controller
   120→   120→    const abortController = new AbortController();
   121→   121→    this.activeExecutions.set(execution.id, abortController);
   122→   122→
   123→   123→    this.emitEvent('execution_start', execution.id, agentId, { prompt: prompt.substring(0, 200) });
   124→   124→
   125→   125→    try {
   126→   126→      // ═══════════════════════════════════════════
   127→   127→      // STEP 1: Check Cache
   128→   128→      // ═══════════════════════════════════════════
   129→   129→      const cacheStart = Date.now();
   130→   130→      const cached = await this.cache.getByAgentPrompt(agentId, prompt);
   131→   131→      addTimeline('cache_check', cached ? 'Cache HIT' : 'Cache MISS', Date.now() - cacheStart);
   132→   132→
   133→   133→      if (cached) {
   134→   134→        this.emitEvent('cache_hit', execution.id, agentId, { cacheHit: true });
   135→   135→        execution.output = cached.output;
   136→   136→        execution.status = 'completed';
   137→   137→        execution.totalCost = 0;
   138→   138→        execution.completedAt = new Date();
   139→   139→        return execution;
   140→   140→      }
   141→   141→
   142→   142→      // ═══════════════════════════════════════════
   143→   143→      // STEP 2: Optimize LLM Selection (RL-inspired)
   144→   144→      // ═══════════════════════════════════════════
   145→   145→      const selectStart = Date.now();
   146→   146→      const selectedProviders = this.costOptimizer.selectProviders(config);
   147→   147→      addTimeline('provider_selection', `Selected ${selectedProviders.length} providers: ${selectedProviders.join(', ')}`, Date.now() - selectStart);
   148→   148→
   149→   149→      this.emitEvent('execution_start', execution.id, agentId, {
   150→   150→        providers: selectedProviders,
   151→   151→        dagNodes: config.dagTemplate.nodes.length,
   152→   152→      });
   153→   153→
   154→   154→      // ═══════════════════════════════════════════
   155→   155→      // STEP 2.5: Build Project Context (if prompt references files)
   156→   156→      // ═══════════════════════════════════════════
   157→   157→      let currentOutput = prompt;
   158→   158→      const contextStart = Date.now();
   159→   159→      const projectContext = this.buildProjectContext(prompt, 8000);
   160→   160→      if (projectContext) {
   161→   161→        currentOutput = projectContext + '\n\n' + currentOutput;
   162→   162→        addTimeline('context_build', `Built project context: ${this.tokenEstimator.estimate(projectContext)} tokens injected`, Date.now() - contextStart);
   163→   163→      } else {
   164→   164→        addTimeline('context_build', 'No project context needed', Date.now() - contextStart);
   165→   165→      }
   166→   166→
   167→   167→      // ═══════════════════════════════════════════
   168→   168→      // STEP 3: Execute DAG Level by Level (Parallel)
   169→   169→      // ═══════════════════════════════════════════
   170→   170→      const levels = topologicalSort(config.dagTemplate);
   171→   171→      let lastReflectionFeedback: string | undefined;
   172→   172→
   173→   173→      for (let levelIndex = 0; levelIndex < levels.length; levelIndex++) {
   174→   174→        // Check for cancellation
   175→   175→        if (abortController.signal.aborted) {
   176→   176→          execution.status = 'cancelled';
   177→   177→          execution.completedAt = new Date();
   178→   178→          return execution;
   179→   179→        }
   180→   180→
   181→   181→        execution.dagExecution.currentLevel = levelIndex;
   182→   182→        execution.dagExecution.activeNodes = levels[levelIndex].map(n => n.id);
   183→   183→
   184→   184→        const levelStart = Date.now();
   185→   185→        this.emitEvent('dag_level_start', execution.id, agentId, {
   186→   186→          level: levelIndex,
   187→   187→          nodes: levels[levelIndex].map(n => ({ id: n.id, label: n.label })),
   188→   188→        });
   189→   189→
   190→   190→        // Execute all nodes in this level IN PARALLEL
   191→   191→        // Pass reflection feedback into MoA for feedback injection
   192→   192→        const levelPromises = levels[levelIndex].map(node =>
   193→   193→          this.executeNode(
   194→   194→            node, currentOutput, config, selectedProviders, execution.id, agentId,
   195→   195→            lastReflectionFeedback
   196→   196→          )
   197→   197→        );
   198→   198→        const levelResults = await Promise.allSettled(levelPromises);
   199→   199→
   200→   200→        // Process results
   201→   201→        const successfulResults = levelResults
   202→   202→          .map((r, i) => {
   203→   203→            if (r.status === 'fulfilled') {
   204→   204→              return { node: levels[levelIndex][i], result: r.value };
   205→   205→            }
   206→   206→            console.error(`[SuperAgent] Node ${levels[levelIndex][i].id} failed:`, r.reason);
   207→   207→            return null;
   208→   208→          })
   209→   209→          .filter((r): r is { node: DAGNode; result: { output: string; tokensUsed: number; cost: number } } => r !== null);
   210→   210→
   211→   211→        for (const { node, result } of successfulResults) {
   212→   212→          execution.dagExecution.results[node.id] = result.output;
   213→   213→          execution.dagExecution.completedNodes.push(node.id);
   214→   214→          execution.totalTokensUsed += result.tokensUsed || 0;
   215→   215→          execution.totalCost += result.cost || 0;
   216→   216→
   217→   217→          this.emitEvent('dag_node_complete', execution.id, agentId, {
   218→   218→            nodeId: node.id,
   219→   219→            nodeLabel: node.label,
   220→   220→            outputLength: result.output.length,
   221→   221→          });
   222→   222→        }
   223→   223→
   224→   224→        // Merge outputs for next level — smart merge based on node types
   225→   225→        currentOutput = this.mergeLevelOutputs(levels[levelIndex], successfulResults);
   226→   226→
   227→   227→        execution.dagExecution.pendingNodes = execution.dagExecution.pendingNodes.filter(
   228→   228→          id => !execution.dagExecution.completedNodes.includes(id)
   229→   229→        );
   230→   230→
   231→   231→        addTimeline('dag_level', `Level ${levelIndex}: ${successfulResults.length}/${levels[levelIndex].length} nodes completed`, Date.now() - levelStart);
   232→   232→
   233→   233→        this.emitEvent('dag_level_complete', execution.id, agentId, {
   234→   234→          level: levelIndex,
   235→   235→          completedCount: successfulResults.length,
   236→   236→        });
   237→   237→      }
   238→   238→
   239→   239→      // ═══════════════════════════════════════════
   240→   240→      // STEP 4: Reflection Loop with 3-Level Auto-Fix
   241→   241→      // ═══════════════════════════════════════════
   242→   242→      execution.status = 'reflecting';
   243→   243→      const reflectionStart = Date.now();
   244→   244→      this.emitEvent('reflection_start', execution.id, agentId, { iteration: 0 });
   245→   245→
   246→   246→      let reflectionResult = await this.reflectionAgent.evaluate(
   247→   247→        currentOutput,
   248→   248→        config.reflectionCriteria,
   249→   249→        config.confidenceThreshold
   250→   250→      );
   251→   251→
   252→   252→      execution.reflectionScores.push(...reflectionResult.scores);
   253→   253→      execution.iterations = 1;
   254→   254→      lastReflectionFeedback = reflectionResult.feedback;
   255→   255→
   256→   256→      addTimeline('reflection', `Iteration 1: passed=${reflectionResult.passed}, confidence=${reflectionResult.confidence.toFixed(2)}`, Date.now() - reflectionStart);
   257→   257→
   258→   258→      this.emitEvent('reflection_complete', execution.id, agentId, {
   259→   259→        passed: reflectionResult.passed,
   260→   260→        confidence: reflectionResult.confidence,
   261→   261→        iteration: 1,
   262→   262→      });
   263→   263→
   264→   264→      // Auto-fix loop if below threshold
   265→   265→      let autoFixAttempt = 0;
   266→   266→      const maxFixAttempts = config.maxIterations;
   267→   267→      const autoFixAttempts: AutoFixAttempt[] = [];
   268→   268→
   269→   269→      while (!reflectionResult.passed && autoFixAttempt < maxFixAttempts) {
   270→   270→        autoFixAttempt++;
   271→   271→        execution.status = 'fixing';
   272→   272→        const fixStart = Date.now();
   273→   273→
   274→   274→        this.emitEvent('autofix_start', execution.id, agentId, {
   275→   275→          attempt: autoFixAttempt,
   276→   276→          maxAttempts: maxFixAttempts,
   277→   277→        });
   278→   278→
   279→   279→        const fixResult = await this.executeAutoFix(
   280→   280→          currentOutput,
   281→   281→          reflectionResult.feedback,
   282→   282→          selectedProviders,
   283→   283→          autoFixAttempt,
   284→   284→          maxFixAttempts
   285→   285→        );
   286→   286→
   287→   287→        autoFixAttempts.push(fixResult);
   288→   288→
   289→   289→        if (fixResult.success) {
   290→   290→          currentOutput = fixResult.output;
   291→   291→
   292→   292→          // Re-evaluate after fix
   293→   293→          const reEvalStart = Date.now();
   294→   294→          this.emitEvent('reflection_start', execution.id, agentId, { iteration: autoFixAttempt + 1 });
   295→   295→          reflectionResult = await this.reflectionAgent.evaluate(
   296→   296→            currentOutput,
   297→   297→            config.reflectionCriteria,
   298→   298→            config.confidenceThreshold
   299→   299→          );
   300→   300→          execution.reflectionScores.push(...reflectionResult.scores);
   301→   301→          execution.iterations = 1 + autoFixAttempt;
   302→   302→          lastReflectionFeedback = reflectionResult.feedback;
   303→   303→
   304→   304→          addTimeline('reflection', `Re-eval iteration ${autoFixAttempt + 1}: passed=${reflectionResult.passed}, confidence=${reflectionResult.confidence.toFixed(2)}`, Date.now() - reEvalStart);
   305→   305→
   306→   306→          this.emitEvent('reflection_complete', execution.id, agentId, {
   307→   307→            passed: reflectionResult.passed,
   308→   308→            confidence: reflectionResult.confidence,
   309→   309→            iteration: autoFixAttempt + 1,
   310→   310→          });
   311→   311→        }
   312→   312→
   313→   313→        addTimeline('autofix', `Attempt ${autoFixAttempt}: success=${fixResult.success}, level=${fixResult.level}`, Date.now() - fixStart);
   314→   314→
   315→   315→        this.emitEvent('autofix_complete', execution.id, agentId, {
   316→   316→          attempt: autoFixAttempt,
   317→   317→          success: fixResult.success,
   318→   318→          level: fixResult.level,
   319→   319→        });
   320→   320→      }
   321→   321→
   322→   322→      // ═══════════════════════════════════════════
   323→   323→      // STEP 5: Feedback Loop — Re-run MoA refine with reflection feedback
   324→   324→      // ═══════════════════════════════════════════
   325→   325→      if (!reflectionResult.passed && lastReflectionFeedback && config.moaConfig?.feedbackInjection) {
   326→   326→        const feedbackStart = Date.now();
   327→   327→        this.emitEvent('feedback_injected', execution.id, agentId, {
   328→   328→          feedbackLength: lastReflectionFeedback.length,
   329→   329→          reason: 'Reflection did not pass — injecting feedback into MoA refine',
   330→   330→        });
   331→   331→
   332→   332→        // Run a targeted MoA refine with the reflection feedback
   333→   333→        const feedbackRefineResult = await this.executeFeedbackRefine(
   334→   334→          currentOutput,
   335→   335→          lastReflectionFeedback,
   336→   336→          config,
   337→   337→          selectedProviders
   338→   338→        );
   339→   339→
   340→   340→        if (feedbackRefineResult) {
   341→   341→          currentOutput = feedbackRefineResult;
   342→   342→          addTimeline('dag_level', 'Feedback-refine loop applied', Date.now() - feedbackStart);
   343→   343→        }
   344→   344→      }
   345→   345→
   346→   346→      // ═══════════════════════════════════════════
   347→   347→      // STEP 6: Record execution result for CostOptimizer RL
   348→   348→      // ═══════════════════════════════════════════
   349→   349→      try {
   350→   350→        for (const provider of selectedProviders) {
   351→   351→          await this.costOptimizer.recordExecutionResult(
   352→   352→            provider,
   353→   353→            config.id,
   354→   354→            reflectionResult.confidence,
   355→   355→            execution.totalCost / selectedProviders.length,
   356→   356→            execution.totalTokensUsed > 0
   357→   357→              ? Math.round(execution.totalTokensUsed / selectedProviders.length)
   358→   358→              : 0,
   359→   359→            0, // rough latency per provider
   360→   360→            agentId,
   361→   361→          );
   362→   362→        }
   363→   363→      } catch (err) {
   364→   364→        console.error('[SuperAgent] Failed to record execution result for CostOptimizer:', err);
   365→   365→      }
   366→   366→
   367→   367→      // ═══════════════════════════════════════════
   368→   368→      // STEP 7: Finalize & Cache
   369→   369→      // ═══════════════════════════════════════════
   370→   370→      execution.output = currentOutput;
   371→   371→      execution.status = reflectionResult.passed ? 'completed' : 'completed_with_warnings';
   372→   372→      execution.completedAt = new Date();
   373→   373→
   374→   374→      // Cache the result (DistributedCache with cross-instance invalidation)
   375→   375→      await this.cache.setByAgentPrompt(agentId, prompt, { output: currentOutput });
   376→   376→
   377→   377→      addTimeline('finalization', `Status: ${execution.status}, cost: $${execution.totalCost.toFixed(4)}, tokens: ${execution.totalTokensUsed}`);
   378→   378→
   379→   379→      this.emitEvent('execution_complete', execution.id, agentId, {
   380→   380→        status: execution.status,
   381→   381→        tokensUsed: execution.totalTokensUsed,
   382→   382→        cost: execution.totalCost,
   383→   383→        iterations: execution.iterations,
   384→   384→        confidence: reflectionResult.confidence,
   385→   385→        timelineEntries: timeline.length,
   386→   386→      });
   387→   387→
   388→   388→      return execution;
   389→   389→    } catch (error) {
   390→   390→      execution.status = 'failed';
   391→   391→      execution.error = error instanceof Error ? error.message : 'Unknown error';
   392→   392→      execution.completedAt = new Date();
   393→   393→
   394→   394→      addTimeline('finalization', `FAILED: ${execution.error}`);
   395→   395→
   396→   396→      this.emitEvent('execution_error', execution.id, agentId, {
   397→   397→        error: execution.error,
   398→   398→      });
   399→   399→
   400→   400→      return execution;
   401→   401→    } finally {
   402→   402→      this.activeExecutions.delete(execution.id);
   403→   403→    }
   404→   404→  }
   405→   405→
   406→   406→  /**
   407→   407→   * Cancel a running execution
   408→   408→   */
   409→   409→  cancel(executionId: string): boolean {
   410→   410→    const controller = this.activeExecutions.get(executionId);
   411→   411→    if (controller) {
   412→   412→      controller.abort();
   413→   413→      return true;
   414→   414→    }
   415→   415→    return false;
   416→   416→  }
   417→   417→
   418→   418→  // ═══════════════════════════════════════════════════════
   419→   419→  // Node Execution (with MoA cascading + feedback injection)
   420→   420→  // ═══════════════════════════════════════════════════════
   421→   421→
   422→   422→  private async executeNode(
   423→   423→    node: DAGNode,
   424→   424→    input: string,
   425→   425→    config: AgentConfig,
   426→   426→    providers: LLMId[],
   427→   427→    executionId: string,
   428→   428→    agentId: AgentId,
   429→   429→    reflectionFeedback?: string
   430→   430→  ): Promise<{ output: string; tokensUsed: number; cost: number }> {
   431→   431→    this.emitEvent('dag_node_start', executionId, agentId, {
   432→   432→      nodeId: node.id,
   433→   433→      nodeLabel: node.label,
   434→   434→      agentRole: node.agentRole,
   435→   435→    });
   436→   436→
   437→   437→    // Forward MoA events with execution context
   438→   438→    const moaHandler = (event: MoAEvent) => {
   439→   439→      this.emitEvent('moa_round', executionId, agentId, {
   440→   440→        nodeId: event.nodeId || node.id,
   441→   441→        round: event.round,
   442→   442→        provider: event.provider,
   443→   443→        data: event.data,
   444→   444→      });
   445→   445→    };
   446→   446→
   447→   447→    this.codeGenerator.on('moa-event', moaHandler);
   448→   448→    try {
   449→   449→      const result = await this.codeGenerator.generateWithFeedback(
   450→   450→        node, input, config, providers, reflectionFeedback
   451→   451→      );
   452→   452→      return {
   453→   453→        output: result.output,
   454→   454→        tokensUsed: result.tokensUsed,
   455→   455→        cost: result.cost,
   456→   456→      };
   457→   457→    } finally {
   458→   458→      this.codeGenerator.off('moa-event', moaHandler);
   459→   459→    }
   460→   460→  }

   462→   462→  // ═══════════════════════════════════════════════════════
   463→   463→  // Feedback Refine Loop — MoA refine with reflection feedback
   464→   464→  // ═══════════════════════════════════════════════════════
   465→   465→
   466→   466→  /**
   467→   467→   * When the reflection loop doesn't pass after all auto-fix attempts,
   468→   468→   * inject the reflection feedback into a targeted MoA refine round.
   469→   469→   * This creates a synthetic DAG node that runs only the refine round
   470→   470→   * with the accumulated feedback.
   471→   471→   */
   472→   472→  private async executeFeedbackRefine(
   473→   473→    currentOutput: string,
   474→   474→    reflectionFeedback: string,
   475→   475→    config: AgentConfig,
   476→   476→    providers: LLMId[]
   477→   477→  ): Promise<string | null> {
   478→   478→    try {
   479→   479→      // Create a synthetic node for the feedback refine
   480→   480→      const syntheticNode: DAGNode = {
   481→   481→        id: 'feedback-refine',
   482→   482→        label: 'Feedback Refine',
   483→   483→        level: 999,
   484→   484→        type: 'process',
   485→   485→        agentRole: 'refiner',
   486→   486→      };
   487→   487→
   488→   488→      const result = await this.codeGenerator.generateWithFeedback(
   489→   489→        syntheticNode,
   490→   490→        currentOutput,
   491→   491→        config,
   492→   492→        providers,
   493→   493→        reflectionFeedback
   494→   494→      );
   495→   495→
   496→   496→      return result.output || null;
   497→   497→    } catch (error) {
   498→   498→      console.error('[SuperAgent] Feedback refine loop failed:', error);
   499→   499→      return null;
   500→   500→    }
   501→   501→  }

   503→   503→  // ═══════════════════════════════════════════════════════
   504→   504→  // Smart Output Merging per Level
   505→   505→  // ═══════════════════════════════════════════════════════
   506→   506→
   507→   507→  private mergeLevelOutputs(
   508→   508→    nodes: DAGNode[],
   509→   509→    results: Array<{ node: DAGNode; result: { output: string; tokensUsed: number; cost: number } }>
   510→   510→  ): string {
   511→   511→    if (results.length === 0) return '';
   512→   512→    if (results.length === 1) return results[0].result.output;
   513→   513→
   514→   514→    // For multiple nodes at same level, merge contextually
   515→   515→    const merged = results.map(({ node, result }) => {
   516→   516→      const header = `=== ${node.label.toUpperCase()} ===`;
   517→   517→      return `${header}\n${result.output}`;
   518→   518→    }).join('\n\n');
   519→   519→
   520→   520→    return merged;
   521→   521→  }

   523→   523→  // ═══════════════════════════════════════════════════════
   524→   524→  // 3-Level Auto-Fix Engine (delegated to ReflectionAgent)
   525→   525→  // ═══════════════════════════════════════════════════════
   526→   526→
   527→   527→  private async executeAutoFix(
   528→   528→    output: string,
   529→   529→    feedback: string,
   530→   530→    providers: LLMId[],
   531→   531→    attempt: number,
   532→   532→    maxAttempts: number
   533→   533→  ): Promise<AutoFixAttempt> {
   534→   534→    const fixAttempt: AutoFixAttempt = {
   535→   535→      level: 'pattern',
   536→   536→      attempt,
   537→   537→      maxAttempts,
   538→   538→      input: output,
   539→   539→      output: '',
   540→   540→      success: false,
   541→   541→    };
   542→   542→
   543→   543→    // Level 1: Pattern-based fixes (fastest, cheapest)
   544→   544→    const patternFixed = this.reflectionAgent.patternFix(output, feedback);
   545→   545→    if (patternFixed) {
   546→   546→      fixAttempt.level = 'pattern';
   547→   547→      fixAttempt.success = true;
   548→   548→      fixAttempt.output = patternFixed;
   549→   549→      return fixAttempt;
   550→   550→    }
   551→   551→
   552→   552→    // Level 2: AST-based structural fixes (moderate cost)
   553→   553→    const astFixed = await this.reflectionAgent.astFix(output, feedback);
   554→   554→    if (astFixed) {
   555→   555→      fixAttempt.level = 'ast';
   556→   556→      fixAttempt.success = true;
   557→   557→      fixAttempt.output = astFixed;
   558→   558→      return fixAttempt;
   559→   559→    }
   560→   560→
   561→   561→    // Level 3: LLM-based fixes (most expensive, most capable)
   562→   562→    const llmFixed = await this.reflectionAgent.llmFix(output, feedback, providers);
   563→   563→    if (llmFixed) {
   564→   564→      fixAttempt.level = 'llm';
   565→   565→      fixAttempt.success = true;
   566→   566→      fixAttempt.output = llmFixed;
   567→   567→      return fixAttempt;
   568→   568→    }
   569→   569→
   570→   570→    return fixAttempt;
   571→   571→  }

   574→   574→  // Project Context Building (Phase 5)
   575→   575→  // ═══════════════════════════════════════════════════════
   576→   576→
   577→   577→  /**
   578→   578→   * Build project context from project files for injection into the prompt.
   579→   579→   * Uses the ContextGraph, DependencyResolver, TokenBudgetManager, and
   580→   580→   * ContextCompressionService pipeline to build, scope, budget, and compress
   581→   581→   * project context for LLM consumption.
   582→   582→   *
   583→   583→   * @param prompt - The user's prompt (used to detect file references)
   584→   584→   * @param maxTokens - Maximum tokens to allocate for context
   585→   585→   * @returns Compressed context string or null if no project files are referenced
   586→   586→   */
   587→   587→  buildProjectContext(prompt: string, maxTokens: number): string | null {
   588→   588→    // Detect if the prompt references project files
   589→   589→    // Look for common patterns: file paths, "in my project", "my code", etc.
   590→   590→    const fileReferencePatterns = [
   591→   591→      /(?:src|lib|app|components?|services?|utils?|hooks?|pages?|api|routes?)\/[\w./-]+\.\w+/gi,
   592→   592→      /file[s]?\s*(?:at|in|named|called)\s+["']?[\w./-]+\.\w+/gi,
   593→   593→      /(?:my|the|this)\s+(?:project|code|app|codebase)/i,
   594→   594→      /(?:update|modify|change|fix|refactor)\s+(?:the\s+)?(?:file|module|component)/i,
   595→   595→    ];
   596→   596→
   597→   597→    const hasFileReferences = fileReferencePatterns.some(p => p.test(prompt));
   598→   598→    if (!hasFileReferences) return null;
   599→   599→
   600→   600→    // Reset the context graph for this execution
   601→   601→    const graph = new ContextGraph();
   602→   602→
   603→   603→    // If there are files in the existing context graph, use them
   604→   604→    // Otherwise, we have no files to build context from — return null
   605→   605→    const graphStats = this.contextGraph.getStats();
   606→   606→    if (graphStats.totalFiles === 0) return null;
   607→   607→
   608→   608→    // Use the existing context graph to build context
   609→   609→    const allFiles = this.contextGraph.getTopologicalOrder();
   610→   610→
   611→   611→    // Build a priority map: entry point files get higher priority
   612→   612→    const priorities = new Map<string, number>();
   613→   613→    for (const filePath of allFiles) {
   614→   614→      // Entry points (main.ts, index.ts, app.ts) get highest priority
   615→   615→      if (/^(main|index|app|server)\.(ts|tsx|js|jsx)$/i.test(filePath.split('/').pop() || '')) {
   616→   616→        priorities.set(filePath, 1.0);
   617→   617→      }
   618→   618→      // Route/controller files get high priority
   619→   619→      else if (/(route|controller|handler|api)/i.test(filePath)) {
   620→   620→        priorities.set(filePath, 0.8);
   621→   621→      }
   622→   622→      // Service files get medium priority
   623→   623→      else if (/(service|repository|store)/i.test(filePath)) {
   624→   624→        priorities.set(filePath, 0.6);
   625→   625→      }
   626→   626→      // Utility/helper files get lower priority
   627→   627→      else if (/(util|helper|hook|type|interface|constant)/i.test(filePath)) {
   628→   628→        priorities.set(filePath, 0.4);
   629→   629→      }
   630→   630→      // Everything else gets default priority
   631→   631→      else {
   632→   632→        priorities.set(filePath, 0.5);
   633→   633→      }
   634→   634→    }
   635→   635→
   636→   636→    // Allocate token budget across files
   637→   637→    const budget = this.tokenBudgetManager.allocateBudget(maxTokens, allFiles, priorities);
   638→   638→
   639→   639→    // Build files map for compression
   640→   640→    const filesMap = new Map<string, string>();
   641→   641→    for (const filePath of allFiles) {
   642→   642→      const deps = this.contextGraph.getDirectDependencies(filePath);
   643→   643→      // We don't have direct access to file content from the graph,
   644→   644→      // so we create a placeholder structure for context
   645→   645→      filesMap.set(filePath, `// File: ${filePath}\n// Dependencies: ${deps.join(', ') || 'none'}`);
   646→   646→    }
   647→   647→
   648→   648→    // Build the token budget object for compression
   649→   649→    const perFile: Record<string, number> = {};
   650→   650→    for (const [file, tokens] of budget) {
   651→   651→      perFile[file] = tokens;
   652→   652→    }
   653→   653→
   654→   654→    const tokenBudget: TokenBudget = {
   655→   655→      total: maxTokens,
   656→   656→      used: 0,
   657→   657→      perFile,
   658→   658→    };
   659→   659→
   660→   660→    // Compress context within budget
   661→   661→    const compressed = this.contextCompressionService.compressForContext(filesMap, tokenBudget);
   662→   662→
   663→   663→    // Build the final context string
   664→   664→    const contextParts: string[] = [];
   665→   665→    contextParts.push('=== PROJECT CONTEXT ===');
   666→   666→    contextParts.push(`Total files: ${allFiles.length}`);
   667→   667→    contextParts.push(`Token budget: ${maxTokens}`);
   668→   668→    contextParts.push('');
   669→   669→
   670→   670→    for (const [filePath, content] of compressed) {
   671→   671→      contextParts.push(`--- ${filePath} ---`);
   672→   672→      contextParts.push(content);
   673→   673→      contextParts.push('');
   674→   674→    }
   675→   675→
   676→   676→    return contextParts.join('\n');
   677→   677→  }

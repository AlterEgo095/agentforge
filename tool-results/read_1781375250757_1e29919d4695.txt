   114→  // ─── Core Pipeline ─────────────────────────────────────
   115→
   116→  protected async generate(
   117→    prompt: string,
   118→    context: Record<string, unknown>,
   119→    sessionId: string,
   120→  ): Promise<DocOutput> {
   121→    // Phase 1: Parse document type from prompt
   122→    const docType = this.parseDocumentType(prompt);
   123→    const topic = this.extractTopic(prompt);
   124→
   125→    this.emitEvent('doc_outline', sessionId, { docType, topic });
   126→
   127→    // Phase 2: Build document outline with proper hierarchy
   128→    const outline = this.buildOutline(docType, topic);
   129→
   130→    // Phase 3: Generate content for each section
   131→    // (structured data generation — actual LLM content is provided by the orchestrator)
   132→    const sections = this.generateSectionContent(outline, prompt, context);
   133→
   134→    this.emitEvent('doc_content', sessionId, {
   135→      sectionCount: sections.length,
   136→      totalWordEstimate: this.estimateSectionWords(sections),
   137→    });
   138→
   139→    // Phase 4: Build table of contents from section hierarchy
   140→    const tableOfContents = this.buildToc(sections);
   141→
   142→    // Phase 5: Add references if sources are mentioned or available
   143→    const references = this.extractReferences(prompt, context);
   144→
   145→    // Phase 6: Compute metadata
   146→    const totalWordCount = this.computeTotalWordCount(sections);
   147→    const readingTimeMinutes = Math.max(1, Math.round(totalWordCount / 238)); // avg 238 wpm
   148→
   149→    this.emitEvent('doc_formatting', sessionId, {
   150→      wordCount: totalWordCount,
   151→      readingTimeMinutes,
   152→      format: 'document',
   153→    });
   154→
   155→    return {
   156→      title: this.generateTitle(docType, topic),
   157→      subtitle: this.generateSubtitle(docType, topic),
   158→      author: context.author as string | undefined,
   159→      sections,
   160→      tableOfContents,
   161→      references,
   162→      metadata: {
   163→        wordCount: totalWordCount,
   164→        readingTimeMinutes,
   165→        format: docType,
   166→        language: 'en',
   167→        sourceCount: references.length,
   168→      },
   169→    };
   170→  }
   171→
   172→  // ─── Validation ────────────────────────────────────────
   173→
   174→  protected async validateOutput(output: DocOutput, _prompt: string): Promise<QualityReport> {
   175→    const dimensions: Record<string, number> = {};
   176→    const suggestions: string[] = [];
   177→
   178→    // 1. Content Depth (0–1)
   179→    dimensions['Content Depth'] = this.scoreContentDepth(output, suggestions);
   180→
   181→    // 2. Structure & Organization (0–1)
   182→    dimensions['Structure & Organization'] = this.scoreStructure(output, suggestions);
   183→
   184→    // 3. Formatting & Typography (0–1)
   185→    dimensions['Formatting & Typography'] = this.scoreFormatting(output, suggestions);
   186→
   187→    // 4. Accuracy & Citations (0–1)
   188→    dimensions['Accuracy & Citations'] = this.scoreCitations(output, suggestions);
   189→
   190→    // 5. Clarity & Readability (0–1)
   191→    dimensions['Clarity & Readability'] = this.scoreReadability(output, suggestions);
   192→
   193→    const overall = this.computeWeightedScore(dimensions);
   194→    const passed = this.isQualityPassing(overall);
   195→
   196→    return { overall, dimensions, passed, suggestions };
   197→  }

   201→  protected async renderOutput(output: DocOutput, format?: string): Promise<unknown> {
   202→    switch (format) {
   203→      case 'markdown':
   204→        return this.renderMarkdown(output);
   205→
   206→      case 'html':
   207→        return this.renderHtml(output);
   208→
   209→      case 'pdf':

   212→        return {
   213→          type: 'pdf',
   214→          title: output.title,
   215→          subtitle: output.subtitle,

   217→          sectionCount: output.sections.length,

   220→

   227→  // ─── Helper: Document Type Parsing ─────────────────────
   228→
   229→  /**
   230→   * Detect the document type from the user prompt by matching against
   231→   * known keyword sets. Falls back to "report".
   232→   */
   233→  parseDocumentType(prompt: string): string {
   234→    const lower = prompt.toLowerCase();
   235→    let bestMatch = 'report';
   236→    let bestScore = 0;
   237→
   238→    for (const [docType, keywords] of Object.entries(DOCUMENT_TYPE_KEYWORDS)) {
   239→      let score = 0;
   240→      for (const keyword of keywords) {
   241→        if (lower.includes(keyword)) {
   242→          score += keyword.length; // longer keyword matches are stronger signals
   243→        }
   244→      }
   245→      if (score > bestScore) {
   246→        bestScore = score;
   247→        bestMatch = docType;
   248→      }
   249→    }
   250→
   251→    return bestMatch;
   252→  }
   253→
   254→  /**
   255→   * Extract the core topic / subject from the prompt.
   256→   * Uses a simple heuristic: strips common command prefixes and returns
   257→   * the remainder trimmed to a reasonable length.
   258→   */
   259→  private extractTopic(prompt: string): string {
   260→    const prefixes = [
   261→      /^(?:write|create|generate|draft|produce|build)\s+(?:a\s+)?(?:report|proposal|manual|whitepaper|memo|letter|article|specification|spec|plan|document)\s+(?:about|on|for|regarding)\s+/i,
   262→      /^(?:write|create|generate|draft|produce|build)\s+(?:a\s+)?/i,
   263→    ];
   264→
   265→    let topic = prompt.trim();
   266→    for (const prefix of prefixes) {
   267→      topic = topic.replace(prefix, '');
   268→    }
   269→
   270→    // Truncate to first sentence or 120 characters whichever is shorter
   271→    const firstSentence = topic.match(/^[^.!?\n]+/);
   272→    if (firstSentence) {
   273→      topic = firstSentence[0].trim();
   274→    }
   275→
   276→    return topic.length > 120 ? topic.slice(0, 117) + '...' : topic;
   277→  }

   279→  // ─── Helper: Outline Building ──────────────────────────
   280→
   281→  /**
   282→   * Create a section hierarchy from a document-type template.
   283→   * Each section gets a unique ID, placeholder content, and recursive
   284→   * children based on the template.
   285→   */
   286→  buildOutline(docType: string, topic: string): DocSection[] {
   287→    const templates = DOCUMENT_TYPE_TEMPLATES[docType] ?? DOCUMENT_TYPE_TEMPLATES['report'];
   288→    return templates.map((tpl, index) => this.instantiateSection(tpl, topic, [index]));
   289→  }
   290→
   291→  private instantiateSection(tpl: SectionTemplate, topic: string, path: number[]): DocSection {
   292→    const id = `sec-${path.join('-')}`;
   293→    const section: DocSection = {
   294→      id,
   295→      heading: tpl.heading,
   296→      level: tpl.level,
   297→      content: this.generatePlaceholderContent(tpl.heading, tpl.level, topic),
   298→    };
   299→
   300→    if (tpl.children && tpl.children.length > 0) {
   301→      section.children = tpl.children.map((child, childIndex) =>
   302→        this.instantiateSection(child, topic, [...path, childIndex]),
   303→      );
   304→    }
   305→
   306→    return section;
   307→  }
   308→
   309→  /**
   310→   * Generate structured placeholder content for a section.
   311→   * Real LLM-generated content would be injected by the orchestrator;
   312→   * this provides a well-structured scaffold with appropriate depth cues.
   313→   */
   314→  private generatePlaceholderContent(heading: string, level: 1 | 2 | 3 | 4, topic: string): string {
   315→    const minWords = level <= 2 ? 150 : 75;
   316→    const base = `This section covers ${heading.toLowerCase()} in the context of ${topic}. `;
   317→
   318→    const elaborations = [
   319→      `The following provides a comprehensive overview of ${heading.toLowerCase()}, including key considerations, analysis, and actionable insights relevant to ${topic}.`,
   320→      `It is essential to understand the foundational aspects presented here, as they form the basis for subsequent sections and recommendations throughout this document.`,
   321→      `The discussion that follows draws upon established practices and contemporary research, ensuring both rigor and relevance in addressing the core themes of ${topic}.`,
   322→      `Stakeholders should consider the interdependencies between this area and the broader strategic context, as outlined in the executive summary and conclusion sections.`,
   323→      `Further detail and supporting evidence can be found in the referenced sources appended to this document.`,
   324→    ];
   325→
   326→    let content = base + elaborations.join(' ');
   327→
   328→    // Ensure minimum word count for level 1–2 sections
   329→    while (this.countWords(content) < minWords) {
   330→      content += ` Additional context and elaboration on ${heading.toLowerCase()} is provided to ensure comprehensive coverage of this important topic. The analysis considers multiple perspectives and draws on both qualitative and quantitative evidence where available.`;
   331→    }
   332→
   333→    return content;
   334→  }

   336→  // ─── Helper: Section Content Generation ────────────────
   337→
   338→  /**
   339→   * Enrich sections with contextual content, tables, images, or code
   340→   * blocks based on the document type and prompt signals.
   341→   */
   342→  private generateSectionContent(
   343→    outline: DocSection[],
   344→    prompt: string,
   345→    context: Record<string, unknown>,
   346→  ): DocSection[] {
   347→    return outline.map((section) => {
   348→      const enriched = { ...section };
   349→
   350→      // If the orchestrator has provided previous output feedback, incorporate it
   351→      if (context.previousOutput && typeof context.previousOutput === 'object') {
   352→        const prev = context.previousOutput as Record<string, unknown>;
   353→        if (prev.sections && Array.isArray(prev.sections)) {
   354→          const matching = (prev.sections as DocSection[]).find((s) => s.heading === section.heading);
   355→          if (matching && matching.content) {
   356→            enriched.content = matching.content;
   357→          }
   358→        }
   359→      }
   360→
   361→      // Recursively enrich children
   362→      if (enriched.children && enriched.children.length > 0) {
   363→        enriched.children = this.generateSectionContent(enriched.children, prompt, context);
   364→      }
   365→
   366→      // Add structural elements based on heading semantics
   367→      enriched.tables = this.suggestTables(enriched.heading, prompt);
   368→      enriched.images = this.suggestImages(enriched.heading, prompt);
   369→      enriched.codeBlocks = this.suggestCodeBlocks(enriched.heading, prompt);
   370→
   371→      return enriched;
   372→    });
   373→  }

   430→
   431→  /**
   432→   * Generate a recursive Table of Contents from the section hierarchy.
   433→   */
   434→  buildToc(sections: DocSection[]): TocEntry[] {
   435→    return sections.map((section) => {
   436→      const entry: TocEntry = {
   437→        id: section.id,
   438→        title: section.heading,
   439→        level: section.level,
   440→      };
   441→
   442→      if (section.children && section.children.length > 0) {
   443→        entry.children = this.buildToc(section.children);
   444→      }
   445→
   446→      return entry;
   447→    });
   448→  }

   450→  // ─── Helper: References ────────────────────────────────
   451→
   452→  /**
   453→   * Extract references from prompt and context. In production this
   454→   * would be powered by the RECHERCHE agent; here we provide a
   455→   * scaffold that detects explicit source mentions.
   456→   */
   457→  private extractReferences(prompt: string, context: Record<string, unknown>): DocReference[] {
   458→    const references: DocReference[] = [];
   459→    const sourcePattern = /(?:source|ref|cite|according to|referenced?)(?:\s+(?:to|from))?\s+["']([^"']+)["']/gi;
   460→    let match: RegExpExecArray | null;
   461→
   462→    while ((match = sourcePattern.exec(prompt)) !== null) {
   463→      references.push({
   464→        id: `ref-${references.length + 1}`,
   465→        type: 'other',
   466→        title: match[1],
   467→      });
   468→    }
   469→
   470→    // Also include any references passed through context (e.g. from RECHERCHE agent)
   471→    if (context.references && Array.isArray(context.references)) {
   472→      for (const ref of context.references as DocReference[]) {
   473→        if (ref.id && ref.title) {
   474→          references.push(ref);
   475→        }
   476→      }
   477→    }
   478→
   479→    return references;
   480→  }

   511→  // ─── Helper: Word Counting ─────────────────────────────
   512→
   513→  /**
   514→   * Count words in a text string. Splits on whitespace and filters empties.
   515→   */
   516→  countWords(text: string): number {
   517→    return text.split(/\s+/).filter((w) => w.length > 0).length;
   518→  }
   519→
   520→  private computeTotalWordCount(sections: DocSection[]): number {
   521→    let total = 0;
   522→    for (const section of sections) {
   523→      total += this.countWords(section.content);
   524→      if (section.tables) {
   525→        for (const table of section.tables) {
   526→          total += table.headers.join(' ').split(/\s+/).length;
   527→          for (const row of table.rows) {
   528→            total += row.join(' ').split(/\s+/).length;
   529→          }
   530→        }
   531→      }
   532→      if (section.codeBlocks) {
   533→        for (const block of section.codeBlocks) {
   534→          total += this.countWords(block.code);
   535→        }
   536→      }
   537→      if (section.children) {
   538→        total += this.computeTotalWordCount(section.children);
   539→      }
   540→    }
   541→    return total;
   542→  }
   543→
   544→  private estimateSectionWords(sections: DocSection[]): number {
   545→    return this.computeTotalWordCount(sections);
   546→  }

   548→  // ─── Quality Scoring Helpers ───────────────────────────
   549→
   550→  private scoreContentDepth(output: DocOutput, suggestions: string[]): number {
   551→    let totalSections = 0;
   552→    let deepSections = 0;
   553→
   554→    const walk = (sections: DocSection[]): void => {
   555→      for (const section of sections) {
   556→        totalSections++;
   557→        const wordCount = this.countWords(section.content);
   558→        // Level 1–2 sections should have at least 150 words; levels 3–4 at least 75
   559→        const minimum = section.level <= 2 ? 150 : 75;
   560→        if (wordCount >= minimum) {
   561→          deepSections++;
   562→        } else {
   563→          suggestions.push(
   564→            `Section "${section.heading}" has only ${wordCount} words (minimum: ${minimum}). Consider expanding the content.`,
   565→          );
   566→        }
   567→        if (section.children) walk(section.children);
   568→      }
   569→    };
   570→
   571→    walk(output.sections);
   572→
   573→    if (totalSections === 0) return 0;
   574→    return deepSections / totalSections;
   575→  }

   577→  private scoreStructure(output: DocOutput, suggestions: string[]): number {
   578→    let score = 1.0;
   579→
   580→    // Must have at least 2 top-level sections
   581→    if (output.sections.length < 2) {
   582→      score -= 0.3;
   583→      suggestions.push('Document should have at least 2 top-level sections.');
   584→    }
   585→
   586→    // Validate heading hierarchy — each section's level should not skip (e.g. 1 → 3)
   587→    let prevLevel = 0;
   588→    const checkHierarchy = (sections: DocSection[]): void => {
   589→      for (const section of sections) {
   590→        if (prevLevel > 0 && section.level > prevLevel + 1) {
   591→          score -= 0.1;
   592→          suggestions.push(
   593→            `Heading hierarchy skip detected: level ${prevLevel} → ${section.level} at "${section.heading}".`,
   594→          );
   595→        }
   596→        prevLevel = section.level;
   597→        if (section.children) checkHierarchy(section.children);
   598→      }
   599→    };
   600→    checkHierarchy(output.sections);
   601→
   602→    // TOC should match sections
   603→    if (output.tableOfContents.length !== output.sections.length) {
   604→      score -= 0.15;
   605→      suggestions.push('Table of contents entry count does not match top-level sections.');
   606→    }
   607→
   608→    return Math.max(0, Math.min(1, score));
   609→  }

   611→  private scoreFormatting(output: DocOutput, suggestions: string[]): number {
   612→    let score = 1.0;
   613→    const headingSet = new Set<string>();
   614→    let duplicateHeadings = 0;
   615→
   616→    const walk = (sections: DocSection[]): void => {
   617→      for (const section of sections) {
   618→        if (headingSet.has(section.heading.toLowerCase())) {
   619→          duplicateHeadings++;
   620→        }
   621→        headingSet.add(section.heading.toLowerCase());
   622→
   623→        // Check for empty content
   624→        if (!section.content || section.content.trim().length === 0) {
   625→          score -= 0.15;
   626→          suggestions.push(`Section "${section.heading}" has empty content.`);
   627→        }
   628→
   629→        // Validate tables have rows
   630→        if (section.tables) {
   631→          for (let i = 0; i < section.tables.length; i++) {
   632→            const table = section.tables[i];
   633→            if (table.rows.length === 0) {
   634→              score -= 0.05;
   635→              suggestions.push(`Table ${i + 1} in "${section.heading}" has no rows.`);
   636→            }
   637→            if (table.headers.length === 0) {
   638→              score -= 0.05;
   639→              suggestions.push(`Table ${i + 1} in "${section.heading}" has no headers.`);
   640→            }
   641→          }
   642→        }
   643→
   644→        // Validate code blocks have code
   645→        if (section.codeBlocks) {
   646→          for (let i = 0; i < section.codeBlocks.length; i++) {
   647→            if (!section.codeBlocks[i].code || section.codeBlocks[i].code.trim().length === 0) {
   648→              score -= 0.05;
   649→              suggestions.push(`Code block ${i + 1} in "${section.heading}" is empty.`);
   650→            }
   651→          }
   652→        }
   653→
   654→        if (section.children) walk(section.children);
   655→      }
   656→    };
   657→
   658→    walk(output.sections);
   659→
   660→    if (duplicateHeadings > 0) {
   661→      score -= 0.1 * duplicateHeadings;
   662→      suggestions.push(`${duplicateHeadings} duplicate heading(s) detected. Use unique headings for each section.`);
   663→    }
   664→
   665→    return Math.max(0, Math.min(1, score));
   666→  }

   668→  private scoreCitations(output: DocOutput, suggestions: string[]): number {
   669→    const refCount = output.references.length;
   670→
   671→    if (refCount === 0) {
   672→      // Not necessarily a failure for shorter document types
   673→      const docTypesRequiringRefs = ['whitepaper', 'report', 'specification', 'proposal'];
   674→      if (docTypesRequiringRefs.includes(output.metadata.format)) {
   675→        suggestions.push(`${output.metadata.format} documents typically require references. Consider adding sources.`);
   676→        return 0.4;
   677→      }
   678→      return 0.8; // acceptable for memos, letters, etc.
   679→    }
   680→
   681→    // Check reference completeness
   682→    let incomplete = 0;
   683→    for (const ref of output.references) {
   684→      if (!ref.title || ref.title.trim().length === 0) incomplete++;
   685→    }
   686→
   687→    if (incomplete > 0) {
   688→      suggestions.push(`${incomplete} reference(s) have incomplete information.`);
   689→    }
   690→
   691→    // Score based on completeness ratio
   692→    const completeness = 1 - incomplete / refCount;
   693→    // Bonus for having multiple well-formed references
   694→    const countBonus = Math.min(refCount / 5, 1) * 0.2;
   695→    return Math.min(1, completeness * 0.8 + countBonus);
   696→  }

   698→  private scoreReadability(output: DocOutput, suggestions: string[]): number {
   699→    let totalWords = 0;
   700→    let totalSentences = 0;
   701→    let totalSyllables = 0;
   702→
   703→    const walk = (sections: DocSection[]): void => {
   704→      for (const section of sections) {
   705→        const text = section.content;
   706→        totalWords += this.countWords(text);
   707→        totalSentences += this.countSentences(text);
   708→        totalSyllables += this.estimateSyllables(text);
   709→
   710→        if (section.children) walk(section.children);
   711→      }
   712→    };
   713→
   714→    walk(output.sections);
   715→
   716→    if (totalSentences === 0 || totalWords === 0) return 0;
   717→
   718→    // Flesch-Kincaid Grade Level (approximate)
   719→    const fkGrade = 0.39 * (totalWords / totalSentences) + 11.8 * (totalSyllables / totalWords) - 15.59;
   720→
   721→    // Ideal grade level for professional documents: 8–12
   722→    // Score 1.0 for grade 10, decreasing as we move away
   723→    const idealGrade = 10;
   724→    const deviation = Math.abs(fkGrade - idealGrade);
   725→    const score = Math.max(0, 1 - deviation / 10);
   726→
   727→    if (fkGrade > 14) {
   728→      suggestions.push('Document may be too complex. Consider simplifying sentence structures.');
   729→    } else if (fkGrade < 6) {
   730→      suggestions.push('Document readability is very low grade-level. Consider if the tone is appropriate for the audience.');
   731→    }
   732→
   733→    return Math.min(1, Math.max(0, score));
   734→  }

   736→  private countSentences(text: string): number {
   737→    const sentences = text.split(/[.!?]+/).filter((s) => s.trim().length > 0);
   738→    return Math.max(1, sentences.length);
   739→  }

   741→  private estimateSyllables(text: string): number {
   742→    // Simple heuristic: count vowel groups per word
   743→    const words = text.split(/\s+/).filter((w) => w.length > 0);
   744→    let total = 0;
   745→    for (const word of words) {
   746→      const vowelGroups = word.toLowerCase().replace(/[^a-z]/g, '').match(/[aeiouy]+/g);
   747→      total += vowelGroups ? Math.max(1, vowelGroups.length) : 1;
   748→    }
   749→    return total;
   750→  }

   754→  renderMarkdown(output: DocOutput): string {
   755→    const lines: string[] = [];
   756→
   757→    // Title block
   758→    lines.push(`# ${output.title}`);
   759→    if (output.subtitle) lines.push(`## ${output.subtitle}`);
   760→    if (output.author) lines.push(`*By ${output.author}*`);
   761→    lines.push('');
   762→
   763→    // Table of Contents
   764→    lines.push('## Table of Contents');
   765→    lines.push('');
   766→    for (const entry of output.tableOfContents) {
   767→      lines.push(this.renderTocEntryMarkdown(entry));
   768→    }
   769→    lines.push('');
   770→
   771→    // Sections
   772→    for (const section of output.sections) {
   773→      lines.push(this.renderSectionMarkdown(section));
   774→    }
   775→
   776→    // References
   777→    if (output.references.length > 0) {
   778→      lines.push('## References');
   779→      lines.push('');
   780→      for (const ref of output.references) {
   781→        lines.push(this.renderReferenceMarkdown(ref));
   782→      }
   783→      lines.push('');
   784→    }
   785→
   786→    // Footer metadata
   787→    lines.push('---');
   788→    lines.push(`*Word count: ${output.metadata.wordCount} | Reading time: ~${output.metadata.readingTimeMinutes} min*`);
   789→
   790→    return lines.join('\n');
   791→  }

   793→  private renderTocEntryMarkdown(entry: TocEntry, indent = 0): string {
   794→    const prefix = '  '.repeat(indent) + '- ';
   795→    let line = `${prefix}[${entry.title}](#${entry.id})`;
   796→    const childLines = entry.children
   797→      ? entry.children.map((child: TocEntry) => this.renderTocEntryMarkdown(child, indent + 1)).join('\n')
   798→      : '';
   799→    return childLines ? line + '\n' + childLines : line;
   800→  }

   802→  private renderSectionMarkdown(section: DocSection): string {
   803→    const lines: string[] = [];
   804→    const headingPrefix = '#'.repeat(section.level + 1); // level 1 → ##, level 2 → ###, etc.
   805→    lines.push(`${headingPrefix} ${section.heading}`);
   806→    lines.push('');
   807→    lines.push(section.content);
   808→    lines.push('');
   809→
   810→    // Tables
   811→    if (section.tables) {
   812→      for (const table of section.tables) {
   813→        lines.push(this.renderTableMarkdown(table));
   814→        lines.push('');
   815→      }
   816→    }
   817→
   818→    // Images
   819→    if (section.images) {
   820→      for (const image of section.images) {
   821→        const alt = image.alt || 'image';
   822→        const md = image.url ? `![${alt}](${image.url})` : `![${alt}]`;
   823→        lines.push(md);
   824→        if (image.caption) lines.push(`*${image.caption}*`);
   825→        lines.push('');
   826→      }
   827→    }
   828→
   829→    // Code blocks
   830→    if (section.codeBlocks) {
   831→      for (const block of section.codeBlocks) {
   832→        lines.push('```' + block.language);
   833→        lines.push(block.code);
   834→        lines.push('```');
   835→        if (block.caption) lines.push(`*${block.caption}*`);
   836→        lines.push('');
   837→      }
   838→    }
   839→
   840→    // Children
   841→    if (section.children) {
   842→      for (const child of section.children) {
   843→        lines.push(this.renderSectionMarkdown(child));
   844→      }
   845→    }
   846→
   847→    return lines.join('\n');
   848→  }

   850→  private renderTableMarkdown(table: DocTable): string {
   851→    const lines: string[] = [];
   852→    lines.push('| ' + table.headers.join(' | ') + ' |');
   853→    lines.push('| ' + table.headers.map(() => '---').join(' | ') + ' |');
   854→    for (const row of table.rows) {
   855→      lines.push('| ' + row.join(' | ') + ' |');
   856→    }
   857→    if (table.caption) {
   858→      lines.push(`*${table.caption}*`);
   859→    }
   860→    return lines.join('\n');
   861→  }

   863→  private renderReferenceMarkdown(ref: DocReference): string {
   864→    const parts: string[] = [];
   865→    if (ref.authors && ref.authors.length > 0) parts.push(ref.authors.join(', '));
   866→    if (ref.year) parts.push(`(${ref.year})`);

   868→    if (ref.url) parts.push(`[Link](${ref.url})`);
   869→    return `- [${ref.id}] ${parts.join(' ')}`;

   938→  private renderTocHtml(entries: TocEntry[], depth: number): string {
   939→    const indent = '    '.repeat(depth);
   940→    const lines: string[] = [];
   941→    lines.push(`${indent}<ul>`);
   942→    for (const entry of entries) {
   943→      lines.push(`${indent}  <li><a href="#${entry.id}">${this.escapeHtml(entry.title)}</a>`);
   944→      if (entry.children && entry.children.length > 0) {
   945→        lines.push(this.renderTocHtml(entry.children, depth + 1));
   946→      }
   947→      lines.push(`${indent}  </li>`);
   948→    }
   949→    lines.push(`${indent}</ul>`);
   950→    return lines.join('\n');
   951→  }

   953→  private renderSectionHtml(section: DocSection): string {
   954→    const tag = `h${section.level + 1}`; // level 1 → h2, level 2 → h3, etc.
   955→    const lines: string[] = [];
   956→    lines.push(`  <section id="${section.id}">`);
   957→    lines.push(`    <${tag}>${this.escapeHtml(section.heading)}</${tag}>`);
   958→    lines.push(`    <p>${this.escapeHtml(section.content)}</p>`);
   959→
   960→    // Tables
   961→    if (section.tables) {
   962→      for (const table of section.tables) {
   963→        lines.push(this.renderTableHtml(table));
   964→      }
   965→    }
   966→
   967→    // Images
   968→    if (section.images) {
   969→      for (const image of section.images) {
   970→        const src = image.url ?? '';
   971→        const alt = this.escapeHtml(image.alt || 'image');
   972→        lines.push(`    <figure>`);
   973→        lines.push(`      <img src="${src}" alt="${alt}" />`);
   974→        if (image.caption) {
   975→          lines.push(`      <figcaption class="caption">${this.escapeHtml(image.caption)}</figcaption>`);
   976→        }
   977→        lines.push(`    </figure>`);
   978→      }
   979→    }
   980→
   981→    // Code blocks
   982→    if (section.codeBlocks) {
   983→      for (const block of section.codeBlocks) {
   984→        lines.push(`    <pre><code class="language-${this.escapeHtml(block.language)}">${this.escapeHtml(block.code)}</code></pre>`);
   985→        if (block.caption) {
   986→          lines.push(`    <p class="caption">${this.escapeHtml(block.caption)}</p>`);
   987→        }
   988→      }
   989→    }
   990→
   991→    // Children
   992→    if (section.children) {
   993→      for (const child of section.children) {
   994→        lines.push(this.renderSectionHtml(child));
   995→      }
   996→    }
   997→
   998→    lines.push('  </section>');
   999→    return lines.join('\n');
  1000→  }

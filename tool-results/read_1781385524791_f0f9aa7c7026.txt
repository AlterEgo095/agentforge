    11→// ============================================================
    12→// Token Refresh Interceptor
    13→// ============================================================
    14→
    15→let refreshPromise: Promise<boolean> | null = null;
    16→
    17→async function refreshTokenIfNeeded(): Promise<boolean> {
    18→  if (refreshPromise) return refreshPromise;
    19→
    20→  refreshPromise = (async () => {
    21→    try {
    22→      const { refreshToken } = useAuthStore.getState();
    23→      if (!refreshToken) return false;
    24→
    25→      // Use a direct fetch (not the request() function) to avoid infinite loop
    26→      const response = await fetch(`${API_BASE}/auth/refresh`, {
    27→        method: 'POST',
    28→        headers: { 'Content-Type': 'application/json' },
    29→        body: JSON.stringify({ refreshToken }),
    30→      });
    31→
    32→      if (!response.ok) return false;
    33→
    34→      const result = await response.json();
    35→      useAuthStore.getState().setTokens(result.data.accessToken, result.data.refreshToken);
    36→      return true;
    37→    } catch {
    38→      return false;
    39→    } finally {
    40→      refreshPromise = null;
    41→    }
    42→  })();
    43→
    44→  return refreshPromise;
    45→}
    46→
    47→async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
    48→  const token = useAuthStore.getState().token;
    49→
    50→  const headers: Record<string, string> = {
    51→    'Content-Type': 'application/json',
    52→    ...(options.headers as Record<string, string>),
    53→  };
    54→
    55→  if (token) {
    56→    headers['Authorization'] = `Bearer ${token}`;
    57→  }
    58→
    59→  const response = await fetch(`${API_BASE}${path}`, {
    60→    ...options,
    61→    headers,
    62→  });
    63→
    64→  if (!response.ok) {
    65→    if (response.status === 401) {
    66→      const refreshed = await refreshTokenIfNeeded();
    67→      if (refreshed) {
    68→        // Retry with new token
    69→        const newToken = useAuthStore.getState().token;
    70→        headers['Authorization'] = `Bearer ${newToken}`;
    71→        const retryResponse = await fetch(`${API_BASE}${path}`, { ...options, headers });
    72→        if (retryResponse.ok) return retryResponse.json();
    73→      }
    74→      // Refresh failed — logout
    75→      useAuthStore.getState().logout();
    76→      window.location.href = '/login';
    77→      throw new Error('Session expired');
    78→    }
    79→
    80→    const error = await response.json().catch(() => ({ error: 'Request failed' }));
    81→    throw new Error(error.error || `HTTP ${response.status}`);
    82→  }
    83→
    84→  return response.json();
    85→}

   487→// SSE Streaming
   488→// ============================================================

   490→/**
   491→ * Execute an agent with SSE streaming.
   492→ * Returns an EventSource-like experience with structured events.
   493→ */
   494→export function executeAgentStream(
   495→  agentId: string,
   496→  prompt: string,
   497→  projectId?: string,
   498→  onEvent?: (event: ExecutionEvent) => void,
   499→  onError?: (error: Error) => void,
   500→  onComplete?: (result: any) => void
   501→): { cancel: () => void } {
   502→  const token = useAuthStore.getState().token;
   503→  const controller = new AbortController();
   504→
   505→  fetch(`${API_BASE}/agents/${agentId}/execute`, {
   506→    method: 'POST',
   507→    headers: {
   508→      'Content-Type': 'application/json',
   509→      'Authorization': `Bearer ${token}`,
   510→    },
   511→    body: JSON.stringify({ prompt, projectId }),
   512→    signal: controller.signal,
   513→  })
   514→    .then(async (response) => {
   515→      if (!response.ok) {
   516→        throw new Error(`HTTP ${response.status}`);
   517→      }
   518→
   519→      const reader = response.body?.getReader();
   520→      if (!reader) throw new Error('No response body');
   521→
   522→      const decoder = new TextDecoder();
   523→      let buffer = '';
   524→
   525→      while (true) {
   526→        const { done, value } = await reader.read();
   527→        if (done) break;
   528→
   529→        buffer += decoder.decode(value, { stream: true });
   530→
   531→        const lines = buffer.split('\n');
   532→        buffer = lines.pop() || '';
   533→
   534→        for (const line of lines) {
   535→          if (line.startsWith('data: ')) {
   536→            try {
   537→              const event = JSON.parse(line.substring(6)) as ExecutionEvent;
   538→              onEvent?.(event);
   539→
   540→              if (event.type === 'execution_complete') {
   541→                onComplete?.(event.data);
   542→              }
   543→            } catch {
   544→              // Skip malformed events
   545→            }
   546→          }
   547→        }
   548→      }
   549→    })
   550→    .catch((error) => {
   551→      if (error.name !== 'AbortError') {
   552→        onError?.(error);
   553→      }
   554→    });
   555→
   556→  return {
   557→    cancel: () => controller.abort(),
   558→  };
   559→}

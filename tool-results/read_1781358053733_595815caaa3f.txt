    35→interface AuthRateLimitOptions {
    36→  windowMs: number;
    37→  maxRequests: number;
    38→  keyPrefix: string;
    39→}
    40→
    41→async function checkAuthRateLimit(key: string, options: AuthRateLimitOptions): Promise<{ allowed: boolean; remaining: number; retryAfterMs: number }> {
    42→  if (!isRedisAvailable()) {
    43→    return { allowed: true, remaining: options.maxRequests, retryAfterMs: 0 };
    44→  }
    45→
    46→  const redis = getPrimaryRedis();
    47→  const now = Date.now();
    48→  const windowKey = `authrl:${options.keyPrefix}:${key}:${Math.floor(now / options.windowMs)}`;
    49→
    50→  try {
    51→    const count = await redis.incr(windowKey);
    52→    if (count === 1) {
    53→      await redis.pexpire(windowKey, options.windowMs);
    54→    }
    55→    const remaining = Math.max(0, options.maxRequests - count);
    56→    const retryAfterMs = Math.ceil(now / options.windowMs) * options.windowMs - now;
    57→    return { allowed: count <= options.maxRequests, remaining, retryAfterMs };
    58→  } catch {
    59→    return { allowed: true, remaining: options.maxRequests, retryAfterMs: 0 };
    60→  }
    61→}

    63→/** Strict rate limiter: 5 requests per 15 minutes per IP — for login, register, MFA login-verify */
    64→const strictAuthRateLimiter = async (c: any, next: any) => {
    65→  const ip = getClientIp(c);
    66→  const result = await checkAuthRateLimit(ip, { windowMs: 15 * 60 * 1000, maxRequests: 5, keyPrefix: 'strict' });
    67→
    68→  c.header('X-RateLimit-Remaining', result.remaining.toString());
    69→
    70→  if (!result.allowed) {
    71→    return c.json({
    72→      success: false,
    73→      error: 'Too many attempts. Please try again later.',
    74→      retryAfter: Math.ceil(result.retryAfterMs / 1000),
    75→    }, 429);
    76→  }
    77→
    78→  await next();
    79→};

    99→/** Account lockout: check if email/IP is locked after 5 failed login attempts */
   100→const ACCOUNT_LOCKOUT_MAX_ATTEMPTS = 5;
   101→const ACCOUNT_LOCKOUT_WINDOW_MS = 15 * 60 * 1000; // 15 minutes
   102→
   103→async function checkAccountLockout(identifier: string): Promise<boolean> {
   104→  if (!isRedisAvailable()) return false;
   105→  const key = `authrl:lockout:${identifier}`;
   106→  try {
   107→    const redis = getPrimaryRedis();
   108→    const count = await redis.get(key);
   109→    return count !== null && parseInt(count) >= ACCOUNT_LOCKOUT_MAX_ATTEMPTS;
   110→  } catch {
   111→    return false; // Redis unavailable — don't block
   112→  }
   113→}
   114→
   115→async function recordFailedLoginAttempt(identifier: string): Promise<void> {
   116→  if (!isRedisAvailable()) return;
   117→  const key = `authrl:lockout:${identifier}`;
   118→  try {
   119→    const redis = getPrimaryRedis();
   120→    const count = await redis.incr(key);
   121→    if (count === 1) {
   122→      await redis.pexpire(key, ACCOUNT_LOCKOUT_WINDOW_MS);
   123→    }
   124→  } catch {
   125→    // Redis unavailable — skip
   126→  }
   127→}

   129→async function resetFailedLoginAttempts(identifier: string): Promise<void> {
   130→  if (!isRedisAvailable()) return;
   131→  const key = `authrl:lockout:${identifier}`;
   132→  try {
   133→    const redis = getPrimaryRedis();
   134→    await redis.del(key);
   135→  } catch {
   136→    // Redis unavailable — skip
   137→  }
   138→}

   177→/**
   178→ * Generate an access token with full OWASP-recommended claims.
   179→ * P2.1.4: iss, aud, sub, iat, exp, jti
   180→ */
   181→async function generateAccessToken(userId: string, tier: string): Promise<string> {
   182→  const now = Math.floor(Date.now() / 1000);
   183→  const payload: JWTPayload = {
   184→    sub: userId,
   185→    tier,
   186→    iss: env.JWT_ISSUER,
   187→    aud: env.JWT_AUDIENCE,
   188→    iat: now,
   189→    exp: now + 900, // 15 minutes
   190→    jti: uuidv4(),
   191→  };
   192→
   193→  return sign(payload, env.JWT_SECRET);
   194→}

   196→/**
   197→ * Generate a refresh token JWT with full claims.
   198→ * P2.1.3: Includes jti for rotation tracking.
   199→ */
   200→async function generateRefreshTokenJWT(userId: string, tokenHash: string): Promise<string> {
   201→  const now = Math.floor(Date.now() / 1000);
   202→  const payload: RefreshJWTPayload = {
   203→    sub: userId,
   204→    type: 'refresh',
   205→    tid: tokenHash.substring(0, 16),
   206→    iss: env.JWT_ISSUER,
   207→    aud: env.JWT_AUDIENCE,
   208→    iat: now,
   209→    exp: now + 604800, // 7 days
   210→    jti: uuidv4(),
   211→  };
   212→
   213→  return sign(payload, env.JWT_SECRET);
   214→}

   216→/**
   217→ * Generate an MFA pending token with full claims.
   218→ */
   219→async function generateMFAPendingToken(userId: string, sessionId: string): Promise<string> {
   220→  const now = Math.floor(Date.now() / 1000);
   221→  const payload: MFAPendingJWTPayload = {
   222→    sub: userId,
   223→    type: 'mfa_pending',
   224→    sessionId,
   225→    iss: env.JWT_ISSUER,
   226→    aud: env.JWT_AUDIENCE,
   227→    iat: now,
   228→    exp: now + 300, // 5 minutes
   229→    jti: uuidv4(),
   230→  };
   231→
   232→  return sign(payload, env.JWT_SECRET);
   233→}

   235→/**
   236→ * Create and persist a refresh token in the database.
   237→ * Returns the raw JWT and the hash for storage.
   238→ * P2.1.3: Stores tokenHash in DB for specific token binding.
   239→ */
   240→async function createRefreshToken(userId: string): Promise<{ jwt: string; tokenHash: string; hashFull: string }> {
   241→  const refreshTokenValue = uuidv4() + '-' + uuidv4();
   242→  const hashFull = createHash('sha256').update(refreshTokenValue).digest('hex');
   243→  const tokenHash = hashFull.substring(0, 64); // Use full hash for specific token binding
   244→
   245→  await db.insert(refreshTokens).values({
   246→    id: uuidv4(),
   247→    userId,
   248→    tokenHash, // P2.1.3: Full hash for specific token identification
   249→    expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
   250→  });
   251→
   252→  const jwt = await generateRefreshTokenJWT(userId, tokenHash);
   253→
   254→  return { jwt, tokenHash, hashFull };
   255→}

   379→auth.post('/login', strictAuthRateLimiter, zValidator('json', LoginSchema), async (c) => {
   380→  const data = c.req.valid('json');
   381→  const ipAddress = getClientIp(c);
   382→  const userAgent = getUserAgent(c);
   383→
   384→  const doLogin = async () => {
   385→    // ── Account Lockout: Check if email or IP is locked ──
   386→    const emailLocked = await checkAccountLockout(data.email);
   387→    const ipLocked = await checkAccountLockout(ipAddress);
   388→    if (emailLocked || ipLocked) {
   389→      return { result: c.json({ success: false, error: 'Account temporarily locked due to too many failed attempts. Try again in 15 minutes.' }, 423), mfaRequired: false };
   390→    }
   391→
   392→    // ── Threat Detection: Check if IP is blocked ──
   393→    if (threatDetectionService.isBlocked(ipAddress)) {
   394→      await auditTrailService.record({
   395→        action: 'auth.login',
   396→        actorId: 'unknown',
   397→        actorType: 'user',
   398→        resourceType: 'user',
   399→        resourceId: 'unknown',
   400→        details: { reason: 'IP blocked by threat detection' },
   401→        ipAddress,
   402→        userAgent,
   403→        success: false,
   404→        riskLevel: 'high',
   405→      });
   406→
   407→      return { result: c.json({ success: false, error: 'Access temporarily restricted. Try again later.' }, 429), mfaRequired: false };
   408→    }
   409→
   410→    const userRows = await db.select().from(users).where(eq(users.email, data.email)).limit(1);
   411→    const user = userRows[0];
   412→
   413→    if (!user || !(await bcrypt.compare(data.password, user.passwordHash))) {
   414→      // ── Account Lockout: Record failed login attempt ──
   415→      await recordFailedLoginAttempt(data.email);
   416→      await recordFailedLoginAttempt(ipAddress);
   417→
   418→      // ── Threat Detection: Record failed login ──
   419→      const threatAlert = threatDetectionService.recordFailedLogin(
   420→        user?.id || data.email,
   421→        ipAddress
   422→      );
   423→
   424→      // Audit: failed login
   425→      await auditTrailService.record({
   426→        action: 'auth.login',
   427→        actorId: user?.id || 'unknown',
   428→        actorType: 'user',
   429→        resourceType: 'user',
   430→        resourceId: user?.id || 'unknown',
   431→        details: { reason: 'Invalid credentials', threatAlert: threatAlert?.id },
   432→        ipAddress,
   433→        userAgent,
   434→        success: false,
   435→        riskLevel: threatAlert ? 'high' : 'medium',
   436→      });
   437→
   438→      return { result: c.json({ success: false, error: 'Invalid email or password' }, 401), mfaRequired: false };
   439→    }
   440→
   441→    // ── Threat Detection: Record successful login (resets user counters) ──
   442→    threatDetectionService.recordSuccessfulLogin(user.id, ipAddress);
   443→
   444→    // ── Account Lockout: Reset failed login attempts on success ──
   445→    await resetFailedLoginAttempts(data.email);
   446→    await resetFailedLoginAttempts(ipAddress);
   447→
   448→    // ── MFA Check: If MFA is enabled, return mfaRequired flag ──
   449→    if (user.mfaEnabled) {
   450→      // Create a pending session (not yet MFA-verified)
   451→      const fingerprint = sessionManager.generateFingerprint(userAgent, ipAddress);
   452→      const session = await sessionManager.createSession(user.id, fingerprint, ipAddress, userAgent);
   453→
   454→      // Issue a temporary MFA token with full claims (P2.1.4)
   455→      const mfaToken = await generateMFAPendingToken(user.id, session.id);
   456→
   457→      // Audit: MFA required
   458→      await auditTrailService.record({
   459→        action: 'auth.login',
   460→        actorId: user.id,
   461→        actorType: 'user',
   462→        resourceType: 'user',
   463→        resourceId: user.id,
   464→        details: { reason: 'MFA verification required' },
   465→        ipAddress,
   466→        userAgent,
   467→        success: false,
   468→        riskLevel: 'low',
   469→      });
   470→
   471→      return {
   472→        result: c.json({
   473→          success: true,
   474→          mfaRequired: true,
   475→          data: {
   476→            mfaToken,
   477→            sessionId: session.id,
   478→            message: 'MFA verification required',
   479→          },
   480→        }),
   481→        userId: user.id,
   482→        mfaRequired: true,
   483→      };
   484→    }
   485→
   486→    // ── No MFA: Issue full tokens ──
   487→    const accessToken = await generateAccessToken(user.id, user.tier);
   488→    const { jwt: refreshTokenJwt } = await createRefreshToken(user.id);
   489→
   490→    // Create session
   491→    const fingerprint = sessionManager.generateFingerprint(userAgent, ipAddress);
   492→    const session = await sessionManager.createSession(user.id, fingerprint, ipAddress, userAgent);
   493→    await sessionManager.verifyMfa(session.id); // Mark as MFA-verified since no MFA required
   494→
   495→    // Audit: successful login
   496→    await auditTrailService.record({
   497→      action: 'auth.login',
   498→      actorId: user.id,
   499→      actorType: 'user',
   500→      resourceType: 'user',
   501→      resourceId: user.id,
   502→      details: { mfaUsed: false },
   503→      ipAddress,
   504→      userAgent,
   505→      success: true,
   506→      riskLevel: 'low',
   507→    });
   508→
   509→    return {
   510→      result: c.json({
   511→        success: true,
   512→        mfaRequired: false,
   513→        data: {
   514→          user: { id: user.id, email: user.email, name: user.name, tier: user.tier, mfaEnabled: false },
   515→          tokens: { accessToken, refreshToken: refreshTokenJwt, expiresIn: 900 },
   516→        },
   517→      }),
   518→      userId: user.id,
   519→      mfaRequired: false,
   520→    };
   521→  };
   522→
   523→  if (tracingService.isEnabled()) {
   524→    return await tracingService.withSpan(
   525→      { name: 'auth.login', attributes: { 'auth.method': 'password' } },
   526→      async (span) => {
   527→        const { result, userId, mfaRequired } = await doLogin();
   528→        if (userId) {
   529→          span.setAttribute('auth.user_id', userId);
   530→          span.setAttribute('auth.mfa_required', mfaRequired ?? false);
   531→          span.setAttribute('auth.success', true);
   532→        }
   533→        return result;
   534→      },
   535→    );
   536→  } else {
   537→    const { result } = await doLogin();
   538→    return result;
   539→  }
   540→});

   543→// MFA Login Verification (second step for MFA-enabled users)
   544→// ═══════════════════════════════════════════════════════════
   545→
   546→auth.post('/mfa/login-verify', strictAuthRateLimiter, zValidator('json', MFALoginVerifySchema), async (c) => {
   547→  const data = c.req.valid('json');
   548→  const ipAddress = getClientIp(c);
   549→  const userAgent = getUserAgent(c);
   550→
   551→  const doMfaVerify = async () => {
   552→    // Validate session
   553→    const sessionResult = await sessionManager.validateSession(data.sessionId);
   554→    if (!sessionResult.valid || !sessionResult.session) {
   555→      return { result: c.json({ success: false, error: 'Invalid or expired session' }, 401), userId: undefined as string | undefined, method: data.code ? 'totp' : 'backup' as string };
   556→    }
   557→
   558→    const session = sessionResult.session;
   559→    const userId = session.userId;
   560→
   561→    // Fetch user with MFA data
   562→    const userRows = await db.select().from(users).where(eq(users.id, userId)).limit(1);
   563→    const user = userRows[0];
   564→
   565→    if (!user || !user.mfaEnabled || !user.mfaSecret) {
   566→      return { result: c.json({ success: false, error: 'MFA not enabled for this account' }, 400), userId, method: data.code ? 'totp' : 'backup' };
   567→    }
   568→
   569→    // Verify TOTP code or backup code — P2.1.1: Use encrypted verification
   570→    let verified = false;
   571→
   572→    if (data.code) {
   573→      // P2.1.1: Decrypt secret and verify TOTP code
   574→      verified = mfaService.verifyWithEncryptedSecret(user.mfaSecret, data.code);
   575→    } else if (data.backupCode) {
   576→      // P2.1.1: Verify against encrypted backup codes
   577→      const backupCodes = (user.mfaBackupCodes as string[]) || [];
   578→      const result = mfaService.verifyEncryptedBackupCode(backupCodes, data.backupCode);

  1050→

  1058→    try {
  1059→      const payload = await verify(refreshToken, env.JWT_SECRET, 'HS256') as {
  1060→        sub: string;
  1061→        type: string;
  1062→        tid: string;
  1063→        jti?: string;
  1064→        iss?: string;
  1065→        aud?: string;
  1066→      };
  1067→
  1068→      // P2.1.4: Validate token type
  1069→      if (payload.type !== 'refresh') {
  1070→        return { result: c.json({ success: false, error: 'Invalid token type' }, 401), userId: payload.sub };
  1071→      }
  1072→
  1073→      // P2.1.4: Validate issuer and audience
  1074→      if (payload.iss && payload.iss !== env.JWT_ISSUER) {
  1075→        return { result: c.json({ success: false, error: 'Invalid token issuer' }, 401), userId: payload.sub };
  1076→      }
  1077→      if (payload.aud && payload.aud !== env.JWT_AUDIENCE) {
  1078→        return { result: c.json({ success: false, error: 'Invalid token audience' }, 401), userId: payload.sub };
  1079→      }
  1080→

  1082→      if (payload.jti && await jwtBlacklist.isBlacklisted(payload.jti)) {
  1083→        return { result: c.json({ success: false, error: 'Token has been revoked' }, 401), userId: payload.sub };
  1084→      }
  1085→

  1087→      const tokenHashPrefix = payload.tid;
  1088→      const storedTokens = await db.select().from(refreshTokens).where(
  1089→        and(
  1090→          eq(refreshTokens.userId, payload.sub),
  1091→          eq(refreshTokens.isRevoked, false),
  1092→        )
  1093→      ).limit(10); // Get all active tokens for this user
  1094→
  1095→      // P2.1.3: Find the specific token by matching the hash prefix
  1096→      const matchingToken = storedTokens.find(t => t.tokenHash.startsWith(tokenHashPrefix) || t.tokenHash === tokenHashPrefix);
  1097→
  1098→      if (!matchingToken) {
  1099→        // P2.1.3: Token not found — potential reuse attack
  1100→        // Revoke ALL tokens for this user as a security measure
  1101→        await db.update(refreshTokens)
  1102→          .set({ isRevoked: true })
  1103→          .where(eq(refreshTokens.userId, payload.sub));
  1104→

  1106→        await auditTrailService.record({
  1107→          action: 'auth.token_refresh',
  1108→          actorId: payload.sub,
  1109→          actorType: 'user',
  1110→          resourceType: 'user',
  1111→          resourceId: payload.sub,
  1112→          details: { reason: 'refresh_token_reuse_detected', allTokensRevoked: true },
  1113→          ipAddress: getClientIp(c),
  1114→          userAgent: getUserAgent(c),
  1115→          success: false,
  1116→          riskLevel: 'high',
  1117→        });
  1118→
  1119→        return { result: c.json({ success: false, error: 'Token reuse detected. All sessions terminated.' }, 401), userId: payload.sub };
  1120→      }
  1121→
  1122→      // P2.1.3: ROTATE — Revoke the used refresh token immediately
  1123→      await db.update(refreshTokens)
  1124→        .set({ isRevoked: true })
  1125→        .where(eq(refreshTokens.id, matchingToken.id));
  1126→
  1127→      // P2.1.4: Blacklist the old JWT's jti to prevent replay
  1128→      if (payload.jti) {
  1129→        await jwtBlacklist.blacklist(payload.jti, 7 * 24 * 60 * 60); // 7 days
  1130→      }
  1131→
  1132→      // Fetch the user to get the current tier
  1133→      const userRows = await db.select().from(users).where(eq(users.id, payload.sub)).limit(1);
  1134→      const user = userRows[0];
  1135→
  1136→      if (!user) {
  1137→        return { result: c.json({ success: false, error: 'User not found' }, 401), userId: payload.sub };
  1138→      }
  1139→
  1140→      // P2.1.3: Issue NEW refresh token (rotation)
  1141→      const accessToken = await generateAccessToken(user.id, user.tier);
  1142→      const { jwt: newRefreshTokenJwt } = await createRefreshToken(user.id);
  1143→
  1144→      // Audit: successful token refresh with rotation
  1145→      await auditTrailService.record({
  1146→        action: 'auth.token_refresh',
  1147→        actorId: user.id,
  1148→        actorType: 'user',
  1149→        resourceType: 'user',
  1150→        resourceId: user.id,
  1151→        details: { rotated: true, oldTokenRevoked: matchingToken.id },
  1152→        ipAddress: getClientIp(c),
  1153→        userAgent: getUserAgent(c),
  1154→        success: true,
  1155→        riskLevel: 'low',
  1156→      });
  1157→
  1158→      return {
  1159→        result: c.json({
  1160→          success: true,
  1161→          data: { accessToken, refreshToken: newRefreshTokenJwt, expiresIn: 900 },
  1162→        }),
  1163→        userId: user.id,
  1164→      };
  1165→    } catch {
  1166→      return { result: c.json({ success: false, error: 'Invalid refresh token' }, 401), userId: undefined };
  1167→    }
  1168→  };
  1169→
  1170→  if (tracingService.isEnabled()) {
  1171→    return await tracingService.withSpan(
  1172→      { name: 'auth.token_refresh', attributes: { 'auth.operation': 'refresh' } },
  1173→      async (span) => {
  1174→        const { result, userId } = await doRefresh();
  1175→        if (userId) {
  1176→          span.setAttribute('auth.user_id', userId);
  1177→        }
  1178→        return result;
  1179→      },
  1180→    );
  1181→  } else {
  1182→    const { result } = await doRefresh();
  1183→    return result;
  1184→  }
  1185→});

  1187→// ═══════════════════════════════════════════════════════════
  1188→// Logout — P2.1.3: Revoke specific token (not all tokens)
  1189→// P2.1.4: Blacklist the JWT's jti
  1190→// ═══════════════════════════════════════════════════════════
  1191→
  1192→auth.post('/logout', async (c) => {
  1193→  const { refreshToken } = await c.req.json();
  1194→  if (!refreshToken) {
  1195→    return c.json({ success: true, message: 'Logged out' });
  1196→  }
  1197→
  1198→  const doLogout = async () => {
  1199→    try {
  1200→      const payload = await verify(refreshToken, env.JWT_SECRET, 'HS256') as { sub: string; tid: string; jti?: string };
  1201→
  1202→      // P2.1.3: Revoke the SPECIFIC refresh token (not all user tokens)
  1203→      const tokenHashPrefix = payload.tid;
  1204→      const storedTokens = await db.select().from(refreshTokens).where(
  1205→        and(
  1206→          eq(refreshTokens.userId, payload.sub),
  1207→          eq(refreshTokens.isRevoked, false),
  1208→        )
  1209→      ).limit(10);
  1210→
  1211→      const matchingToken = storedTokens.find(t => t.tokenHash.startsWith(tokenHashPrefix) || t.tokenHash === tokenHashPrefix);
  1212→
  1213→      if (matchingToken) {
  1214→        // Revoke only this specific token
  1215→        await db.update(refreshTokens)
  1216→          .set({ isRevoked: true })
  1217→          .where(eq(refreshTokens.id, matchingToken.id));
  1218→      }
  1219→
  1220→      // P2.1.4: Blacklist the JWT's jti
  1221→      if (payload.jti) {
  1222→        await jwtBlacklist.blacklist(payload.jti, 7 * 24 * 60 * 60); // 7 days
  1223→      }
  1224→
  1225→      // Audit: logout
  1226→      await auditTrailService.record({
  1227→        action: 'auth.logout',
  1228→        actorId: payload.sub,
  1229→        actorType: 'user',
  1230→        resourceType: 'user',
  1231→        resourceId: payload.sub,
  1232→        details: { specificTokenRevoked: !!matchingToken },
  1233→        ipAddress: getClientIp(c),
  1234→        userAgent: getUserAgent(c),
  1235→        success: true,
  1236→        riskLevel: 'low',
  1237→      });
  1238→
  1239→      return { result: c.json({ success: true, message: 'Logged out successfully' }), userId: payload.sub };
  1240→    } catch {
  1241→      return { result: c.json({ success: true, message: 'Logged out' }), userId: undefined };
  1242→    }
  1243→  };
  1244→
  1245→  if (tracingService.isEnabled()) {
  1246→    return await tracingService.withSpan(
  1247→      { name: 'auth.logout', attributes: { 'auth.operation': 'logout' } },
  1248→      async (span) => {

  1250→        if (userId) {
  1251→          span.setAttribute('auth.user_id', userId);
  1252→        }

  1254→      },
  1255→    );
  1256→  } else {
  1257→    const { result } = await doLogout();

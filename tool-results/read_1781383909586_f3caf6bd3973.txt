    17→    17→async function refreshTokenIfNeeded(): Promise<boolean> {
    18→    18→  if (refreshPromise) return refreshPromise;
    19→    19→
    20→    20→  refreshPromise = (async () => {
    21→    21→    try {
    22→    22→      const { refreshToken } = useAuthStore.getState();
    23→    23→      if (!refreshToken) return false;
    24→    24→
    25→    25→      // Use a direct fetch (not the request() function) to avoid infinite loop
    26→    26→      const response = await fetch(`${API_BASE}/auth/refresh`, {
    27→    27→        method: 'POST',
    28→    28→        headers: { 'Content-Type': 'application/json' },
    29→    29→        body: JSON.stringify({ refreshToken }),
    30→    30→      });
    31→    31→
    32→    32→      if (!response.ok) return false;
    33→    33→
    34→    34→      const result = await response.json();
    35→    35→      useAuthStore.getState().setTokens(result.data.accessToken, result.data.refreshToken);
    36→    36→      return true;
    37→    37→    } catch {
    38→    38→      return false;
    39→    39→    } finally {
    40→    40→      refreshPromise = null;
    41→    41→    }
    42→    42→  })();
    43→    43→
    44→    44→  return refreshPromise;
    45→    45→}

    47→    47→async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
    48→    48→  const token = useAuthStore.getState().token;
    49→    49→
    50→    50→  const headers: Record<string, string> = {
    51→    51→    'Content-Type': 'application/json',
    52→    52→    ...(options.headers as Record<string, string>),
    53→    53→  };
    54→    54→
    55→    55→  if (token) {
    56→    56→    headers['Authorization'] = `Bearer ${token}`;
    57→    57→  }
    58→    58→
    59→    59→  const response = await fetch(`${API_BASE}${path}`, {
    60→    60→    ...options,
    61→    61→    headers,
    62→    62→  });
    63→    63→
    64→    64→  if (!response.ok) {
    65→    65→    if (response.status === 401) {
    66→    66→      const refreshed = await refreshTokenIfNeeded();
    67→    67→      if (refreshed) {
    68→    68→        // Retry with new token
    69→    69→        const newToken = useAuthStore.getState().token;
    70→    70→        headers['Authorization'] = `Bearer ${newToken}`;
    71→    71→        const retryResponse = await fetch(`${API_BASE}${path}`, { ...options, headers });
    72→    72→        if (retryResponse.ok) return retryResponse.json();
    73→    73→      }
    74→    74→      // Refresh failed — logout
    75→    75→      useAuthStore.getState().logout();
    76→    76→      window.location.href = '/login';
    77→    77→      throw new Error('Session expired');
    78→    78→    }
    79→    79→
    80→    80→    const error = await response.json().catch(() => ({ error: 'Request failed' }));
    81→    81→    throw new Error(error.error || `HTTP ${response.status}`);
    82→    82→  }
    83→    83→
    84→    84→  return response.json();
    85→    85→}

   487→   487→// SSE Streaming
   488→   488→// ============================================================
   489→   489→
   490→   490→/**
   491→   491→ * Execute an agent with SSE streaming.
   492→   492→ * Returns an EventSource-like experience with structured events.
   493→   493→ */
   494→   494→export function executeAgentStream(
   495→   495→  agentId: string,
   496→   496→  prompt: string,
   497→   497→  projectId?: string,
   498→   498→  onEvent?: (event: ExecutionEvent) => void,
   499→   499→  onError?: (error: Error) => void,
   500→   500→  onComplete?: (result: any) => void
   501→   501→): { cancel: () => void } {
   502→   502→  const token = useAuthStore.getState().token;
   503→   503→  const controller = new AbortController();
   504→   504→
   505→   505→  fetch(`${API_BASE}/agents/${agentId}/execute`, {
   506→   506→    method: 'POST',
   507→   507→    headers: {
   508→   508→      'Content-Type': 'application/json',
   509→   509→      'Authorization': `Bearer ${token}`,
   510→   510→    },
   511→   511→    body: JSON.stringify({ prompt, projectId }),
   512→   512→    signal: controller.signal,
   513→   513→  })
   514→   514→    .then(async (response) => {
   515→   515→      if (!response.ok) {
   516→   516→        throw new Error(`HTTP ${response.status}`);
   517→   517→      }
   518→   518→
   519→   519→      const reader = response.body?.getReader();
   520→   520→      if (!reader) throw new Error('No response body');
   521→   521→
   522→   522→      const decoder = new TextDecoder();
   523→   523→      let buffer = '';
   524→   524→
   525→   525→      while (true) {
   526→   526→        const { done, value } = await reader.read();
   527→   527→        if (done) break;
   528→   528→
   529→   529→        buffer += decoder.decode(value, { stream: true });
   530→   530→
   531→   531→        const lines = buffer.split('\n');
   532→   532→        buffer = lines.pop() || '';
   533→   533→
   534→   534→        for (const line of lines) {
   535→   535→          if (line.startsWith('data: ')) {
   536→   536→            try {
   537→   537→              const event = JSON.parse(line.substring(6)) as ExecutionEvent;
   538→   538→              onEvent?.(event);
   539→   539→
   540→   540→              if (event.type === 'execution_complete') {
   541→   541→                onComplete?.(event.data);
   542→   542→              }
   543→   543→            } catch {
   544→   544→              // Skip malformed events
   545→   545→            }
   546→   546→          }
   547→   547→        }
   548→   548→      }
   549→   549→    })
   550→   550→    .catch((error) => {
   551→   551→      if (error.name !== 'AbortError') {
   552→   552→        onError?.(error);
   553→   553→      }
   554→   554→    });
   555→   555→
   556→   556→  return {
   557→   557→    cancel: () => controller.abort(),
   558→   558→  };
   559→   559→}

    44→    44→// GET /admin/users - List all users
    45→    45→admin.get('/users', requirePermission({ resource: 'users', action: 'read' }), async (c) => {
    46→    46→  try {
    47→    47→    const limit = parseInt(c.req.query('limit') || '50');
    48→    48→    const offset = parseInt(c.req.query('offset') || '0');
    49→    49→
    50→    50→    const userList = await db
    51→    51→      .select({
    52→    52→        id: users.id,
    53→    53→        email: users.email,
    54→    54→        name: users.name,
    55→    55→        tier: users.tier,
    56→    56→        mfaEnabled: users.mfaEnabled,
    57→    57→        createdAt: users.createdAt,
    58→    58→        updatedAt: users.updatedAt,
    59→    59→      })
    60→    60→      .from(users)
    61→    61→      .orderBy(desc(users.createdAt))
    62→    62→      .limit(Math.min(limit, 100))
    63→    63→      .offset(offset);
    64→    64→
    65→    65→    const total = await db.select({ count: count() }).from(users);
    66→    66→
    67→    67→    return c.json({
    68→    68→      success: true,
    69→    69→      data: {
    70→    70→        users: userList,
    71→    71→        total: total[0]?.count || 0,
    72→    72→        limit,
    73→    73→        offset,
    74→    74→      },
    75→    75→    });
    76→    76→  } catch (error) {
    77→    77→    console.error('[Admin] Failed to list users:', error);
    78→    78→    return c.json({ success: false, error: 'Failed to list users' }, 500);
    79→    79→  }

    89→    89→  const data = c.req.valid('json');
    90→    90→  const targetUserId = c.req.query('userId');

    94→    94→  }
    95→    95→
    96→    96→  try {
    97→    97→    const result = await db
    98→    98→      .update(users)
    99→    99→      .set({ ...data, updatedAt: new Date() })
   100→   100→      .where(eq(users.id, targetUserId))
   101→   101→      .returning({
   102→   102→        id: users.id,
   103→   103→        email: users.email,
   104→   104→        name: users.name,
   105→   105→        tier: users.tier,
   106→   106→      });
   107→   107→
   108→   108→    if (!result.length) {
   109→   109→      return c.json({ success: false, error: 'User not found' }, 404);
   110→   110→    }
   111→   111→
   112→   112→    // Audit: admin user update — P2.1.1: No secrets in audit
   113→   113→    await auditTrailService.record({
   114→   114→      action: 'admin.user_update',
   115→   115→      actorId: c.get('userId') as string,
   116→   116→      actorType: 'admin',
   117→   117→      resourceType: 'user',
   118→   118→      resourceId: targetUserId,
   119→   119→      details: { updatedFields: Object.keys(data) },
   120→   120→      ipAddress: c.req.header('X-Forwarded-For')?.split(',')[0]?.trim() || 'unknown',
   121→   121→      userAgent: c.req.header('User-Agent') || 'unknown',
   122→   122→      success: true,
   123→   123→      riskLevel: 'medium',
   124→   124→    });
   125→   125→
   126→   126→    return c.json({ success: true, data: result[0] });
   127→   127→  } catch (error) {
   128→   128→    console.error('[Admin] Failed to update user:', error);
   129→   129→    return c.json({ success: false, error: 'Failed to update user' }, 500);

   189→   189→admin.delete('/users/:id', requirePermission({ resource: 'users', action: 'manage' }), async (c) => {
   190→   190→  const userId = c.req.param('id');
   191→   191→
   192→   192→  try {
   193→   193→    // Safety check: don't allow self-deletion
   194→   194→    const actorId = c.get('userId') as string;
   195→   195→    if (actorId === userId) {
   196→   196→      return c.json({ success: false, error: 'Cannot delete your own account' }, 400);
   197→   197→    }
   198→   198→
   199→   199→    const result = await db.delete(users).where(eq(users.id, userId)).returning({ id: users.id });
   200→   200→
   201→   201→    if (!result.length) {
   202→   202→      return c.json({ success: false, error: 'User not found' }, 404);
   203→   203→    }
   204→   204→
   205→   205→    // Audit: admin user deletion
   206→   206→    await auditTrailService.record({
   207→   207→      action: 'admin.user_delete',
   208→   208→      actorId,
   209→   209→      actorType: 'admin',
   210→   210→      resourceType: 'user',
   211→   211→      resourceId: userId,
   212→   212→      details: {},
   213→   213→      ipAddress: c.req.header('X-Forwarded-For')?.split(',')[0]?.trim() || 'unknown',
   214→   214→      userAgent: c.req.header('User-Agent') || 'unknown',
   215→   215→      success: true,
   216→   216→      riskLevel: 'high',
   217→   217→    });
   218→   218→
   219→   219→    return c.json({ success: true, message: 'User deleted' });
   220→   220→  } catch (error) {
